   117→   117→async function unwrap<T>(promise: Promise<unknown>): Promise<T> {
   118→   118→  const res = await promise;
   119→   119→  if (res && typeof res === "object" && "success" in res && "data" in (res as Record<string, unknown>)) {
   120→   120→    return (res as { data: T }).data;
   121→   121→  }
   122→   122→  return res as T;
   123→   123→}

   879→   879→function ChangePasswordDialog({
   880→   880→  open,
   881→   881→  onOpenChange,
   882→   882→  onSuccess,
   883→   883→}: {
   884→   884→  open: boolean;
   885→   885→  onOpenChange: (v: boolean) => void;
   886→   886→  onSuccess: () => void;
   887→   887→}) {
   888→   888→  const isMobile = useIsMobile();
   889→   889→  const [form, setForm] = useState({ currentPassword: "", newPassword: "", confirmPassword: "" });
   890→   890→  const [showCurrent, setShowCurrent] = useState(false);
   891→   891→  const [showNew, setShowNew] = useState(false);
   892→   892→  const [loading, setLoading] = useState(false);
   893→   893→
   894→   894→  const passwordStrength = (() => {
   895→   895→    const p = form.newPassword;
   896→   896→    if (!p) return { score: 0, label: "", color: "" };
   897→   897→    let score = 0;
   898→   898→    if (p.length >= 8) score++;
   899→   899→    if (/[A-Z]/.test(p)) score++;
   900→   900→    if (/[a-z]/.test(p)) score++;
   901→   901→    if (/[0-9]/.test(p)) score++;
   902→   902→    if (/[^A-Za-z0-9]/.test(p)) score++;
   903→   903→    const labels = ["", "Weak", "Fair", "Good", "Strong", "Excellent"];
   904→   904→    const colors = ["", "bg-destructive", "bg-warning", "bg-warning", "bg-success", "bg-success"];
   905→   905→    return { score, label: labels[score], color: colors[score] };
   906→   906→  })();
   907→   907→
   908→   908→  const submit = async () => {
   909→   909→    if (form.newPassword !== form.confirmPassword) {
   910→   910→      toast.error("Passwords don't match");
   911→   911→      return;
   912→   912→    }
   913→   913→    if (form.newPassword.length < 8) {
   914→   914→      toast.error("Password must be at least 8 characters");
   915→   915→      return;
   916→   916→    }
   917→   917→    setLoading(true);
   918→   918→    try {
   919→   919→      await api.post("/auth/change-password", {
   920→   920→        currentPassword: form.currentPassword,
   921→   921→        newPassword: form.newPassword,
   922→   922→      });
   923→   923→      toast.success("Password changed. Other sessions have been signed out.");
   924→   924→      setForm({ currentPassword: "", newPassword: "", confirmPassword: "" });
   925→   925→      onSuccess();
   926→   926→    } catch (e: unknown) {
   927→   927→      toast.error((e as Error).message || "Failed to change password");
   928→   928→    } finally {
   929→   929→      setLoading(false);
   930→   930→    }
   931→   931→  };
   932→   932→
   933→   933→  const content = (
   934→   934→    <div className="space-y-4">
   935→   935→      <GlassInput
   936→   936→        label="Current Password"
   937→   937→        type={showCurrent ? "text" : "password"}
