   173→function formatINR(n: number) {
   174→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   175→}
   176→
   177→function formatMonthYear(month: number, year: number) {
   178→  return `${MONTHS[month] ?? "—"} ${year}`;
   179→}
   180→
   181→function formatDate(iso: string | null) {
   182→  if (!iso) return "—";
   183→  const d = new Date(iso);
   184→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" });
   185→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   186→  return `${datePart}, ${timePart}`;
   187→}
   188→
   189→/** Full date-time format: "29 June 2026, 05:28 am" */
   190→function formatDateTime(iso: string | null) {
   191→  if (!iso) return "—";
   192→  const d = new Date(iso);
   193→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric" });
   194→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   195→  return `${datePart}, ${timePart}`;
   196→}
   197→
   198→/** A bill is overdue if it has a due date in the past AND there's still an outstanding due amount. */
   199→function isOverdue(bill: Bill): boolean {
   200→  if (!bill.dueDate || bill.deletedAt) return false;
   201→  if (bill.status === "PAID" || bill.status === "VOID") return false;
   202→  return new Date(bill.dueDate) < new Date() && bill.dueAmount > 0;
   203→}

   268→    },

   271→    placeholderData: (prev) => prev,
   272→  });
   273→
   274→  // Fetch soft-deleted bills (deletion queue) — admin only
   275→  const { data: deletedBills = [] } = useQuery({
   276→    queryKey: ["bills", "deleted", { month: selectedMonth, year: selectedYear }],

   291→        { month: genMonth, year: genYear, dueDate: genDueDate || undefined }

   985→function KpiCard({
   986→  label,
   987→  value,
   988→  icon,
   989→  color,
   990→  prefix,
   991→  sub,
   992→}: {
   993→  label: string;
   994→  value: number;
   995→  icon: React.ReactNode;
   996→  color: "primary" | "success" | "warning" | "danger" | "info";
   997→  prefix?: string;
   998→  sub?: string;
   999→}) {
  1000→  const colorVar =
  1001→    color === "primary"
  1002→      ? "var(--primary)"
  1003→      : color === "success"
  1004→        ? "var(--success)"
  1005→        : color === "warning"
  1006→          ? "var(--warning)"
  1007→          : color === "danger"
  1008→            ? "var(--destructive)"
  1009→            : "var(--info)";
  1010→  return (
  1011→    <GlassCard
  1012→      className="p-4 relative overflow-hidden"
  1013→      glow={color === "danger" ? "danger" : color === "warning" ? "warning" : color === "success" ? "success" : "primary"}
  1014→      whileHover={{ y: -2 }}
  1015→    >
  1016→      <div
  1017→        className="absolute -top-8 -right-8 h-24 w-24 rounded-full blur-3xl opacity-30"
  1018→        style={{ background: colorVar }}
  1019→      />
  1020→      <div className="relative">
  1021→        <div className="flex items-start justify-between mb-3">
  1022→          <div
  1023→            className="grid place-items-center h-9 w-9 rounded-2xl"
  1024→            style={{
  1025→              background: `color-mix(in oklch, ${colorVar} 18%, transparent)`,
  1026→              color: colorVar,
  1027→            }}
  1028→          >
  1029→            {icon}
  1030→          </div>
  1031→        </div>
  1032→        <p className="text-[11px] text-muted-foreground">{label}</p>
  1033→        <div className="text-2xl font-bold tracking-tight tabular-nums">
  1034→          <AnimatedCounter value={value} prefix={prefix || ""} />
  1035→        </div>
  1036→        {sub && (
  1037→          <p className="text-[10px] text-muted-foreground mt-1">{sub}</p>
  1038→        )}
  1039→      </div>
  1040→    </GlassCard>
  1041→  );
  1042→}

  1044→const BillRow = memo(function BillRow({
  1045→  bill,
  1046→  isAdmin,
  1047→  onView,
  1048→  onVoid,
  1049→  onDelete,
  1050→  onRestore,
  1051→}: {
  1052→  bill: Bill;
  1053→  isAdmin: boolean;
  1054→  onView: () => void;
  1055→  onVoid: () => void;
  1056→  onDelete: () => void;
  1057→  onRestore: () => void;
  1058→}) {
  1059→  const isDeleted = !!bill.deletedAt;
  1060→
  1061→  // Build the actions list — same shape as UserRow's `actions` array.
  1062→  const actions: { label: string; icon: typeof Eye; onClick: () => void; variant?: "destructive" }[] = [];
  1063→  if (isDeleted) {
  1064→    if (isAdmin) {
  1065→      actions.push({ label: "Restore Bill", icon: RotateCcw, onClick: onRestore });
  1066→    }
  1067→  } else {
  1068→    actions.push({ label: "View Details", icon: Eye, onClick: onView });
  1069→    if (isAdmin && bill.status !== "VOID") {
  1070→      actions.push({ label: "Void Bill", icon: Ban, onClick: onVoid, variant: "destructive" });
  1071→    }
  1072→    if (isAdmin) {
  1073→      actions.push({ label: "Delete Bill", icon: Trash2, onClick: onDelete, variant: "destructive" });
  1074→    }
  1075→  }
  1076→
  1077→  return (
  1078→    <GlassCard className="p-4" hover={false}>
  1079→      <div className="flex items-start gap-3">
  1080→        <UserAvatar
  1081→          name={bill.user.name}
  1082→          avatarUrl={bill.user.avatarUrl}
  1083→          className="h-10 w-10 rounded-xl"
  1084→          fallbackClassName="text-xs"
  1085→        />
