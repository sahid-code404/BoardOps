   206→   206→
   207→   207→/** Returns { top, bottom } for the date picker.
   208→   208→ *  - Today/Yesterday/Tomorrow: top = relative label, bottom = "EEE, d MMM"
   209→   209→ *  - Other dates: top = "d MMM" (date only), bottom = "EEE" (day name only, no duplicate) */
   210→   210→function getDatePickerLabels(d: Date): { top: string; bottom: string } {
   211→   211→  const today = new Date();
   212→   212→  today.setHours(0, 0, 0, 0);
   213→   213→  const target = new Date(d);
   214→   214→  target.setHours(0, 0, 0, 0);
   215→   215→  const diffMs = target.getTime() - today.getTime();
   216→   216→  const diffDays = Math.round(diffMs / (24 * 60 * 60 * 1000));
   217→   217→
   218→   218→  if (diffDays === 0) return { top: "Today", bottom: format(target, "EEE, d MMM") };
   219→   219→  if (diffDays === -1) return { top: "Yesterday", bottom: format(target, "EEE, d MMM") };
   220→   220→  if (diffDays === 1) return { top: "Tomorrow", bottom: format(target, "EEE, d MMM") };
   221→   221→  // Far dates: top = date only, bottom = day name only (no duplicate)
   222→   222→  return { top: format(target, "d MMM"), bottom: format(target, "EEE") };
   223→   223→}

   913→   913→function SubmitPaymentDialog({
   914→   914→  open,
   915→   915→  onOpenChange,
   916→   916→  onSubmit,
   917→   917→  loading,
   918→   918→}: {
   919→   919→  open: boolean;
   920→   920→  onOpenChange: (o: boolean) => void;
   921→   921→  onSubmit: (payload: {
   922→   922→    amount: number;
   923→   923→    method: PaymentMethod;
   924→   924→    billId?: string;
   925→   925→    reference?: string;
   926→   926→    notes?: string;
   927→   927→  }) => void;
   928→   928→  loading: boolean;
   929→   929→}) {
   930→   930→  const [amount, setAmount] = useState("");
   931→   931→  const [method, setMethod] = useState<PaymentMethod>("UPI");
   932→   932→  const [billId, setBillId] = useState<string>("NONE");
   933→   933→  const [reference, setReference] = useState("");
   934→   934→  const [notes, setNotes] = useState("");
   935→   935→  const [error, setError] = useState<string>("");
   936→   936→
   937→   937→  // Load user's outstanding bills for selection
   938→   938→  const { data: bills = [] } = useQuery({
   939→   939→    queryKey: ["bills"],
   940→   940→    queryFn: async () => {
   941→   941→      const r = await api.get<ApiResponse<BillListItem[]>>("/bills");
   942→   942→      return r.data;
   943→   943→    },
   944→   944→    enabled: open,
   945→   945→  });
   946→   946→
   947→   947→  const outstanding = bills.filter(
   948→   948→    (b) => b.status !== "PAID" && b.status !== "VOID" && b.dueAmount > 0
   949→   949→  );
   950→   950→
   951→   951→  function reset() {
   952→   952→    setAmount("");
   953→   953→    setMethod("UPI");
   954→   954→    setBillId("NONE");
   955→   955→    setReference("");
   956→   956→    setNotes("");
   957→   957→    setError("");
   958→   958→  }
   959→   959→
   960→   960→  function handleClose(o: boolean) {
   961→   961→    if (!o) reset();

   963→   963→  }
   964→   964→
   965→   965→  function handleSubmit() {
   966→   966→    const amt = parseFloat(amount);
   967→   967→    if (!amt || amt <= 0) {
   968→   968→      setError("Enter a valid amount greater than 0");
