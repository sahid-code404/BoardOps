   184→   184→
   185→   185→export function PaymentsView() {
   186→   186→  const user = useAuthStore((s) => s.user);
   187→   187→  const isAdmin = user?.role === "ADMIN";
   188→   188→  const qc = useQueryClient();
   189→   189→
   190→   190→  const [search, setSearch] = useState("");
   191→   191→  const [statusFilter, setStatusFilter] = useState<PaymentStatus | "ALL">("ALL");
   192→   192→  const [methodFilter, setMethodFilter] = useState<PaymentMethod | "ALL">("ALL");
   193→   193→
   194→   194→  const [submitOpen, setSubmitOpen] = useState(false);
   195→   195→  const [actionTarget, setActionTarget] = useState<{
   196→   196→    payment: Payment;
   197→   197→    action: "APPROVE" | "REJECT";
   198→   198→  } | null>(null);
   199→   199→
   200→   200→  const { data: payments = [], isLoading } = useQuery({
   201→   201→    queryKey: ["payments"],
   202→   202→    queryFn: async () => {
   203→   203→      const r = await api.get<ApiResponse<Payment[]>>("/payments");
   204→   204→      return r.data;
   205→   205→    },
   206→   206→  });
   207→   207→
   208→   208→  const submitMutation = useMutation({
   209→   209→    mutationFn: (payload: {
   210→   210→      amount: number;
   211→   211→      method: PaymentMethod;
   212→   212→      billId?: string;
   213→   213→      reference?: string;
   214→   214→      notes?: string;
   215→   215→    }) =>
   216→   216→      api.post<ApiResponse<Payment>>("/payments", payload),
   217→   217→    onSuccess: () => {
   218→   218→      toast.success("Payment submitted — pending admin approval");
   219→   219→      setSubmitOpen(false);
   220→   220→      qc.invalidateQueries({ queryKey: ["payments"] });
   221→   221→    },
   222→   222→    onError: (e: Error) => toast.error(e.message || "Failed to submit payment"),
   223→   223→  });
   224→   224→
   225→   225→  const actionMutation = useMutation({
   226→   226→    mutationFn: ({
   227→   227→      id,
   228→   228→      action,
   229→   229→    }: {
   230→   230→      id: string;
   231→   231→      action: "APPROVE" | "REJECT";
   232→   232→    }) => api.patch<ApiResponse<Payment>>(`/payments/${id}`, { action }),
   233→   233→    onSuccess: (_data, vars) => {
   234→   234→      toast.success(
   235→   235→        vars.action === "APPROVE"
   236→   236→          ? "Payment approved — bill updated"
   237→   237→          : "Payment rejected"
   238→   238→      );
   239→   239→      setActionTarget(null);
   240→   240→      qc.invalidateQueries({ queryKey: ["payments"] });
   241→   241→      qc.invalidateQueries({ queryKey: ["bills"] });
   242→   242→    },
   243→   243→    onError: (e: Error) => toast.error(e.message || "Action failed"),
   244→   244→  });
   245→   245→
   246→   246→  // KPIs
   247→   247→  const kpis = useMemo(() => {
   248→   248→    const approved = payments.filter((p) => p.status === "APPROVED");
   249→   249→    const totalApproved = approved.reduce((s, p) => s + p.amount, 0);
   250→   250→    const pending = payments.filter((p) => p.status === "PENDING").length;
   251→   251→    const rejected = payments.filter((p) => p.status === "REJECTED").length;
   252→   252→    const now = new Date();
   253→   253→    const monthTotal = approved
   254→   254→      .filter((p) => {
   255→   255→        const d = new Date(p.createdAt);
   256→   256→        return (
   257→   257→          d.getMonth() === now.getMonth() &&
   258→   258→          d.getFullYear() === now.getFullYear()
   259→   259→        );
   260→   260→      })
   261→   261→      .reduce((s, p) => s + p.amount, 0);
   262→   262→    return { totalApproved, pending, rejected, monthTotal };
   263→   263→  }, [payments]);
   264→   264→
   265→   265→  // Filtered list
   266→   266→  const filtered = useMemo(() => {
   267→   267→    const q = search.trim().toLowerCase();
   268→   268→    return payments.filter((p) => {
   269→   269→      if (statusFilter !== "ALL" && p.status !== statusFilter) return false;
   270→   270→      if (methodFilter !== "ALL" && p.method !== methodFilter) return false;
   271→   271→      if (!q) return true;
   272→   272→      return (
   273→   273→        p.user.name?.toLowerCase().includes(q) ||
   274→   274→        p.user.email?.toLowerCase().includes(q) ||
   275→   275→        (p.reference || "").toLowerCase().includes(q)
   276→   276→      );
   277→   277→    });
   278→   278→  }, [payments, search, statusFilter, methodFilter]);
   279→   279→
   280→   280→  const pendingPayments = payments.filter((p) => p.status === "PENDING");
   281→   281→
   282→   282→  if (isLoading) {
   283→   283→    return (
   284→   284→      <div className="space-y-4">
   285→   285→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   286→   286→          {Array.from({ length: 4 }).map((_, i) => (
   287→   287→            <ShimmerSkeleton key={i} className="h-32" />
   288→   288→          ))}
   289→   289→        </div>
   290→   290→        <ShimmerSkeleton className="h-14 w-full" />
   291→   291→        <div className="grid gap-3">
   292→   292→          {Array.from({ length: 5 }).map((_, i) => (
   293→   293→            <ShimmerSkeleton key={i} className="h-20" />
   294→   294→          ))}
   295→   295→        </div>
   296→   296→      </div>
   297→   297→    );
   298→   298→  }
   299→   299→
   300→   300→  return (
   301→   301→    <StaggerGroup className="space-y-4 md:space-y-6">
   302→   302→      {/* Action bar */}
   303→   303→      <StaggerItem>
   304→   304→        <div className="flex items-center justify-end gap-3">
   305→   305→          <p className="text-sm text-muted-foreground hidden sm:block">
   306→   306→            {isAdmin
   307→   307→              ? "Approve payments & track transactions"
   308→   308→              : "Submit payments & track history"}
   309→   309→          </p>
   310→   310→          {!isAdmin && (
   311→   311→            <GlassButton
   312→   312→              size="lg"
   313→   313→              onClick={() => setSubmitOpen(true)}
   314→   314→              className="shrink-0"
   315→   315→            >
   316→   316→              <Plus className="h-4 w-4" />
   317→   317→              Submit Payment
   318→   318→            </GlassButton>
   319→   319→          )}
   320→   320→        </div>
   321→   321→      </StaggerItem>
   322→   322→
   323→   323→      {/* KPIs */}
   324→   324→      <StaggerItem>
   325→   325→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   326→   326→          <KpiCard
   327→   327→            label="Total Approved"
   328→   328→            value={kpis.totalApproved}
   329→   329→            icon={<CheckCircle2 className="h-5 w-5" />}
   330→   330→            color="success"
   331→   331→            prefix="₹"
   332→   332→          />
   333→   333→          <KpiCard
   334→   334→            label="Pending Approvals"
   335→   335→            value={kpis.pending}
   336→   336→            icon={<Clock className="h-5 w-5" />}
   337→   337→            color="warning"
   338→   338→          />
   339→   339→          <KpiCard
   340→   340→            label="Rejected"
   341→   341→            value={kpis.rejected}
   342→   342→            icon={<XCircle className="h-5 w-5" />}
   343→   343→            color="danger"
   344→   344→          />
   345→   345→          <KpiCard
   346→   346→            label="This Month"
   347→   347→            value={kpis.monthTotal}
   348→   348→            icon={<Wallet className="h-5 w-5" />}
   349→   349→            color="info"
   350→   350→            prefix="₹"
   351→   351→          />
   352→   352→        </div>
   353→   353→      </StaggerItem>
   354→   354→
   355→   355→      {/* Admin: Pending payments section */}
   356→   356→      {isAdmin && pendingPayments.length > 0 && (
   357→   357→        <StaggerItem>
   358→   358→          <GlassCard className="p-5 md:p-6" hover={false} glow="warning">
   359→   359→            <div className="flex items-center justify-between mb-4">
   360→   360→              <div>
   361→   361→                <h3 className="font-semibold text-lg flex items-center gap-2">
   362→   362→                  <Clock className="h-5 w-5 text-warning" />
   363→   363→                  Pending Approvals
   364→   364→                  <Badge variant="outline" className="bg-warning/15 text-warning border-warning/30 rounded-full">
   365→   365→                    {pendingPayments.length}
   366→   366→                  </Badge>
   367→   367→                </h3>
   368→   368→                <p className="text-xs text-muted-foreground">
   369→   369→                  Review and approve or reject incoming payments.
   370→   370→                </p>
   371→   371→              </div>
   372→   372→            </div>
   373→   373→            <StaggerGroup className="space-y-2">
   374→   374→              {pendingPayments.slice(0, 4).map((p) => (
   375→   375→                <StaggerItem key={p.id}>
   376→   376→                  <PendingRow
   377→   377→                    payment={p}
   378→   378→                    onApprove={() =>
   379→   379→                      setActionTarget({ payment: p, action: "APPROVE" })
   380→   380→                    }
   381→   381→                    onReject={() =>
   382→   382→                      setActionTarget({ payment: p, action: "REJECT" })
   383→   383→                    }
   384→   384→                    loading={
   385→   385→                      actionMutation.isPending &&
   386→   386→                      actionTarget?.payment.id === p.id
   387→   387→                    }
   388→   388→                  />
   389→   389→                </StaggerItem>
   390→   390→              ))}
   391→   391→            </StaggerGroup>
   392→   392→          </GlassCard>
   393→   393→        </StaggerItem>
   394→   394→      )}
   395→   395→
   396→   396→      {/* Filters */}
   397→   397→      <StaggerItem>
   398→   398→        <GlassCard className="p-3 md:p-4" hover={false}>
   399→   399→          <div className="flex flex-col md:flex-row gap-3">
   400→   400→            <div className="flex-1 min-w-0">
   401→   401→              <GlassInput
   402→   402→                placeholder="Search by name, email, reference…"
   403→   403→                value={search}
   404→   404→                onChange={(e) => setSearch(e.target.value)}
   405→   405→                icon={<Search />}
   406→   406→              />
   407→   407→            </div>
   408→   408→            <div className="flex items-center gap-2 overflow-x-auto no-scrollbar">
   409→   409→              {(
   410→   410→                ["ALL", "APPROVED", "PENDING", "REJECTED", "REFUNDED"] as const
   411→   411→              ).map((s) => {
   412→   412→                const active = statusFilter === s;
   413→   413→                return (
   414→   414→                  <button
   415→   415→                    key={s}
   416→   416→                    onClick={() => setStatusFilter(s)}
   417→   417→                    className={cn(
   418→   418→                      "px-3.5 py-2 text-xs font-medium rounded-full whitespace-nowrap transition-all",
   419→   419→                      active
   420→   420→                        ? "bg-primary text-primary-foreground shadow-md shadow-primary/30"
   421→   421→                        : "glass-soft text-muted-foreground hover:text-foreground"
   422→   422→                    )}
   423→   423→                  >
   424→   424→                    {s === "ALL" ? "All" : STATUS_STYLES[s].label}
   425→   425→                  </button>
   426→   426→                );
   427→   427→              })}
   428→   428→            </div>
   429→   429→            <Select
   430→   430→              value={methodFilter}
   431→   431→              onValueChange={(v) => setMethodFilter(v as PaymentMethod | "ALL")}
   432→   432→            >
   433→   433→              <SelectTrigger className="w-full md:w-40 h-11 rounded-2xl">
   434→   434→                <SelectValue placeholder="Method" />
   435→   435→              </SelectTrigger>
   436→   436→              <SelectContent>
   437→   437→                <SelectItem value="ALL">All Methods</SelectItem>
   438→   438→                {Object.entries(METHOD_META).map(([k, m]) => (
   439→   439→                  <SelectItem key={k} value={k}>
   440→   440→                    {m.label}
   441→   441→                  </SelectItem>
   442→   442→                ))}
   443→   443→              </SelectContent>
   444→   444→            </Select>
   445→   445→          </div>
   446→   446→        </GlassCard>
   447→   447→      </StaggerItem>
   448→   448→
   449→   449→      {/* List */}
   450→   450→      <StaggerItem>
   451→   451→        {filtered.length === 0 ? (
   452→   452→          <GlassCard className="p-10 md:p-16" hover={false}>
   453→   453→            <div className="flex flex-col items-center justify-center text-center gap-3">
   454→   454→              <div className="grid place-items-center h-14 w-14 rounded-3xl bg-muted/40">
   455→   455→                <IndianRupee className="h-6 w-6 text-muted-foreground" />
   456→   456→              </div>
   457→   457→              <div>
   458→   458→                <p className="font-medium">No payments found</p>
   459→   459→                <p className="text-sm text-muted-foreground">
   460→   460→                  Submit a payment or adjust your filters.
   461→   461→                </p>
   462→   462→              </div>
   463→   463→            </div>
   464→   464→          </GlassCard>
   465→   465→        ) : (
   466→   466→          <>
   467→   467→            {/* Mobile cards */}
   468→   468→            <div className="md:hidden space-y-3">
   469→   469→              <StaggerGroup className="space-y-3">
   470→   470→                {filtered.map((p) => (
   471→   471→                  <StaggerItem key={p.id}>
   472→   472→                    <PaymentCard payment={p} isAdmin={isAdmin} />
   473→   473→                  </StaggerItem>
   474→   474→                ))}
   475→   475→              </StaggerGroup>
   476→   476→            </div>
   477→   477→
   478→   478→            {/* Desktop table */}
   479→   479→            <GlassCard className="hidden md:block p-2" hover={false}>
   480→   480→              <Table>
   481→   481→                <TableHeader>
   482→   482→                  <TableRow className="border-border/60">
   483→   483→                    {isAdmin && <TableHead className="pl-4">Resident</TableHead>}
   484→   484→                    <TableHead className={isAdmin ? "" : "pl-4"}>Amount</TableHead>
   485→   485→                    <TableHead>Method</TableHead>
   486→   486→                    <TableHead>Status</TableHead>
   487→   487→                    <TableHead>Reference</TableHead>
   488→   488→                    <TableHead>Date</TableHead>
   489→   489→                    {!isAdmin && <TableHead className="text-right pr-4">Notes</TableHead>}
   490→   490→                  </TableRow>
   491→   491→                </TableHeader>
   492→   492→                <TableBody>
   493→   493→                  {filtered.map((p) => (
   494→   494→                    <TableRow key={p.id} className="border-border/40">
   495→   495→                      {isAdmin && (
   496→   496→                        <TableCell className="pl-4">
