   230→/** Returns { top, bottom } for the date picker.
   231→ *  - Today/Yesterday/Tomorrow: top = relative label, bottom = "EEE, d MMM"
   232→ *  - Other dates: top = "d MMM" (date only), bottom = "EEE" (day name only, no duplicate) */
   233→function getDatePickerLabels(d: Date): { top: string; bottom: string } {
   234→  const today = new Date();
   235→  today.setHours(0, 0, 0, 0);
   236→  const target = new Date(d);
   237→  target.setHours(0, 0, 0, 0);
   238→  const diffMs = target.getTime() - today.getTime();
   239→  const diffDays = Math.round(diffMs / (24 * 60 * 60 * 1000));
   240→
   241→  if (diffDays === 0) return { top: "Today", bottom: format(target, "EEE, d MMM") };
   242→  if (diffDays === -1) return { top: "Yesterday", bottom: format(target, "EEE, d MMM") };
   243→  if (diffDays === 1) return { top: "Tomorrow", bottom: format(target, "EEE, d MMM") };
   244→  // Far dates: top = date only, bottom = day name only (no duplicate)
   245→  return { top: format(target, "d MMM"), bottom: format(target, "EEE") };
   246→}

   910→        </AlertDialogContent>

   912→    </StaggerGroup>
   913→  );

  1031→function PaymentRow({
  1032→  payment,
  1033→  isAdmin,
  1034→  onApprove,
  1035→  onReject,
  1036→  onEdit,
  1037→  onDelete,
  1038→  onVoid,
  1039→  onRestore,
  1040→}: {
  1041→  payment: Payment;
  1042→  isAdmin: boolean;
  1043→  onApprove: () => void;
  1044→  onReject: () => void;
  1045→  onEdit: () => void;
  1046→  onDelete: () => void;
  1047→  onVoid: () => void;
  1048→  onRestore: () => void;
  1049→}) {
  1050→  const isDeleted = !!payment.deletedAt;
  1051→
  1052→  // Build the actions list — admin-only. Deleted rows show Restore; otherwise
  1053→  // PENDING rows get Approve/Reject, non-VOID rows get Edit/Void, all get Delete.
  1054→  const actions: {
  1055→    label: string;
  1056→    icon: typeof CheckCircle2;
  1057→    onClick: () => void;
  1058→    variant?: "destructive";
  1059→  }[] = [];
  1060→
  1061→  if (isDeleted) {
  1062→    if (isAdmin) {
  1063→      actions.push({ label: "Restore Payment", icon: RotateCcw, onClick: onRestore });
  1064→    }
  1065→  } else {
  1066→    if (isAdmin && payment.status === "PENDING") {
  1067→      actions.push({ label: "Approve Payment", icon: CheckCircle2, onClick: onApprove });
  1068→      actions.push({
  1069→        label: "Reject Payment",
  1070→        icon: XCircle,
  1071→        onClick: onReject,
  1072→        variant: "destructive",
  1073→      });
  1074→    }
  1075→    if (isAdmin && payment.status !== "VOID") {
  1076→      actions.push({ label: "Edit Payment", icon: PencilLine, onClick: onEdit });
  1077→      actions.push({
  1078→        label: "Void Payment",
  1079→        icon: Ban,
  1080→        onClick: onVoid,
  1081→        variant: "destructive",
  1082→      });
  1083→    }
  1084→    if (isAdmin) {
  1085→      actions.push({
  1086→        label: "Delete Payment",
  1087→        icon: Trash2,
  1088→        onClick: onDelete,
  1089→        variant: "destructive",
  1090→      });
  1091→    }
  1092→  }
  1093→
  1094→  const methodMeta = METHOD_META[payment.method];
  1095→  const statusMeta = STATUS_STYLES[payment.status];
  1096→
  1097→  return (
  1098→    <GlassCard className="p-4 md:p-5" hover={false}>
  1099→      <div className="flex items-start gap-3 md:gap-4">
  1100→        <Avatar className="h-12 w-12 md:h-14 md:w-14 rounded-2xl shrink-0">
  1101→          <AvatarFallback
  1102→            className={cn(
  1103→              "rounded-2xl bg-gradient-to-br text-white font-semibold",
  1104→              gradientFor(payment.user.name)
  1105→            )}
  1106→          >
  1107→            {initials(payment.user.name) || "U"}
  1108→          </AvatarFallback>
  1109→        </Avatar>
  1110→
  1111→        <div className="flex-1 min-w-0">
  1112→          <div className="flex items-start justify-between gap-2">
  1113→            <div className="min-w-0">
  1114→              <div className="flex items-center gap-2 flex-wrap">
  1115→                <h3
  1116→                  className={cn(
  1117→                    "font-semibold truncate",
  1118→                    isDeleted && "text-muted-foreground line-through"
  1119→                  )}
  1120→                >
  1121→                  {isAdmin ? payment.user.name : methodMeta.label}
  1122→                </h3>
  1123→                {isDeleted ? (
  1124→                  <Badge
  1125→                    variant="outline"
  1126→                    className="text-[10px] bg-destructive/15 text-destructive border-destructive/30"
  1127→                  >
  1128→                    <Clock className="h-2.5 w-2.5" />{" "}
  1129→                    {formatDeletionCountdown(new Date(payment.deletedAt!))}
  1130→                  </Badge>
  1131→                ) : (
  1132→                  <>
  1133→                    <Badge
  1134→                      variant="outline"
  1135→                      className={cn("text-[10px]", statusMeta.className)}
  1136→                    >
  1137→                      {statusMeta.label}
  1138→                    </Badge>
  1139→                    <Badge
  1140→                      variant="outline"
  1141→                      className={cn("text-[10px]", methodMeta.className)}
  1142→                    >
  1143→                      {methodMeta.label}
  1144→                    </Badge>
  1145→                  </>
  1146→                )}
  1147→              </div>
  1148→              <div className="flex flex-col sm:flex-row sm:items-center gap-x-3 gap-y-0.5 mt-1.5 text-xs text-muted-foreground">
  1149→                {isAdmin && (
  1150→                  <span className="inline-flex items-center gap-1 truncate">
  1151→                    <Mail className="h-3 w-3" /> {payment.user.email}
  1152→                  </span>
  1153→                )}
  1154→                <span className="inline-flex items-center gap-1">
  1155→                  <Clock className="h-3 w-3" /> {formatDateTime(payment.createdAt)}
  1156→                </span>
  1157→                {payment.reference && (
  1158→                  <span className="inline-flex items-center gap-1 truncate">
  1159→                    <ArrowUpRight className="h-3 w-3" /> Ref {payment.reference}
  1160→                  </span>
  1161→                )}
  1162→              </div>
  1163→              {/* Amount inline */}
  1164→              <div className="flex flex-wrap items-center gap-x-3 gap-y-0.5 mt-1.5 text-[11px] text-muted-foreground">
  1165→                <span>
  1166→                  Amount{" "}
  1167→                  <span className="font-semibold text-foreground tabular-nums">
  1168→                    {formatINR(payment.amount)}

  1457→function PaymentEditBody({
  1458→  payment,
  1459→  onSubmit,
  1460→  loading,
  1461→  onCancel,
  1462→}: {
  1463→  payment: Payment | null;
  1464→  onSubmit: (
  1465→    id: string,
  1466→    payload: {
  1467→      amount?: number;
  1468→      method?: PaymentMethod;
  1469→      reference?: string | null;
  1470→      notes?: string | null;
  1471→    }
  1472→  ) => void;
  1473→  loading: boolean;
  1474→  onCancel: () => void;
  1475→}) {
  1476→  const [amount, setAmount] = useState(payment ? String(payment.amount) : "");
  1477→  const [method, setMethod] = useState<PaymentMethod>(payment?.method ?? "UPI");
  1478→  const [reference, setReference] = useState(payment?.reference ?? "");
  1479→  const [notes, setNotes] = useState(payment?.notes ?? "");
  1480→  const [errors, setErrors] = useState<Record<string, string>>({});
  1481→
  1482→  // Cannot edit amount on APPROVED + bill-linked payments — backend will
  1483→  // reject with 422. Lock the field and explain why; admin must void + resubmit.
  1484→  const amountLocked = payment?.status === "APPROVED" && !!payment?.billId;
  1485→
  1486→  function handleSubmit() {
  1487→    if (!payment) return;
  1488→    const next: Record<string, string> = {};
  1489→    const amt = parseFloat(amount);
  1490→    if (!amountLocked) {
  1491→      if (!amt || amt <= 0) next.amount = "Enter a valid amount";
  1492→    }
  1493→    setErrors(next);
  1494→    if (Object.keys(next).length > 0) return;
  1495→
  1496→    const payload: {
  1497→      amount?: number;
  1498→      method?: PaymentMethod;
  1499→      reference?: string | null;
  1500→      notes?: string | null;
  1501→    } = {
  1502→      method,
  1503→      reference: reference.trim() || null,
  1504→      notes: notes.trim() || null,
  1505→    };
  1506→    if (!amountLocked) payload.amount = amt;
  1507→    onSubmit(payment.id, payload);
  1508→  }
  1509→
  1510→  return (
  1511→    <>
  1512→      <SheetHeader className="px-6 pt-6 pb-2">
  1513→        <SheetTitle className="flex items-center gap-2 text-xl">
  1514→          <PencilLine className="h-5 w-5 text-primary" />
  1515→          Edit Payment
  1516→        </SheetTitle>
  1517→        <SheetDescription>
  1518→          Update the details of this payment from {payment?.user.name}.
  1519→        </SheetDescription>
  1520→      </SheetHeader>
  1521→
  1522→      <div className="flex-1 overflow-y-auto px-6 py-4 space-y-3 no-scrollbar">
  1523→        <GlassInput
  1524→          label="Amount (₹)"
  1525→          type="number"
  1526→          inputMode="decimal"
  1527→          placeholder="0.00"
  1528→          value={amount}
  1529→          onChange={(e) => setAmount(e.target.value)}
  1530→          error={errors.amount}
  1531→          icon={<IndianRupee />}
  1532→          disabled={amountLocked}
  1533→          hint={
  1534→            amountLocked
  1535→              ? "Amount locked — this approved payment is linked to a bill. Void it and submit a new payment to change the amount."
  1536→              : undefined
  1537→          }
  1538→        />
  1539→
  1540→        <div className="space-y-1.5">
  1541→          <label className="text-xs font-medium text-muted-foreground ml-1">
  1542→            Method
  1543→          </label>
  1544→          <Select
  1545→            value={method}
  1546→            onValueChange={(v) => setMethod(v as PaymentMethod)}
  1547→          >
  1548→            <SelectTrigger className="w-full h-11 rounded-2xl">
  1549→              <SelectValue />
  1550→            </SelectTrigger>
  1551→            <SelectContent>
  1552→              {Object.entries(METHOD_META).map(([k, m]) => (
  1553→                <SelectItem key={k} value={k}>
  1554→                  {m.label}
  1555→                </SelectItem>
  1556→              ))}
  1557→            </SelectContent>
  1558→          </Select>
  1559→        </div>
  1560→
  1561→        <GlassInput
  1562→          label="Reference (optional)"
  1563→          placeholder="UTR / Txn ID"

  1565→          onChange={(e) => setReference(e.target.value)}
  1566→          icon={<ArrowUpRight />}
