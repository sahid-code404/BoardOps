   207→/** Returns { top, bottom } for the date picker.
   208→ *  - Today/Yesterday/Tomorrow: top = relative label, bottom = "EEE, d MMM"
   209→ *  - Other dates: top = "d MMM" (date only), bottom = "EEE" (day name only, no duplicate) */
   210→function getDatePickerLabels(d: Date): { top: string; bottom: string } {
   211→  const today = new Date();
   212→  today.setHours(0, 0, 0, 0);
   213→  const target = new Date(d);
   214→  target.setHours(0, 0, 0, 0);
   215→  const diffMs = target.getTime() - today.getTime();
   216→  const diffDays = Math.round(diffMs / (24 * 60 * 60 * 1000));
   217→
   218→  if (diffDays === 0) return { top: "Today", bottom: format(target, "EEE, d MMM") };
   219→  if (diffDays === -1) return { top: "Yesterday", bottom: format(target, "EEE, d MMM") };
   220→  if (diffDays === 1) return { top: "Tomorrow", bottom: format(target, "EEE, d MMM") };
   221→  // Far dates: top = date only, bottom = day name only (no duplicate)
   222→  return { top: format(target, "d MMM"), bottom: format(target, "EEE") };
   223→}

   577→        onOpenChange={setSubmitOpen}
   578→        onSubmit={(payload) => submitMutation.mutate(payload)}
   579→        loading={submitMutation.isPending}
   580→      />
   581→
   582→      {/* Approve/Reject confirm */}
   583→      <AlertDialog
   584→        open={!!actionTarget}
   585→        onOpenChange={(o) => !o && setActionTarget(null)}
   586→      >
   587→        <AlertDialogContent className="rounded-3xl">
   588→          <AlertDialogHeader>
   589→            <AlertDialogTitle>
   590→              {actionTarget?.action === "APPROVE"
   591→                ? "Approve payment?"

   594→            <AlertDialogDescription>
   595→              {actionTarget && (
   596→                <>
   597→                  {actionTarget.action === "APPROVE" ? (
   598→                    <>
   599→                      Approve{" "}
   600→                      <span className="font-medium text-foreground">
   601→                        {formatINR(actionTarget.payment.amount)}
   602→                      </span>{" "}
   603→                      from{" "}
   604→                      <span className="font-medium text-foreground">
   605→                        {actionTarget.payment.user.name}
   606→                      </span>
   607→                      ? The linked bill will be updated.
   608→                    </>
   609→                  ) : (
   610→                    <>
   611→                      Reject the{" "}
   612→                      <span className="font-medium text-foreground">
   613→                        {formatINR(actionTarget.payment.amount)}
   614→                      </span>{" "}
   615→                      payment from{" "}
   616→                      <span className="font-medium text-foreground">
   617→                        {actionTarget.payment.user.name}
   618→                      </span>
   619→                      ?
   620→                    </>
   621→                  )}
   622→                </>
   623→              )}
   624→            </AlertDialogDescription>
   625→          </AlertDialogHeader>
   626→          <AlertDialogFooter>
   627→            <AlertDialogCancel className="rounded-2xl">Cancel</AlertDialogCancel>
   628→            <AlertDialogAction
   629→              className={cn(
   630→                "rounded-2xl",
   631→                actionTarget?.action === "APPROVE"
   632→                  ? "bg-success text-white hover:bg-success/90"
   633→                  : "bg-destructive text-white hover:bg-destructive/90"
   634→              )}
   635→              onClick={() =>
   636→                actionTarget &&
   637→                actionMutation.mutate({
   638→                  id: actionTarget.payment.id,
   639→                  action: actionTarget.action,
   640→                })
   641→              }
   642→              disabled={actionMutation.isPending}
   643→            >
   644→              {actionMutation.isPending
   645→                ? "Processing…"
   646→                : actionTarget?.action === "APPROVE"
   647→                  ? "Approve"
   648→                  : "Reject"}
   649→            </AlertDialogAction>
   650→          </AlertDialogFooter>
   651→        </AlertDialogContent>
   652→      </AlertDialog>
   653→    </StaggerGroup>
   654→  );
   655→}

   913→function SubmitPaymentDialog({
   914→  open,
   915→  onOpenChange,
   916→  onSubmit,
   917→  loading,
   918→}: {
   919→  open: boolean;
   920→  onOpenChange: (o: boolean) => void;
   921→  onSubmit: (payload: {
   922→    amount: number;
   923→    method: PaymentMethod;
   924→    billId?: string;
   925→    reference?: string;
   926→    notes?: string;
   927→  }) => void;
   928→  loading: boolean;
   929→}) {
   930→  const [amount, setAmount] = useState("");
   931→  const [method, setMethod] = useState<PaymentMethod>("UPI");
   932→  const [billId, setBillId] = useState<string>("NONE");
   933→  const [reference, setReference] = useState("");
   934→  const [notes, setNotes] = useState("");
   935→  const [error, setError] = useState<string>("");
   936→
   937→  // Load user's outstanding bills for selection
   938→  const { data: bills = [] } = useQuery({
   939→    queryKey: ["bills"],
   940→    queryFn: async () => {
   941→      const r = await api.get<ApiResponse<BillListItem[]>>("/bills");
   942→      return r.data;
   943→    },
   944→    enabled: open,
   945→  });
   946→
   947→  const outstanding = bills.filter(
   948→    (b) => b.status !== "PAID" && b.status !== "VOID" && b.dueAmount > 0
   949→  );
   950→
   951→  function reset() {
   952→    setAmount("");
   953→    setMethod("UPI");
   954→    setBillId("NONE");
   955→    setReference("");
   956→    setNotes("");
   957→    setError("");
   958→  }
   959→
   960→  function handleClose(o: boolean) {
   961→    if (!o) reset();
   962→    onOpenChange(o);
   963→  }
   964→
   965→  function handleSubmit() {
   966→    const amt = parseFloat(amount);
   967→    if (!amt || amt <= 0) {
   968→      setError("Enter a valid amount greater than 0");
   969→      return;
   970→    }
   971→    onSubmit({
   972→      amount: amt,
   973→      method,
   974→      billId: billId === "NONE" ? undefined : billId,
   975→      reference: reference.trim() || undefined,
   976→      notes: notes.trim() || undefined,
   977→    });
   978→  }
   979→
   980→  return (
   981→    <Dialog open={open} onOpenChange={handleClose}>
   982→      <DialogContent className="rounded-3xl max-w-md">
   983→        <DialogHeader>
   984→          <DialogTitle className="flex items-center gap-2">
   985→            <Wallet className="h-5 w-5 text-primary" />
   986→            Submit a Payment
   987→          </DialogTitle>
   988→          <DialogDescription>
   989→            Your payment will be reviewed by an administrator before being
   990→            applied to your bill.
   991→          </DialogDescription>
   992→        </DialogHeader>
   993→
   994→        <div className="space-y-3">
   995→          <GlassInput
   996→            label="Amount (₹)"
   997→            type="number"
   998→            inputMode="decimal"
   999→            placeholder="0.00"
  1000→            value={amount}
  1001→            onChange={(e) => {
  1002→              setAmount(e.target.value);
  1003→              setError("");
  1004→            }}
  1005→            error={error}
  1006→            icon={<IndianRupee />}
  1007→          />
  1008→
  1009→          <div className="space-y-1.5">
  1010→            <label className="text-xs font-medium text-muted-foreground ml-1">
  1011→              Method
  1012→            </label>
  1013→            <Select
  1014→              value={method}
  1015→              onValueChange={(v) => setMethod(v as PaymentMethod)}
  1016→            >
  1017→              <SelectTrigger className="w-full h-11 rounded-2xl">
  1018→                <SelectValue />
  1019→              </SelectTrigger>
  1020→              <SelectContent>
  1021→                {Object.entries(METHOD_META).map(([k, m]) => (
  1022→                  <SelectItem key={k} value={k}>
  1023→                    {m.label}
  1024→                  </SelectItem>
  1025→                ))}
  1026→              </SelectContent>
  1027→            </Select>
  1028→          </div>
  1029→
  1030→          {outstanding.length > 0 && (
  1031→            <div className="space-y-1.5">
  1032→              <label className="text-xs font-medium text-muted-foreground ml-1">
  1033→                Apply to bill (optional)
  1034→              </label>
  1035→              <Select value={billId} onValueChange={setBillId}>
  1036→                <SelectTrigger className="w-full h-11 rounded-2xl">
  1037→                  <SelectValue placeholder="Select a bill" />
  1038→                </SelectTrigger>
  1039→                <SelectContent>
  1040→                  <SelectItem value="NONE">No specific bill</SelectItem>
  1041→                  {outstanding.map((b) => (
  1042→                    <SelectItem key={b.id} value={b.id}>
  1043→                      {b.user.name} — {formatMonthLabel(b.periodMonth, b.periodYear)} · ₹
  1044→                      {Math.round(b.dueAmount).toLocaleString("en-IN")} due
  1045→                    </SelectItem>
  1046→                  ))}
  1047→                </SelectContent>
  1048→              </Select>
  1049→            </div>
  1050→          )}
  1051→
  1052→          <GlassInput
  1053→            label="Reference (optional)"
  1054→            placeholder="UTR / Txn ID"
  1055→            value={reference}
  1056→            onChange={(e) => setReference(e.target.value)}
  1057→            icon={<ArrowUpRight />}
  1058→          />
  1059→
  1060→          <GlassTextarea
  1061→            label="Notes (optional)"
  1062→            placeholder="Any note for the admin…"
  1063→            value={notes}
  1064→            onChange={(e) => setNotes(e.target.value)}
  1065→            rows={2}
  1066→          />
  1067→        </div>
  1068→
  1069→        <DialogFooter>
  1070→          <GlassButton variant="ghost" onClick={() => handleClose(false)}>
  1071→            Cancel
  1072→          </GlassButton>
  1073→          <GlassButton onClick={handleSubmit} loading={loading}>
  1074→            <ArrowDownRight className="h-4 w-4" />
  1075→            Submit
  1076→          </GlassButton>
  1077→        </DialogFooter>
  1078→      </DialogContent>
  1079→    </Dialog>
  1080→  );
  1081→}
  1082→
  1083→function formatMonthLabel(month: number, year: number) {
  1084→  return new Date(year, month, 1).toLocaleDateString("en-IN", {
  1085→    month: "short",
  1086→    year: "numeric",
  1087→  });
  1088→}
