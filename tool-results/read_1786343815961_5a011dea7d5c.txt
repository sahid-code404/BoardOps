   101→function toDateString(d: Date): string {
   102→  const y = d.getFullYear();
   103→  const m = String(d.getMonth() + 1).padStart(2, "0");
   104→  const day = String(d.getDate()).padStart(2, "0");
   105→  return `${y}-${m}-${day}`;
   106→}
   107→
   108→/** Returns { top, bottom } for the date picker.
   109→ *  - Today/Yesterday/Tomorrow: top = relative label, bottom = "EEE, d MMM"
   110→ *  - Other dates: top = "d MMM" (date only), bottom = "EEE" (day name only, no duplicate) */
   111→function getDatePickerLabels(d: Date): { top: string; bottom: string } {
   112→  const today = new Date();
   113→  today.setHours(0, 0, 0, 0);
   114→  const target = new Date(d);
   115→  target.setHours(0, 0, 0, 0);
   116→  const diffMs = target.getTime() - today.getTime();
   117→  const diffDays = Math.round(diffMs / (24 * 60 * 60 * 1000));
   118→
   119→  if (diffDays === 0) return { top: "Today", bottom: format(target, "EEE, d MMM") };
   120→  if (diffDays === -1) return { top: "Yesterday", bottom: format(target, "EEE, d MMM") };
   121→  if (diffDays === 1) return { top: "Tomorrow", bottom: format(target, "EEE, d MMM") };
   122→  // Far dates: top = date only, bottom = day name only (no duplicate)
   123→  return { top: format(target, "d MMM"), bottom: format(target, "EEE") };
   124→}

   211→          u.email.toLowerCase().includes(q)

   215→  }, [userMealStatus, mealSearch]);
   216→

   351→                  ) : (
   352→                    filteredUsers.map((u) => {
   353→                      const isExpanded = expandedUser === u.userId;
   354→                      return (
   355→                        <GlassCard key={u.userId} className="overflow-hidden" hover={false}>

   362→                            <UserAvatar

   374→                            </div>
   375→                            {/* Before Enrollment badge — shown when the selected date

   377→                            {u.notEnrolled && (

   379→                                <UserPlus className="h-2.5 w-2.5" /> Before Enrollment
   380→                              </span>
   381→                            )}
   382→                            {/* Expand chevron */}
   383→                            <motion.div animate={{ rotate: isExpanded ? 180 : 0 }} transition={{ duration: 0.2 }}>
   384→                              <ChevronDown className="h-4 w-4 text-muted-foreground shrink-0" />

   388→                          {/* Expanded — meal details with override controls */}
   389→                          <AnimatePresence initial={false}>
   390→                            {isExpanded && (
   391→                              <motion.div
   392→                                initial={{ height: 0, opacity: 0 }}

   544→function KpiCard({
   545→  icon,
   546→  label,
   547→  value,
   548→  color,
   549→  sub,
   550→}: {
   551→  icon: React.ReactNode;
   552→  label: string;

   558→    color === "primary"

   560→      : color === "success"
   561→        ? "var(--success)"
   562→        : color === "warning"
   563→          ? "var(--warning)"
   564→          : color === "danger"

   568→    <GlassCard
   569→      className="p-4 relative overflow-hidden"

   571→      whileHover={{ y: -2 }}
   572→    >
   573→      <div
   574→        className="absolute -top-8 -right-8 h-24 w-24 rounded-full blur-3xl opacity-30"

   576→      />
   577→      <div className="relative">
   578→        <div className="flex items-start justify-between mb-3">
   579→          <div
   580→            className="grid place-items-center h-9 w-9 rounded-2xl"
   581→            style={{
   582→              background: `color-mix(in oklch, ${colorVar} 18%, transparent)`,
   583→              color: colorVar,
   584→            }}
   585→          >
   586→            {icon}
   587→          </div>
   588→        </div>
   589→        <p className="text-[11px] text-muted-foreground">{label}</p>
   590→        <div className="text-2xl font-bold tracking-tight tabular-nums">
   591→          <AnimatedCounter value={value} />
