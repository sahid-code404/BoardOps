   103→function gradientFor(name: string) {
   104→  const idx = name.split("").reduce((acc, c) => acc + c.charCodeAt(0), 0) % AVATAR_GRADIENTS.length;
   105→  return AVATAR_GRADIENTS[idx];

   108→function initials(name: string) {
   109→  return name
   110→    .split(" ")
   111→    .filter(Boolean)
   112→    .slice(0, 2)
   113→    .map((p) => p[0]?.toUpperCase() ?? "")
   114→    .join("");
   115→}

   117→async function unwrap<T>(promise: Promise<unknown>): Promise<T> {
   118→  const res = await promise;
   119→  if (res && typeof res === "object" && "success" in res && "data" in (res as Record<string, unknown>)) {
   120→    return (res as { data: T }).data;
   121→  }
   122→  return res as T;
   123→}

   500→function AvatarUpload({
   501→  avatarUrl,
   502→  name,
   503→  onUpload,
   504→  loading,
   505→}: {
   506→  avatarUrl?: string;
   507→  name: string;
   508→  onUpload: (file: File) => void;
   509→  loading: boolean;
   510→}) {
   511→  const inputRef = useRef<HTMLInputElement>(null);
   512→
   513→  const handleFile = (file: File) => {
   514→    if (!file.type.startsWith("image/")) {
   515→      toast.error("Please select an image file");
   516→      return;
   517→    }
   518→    if (file.size > 4 * 1024 * 1024) {
   519→      toast.error("Image must be under 4 MB");
   520→      return;
   521→    }
   522→    onUpload(file);
   523→  };
   524→
   525→  return (
   526→    <motion.div
   527→      initial={{ scale: 0.85, opacity: 0 }}
   528→      animate={{ scale: 1, opacity: 1 }}
   529→      transition={{ type: "spring", stiffness: 220, damping: 18 }}
   530→      className="relative shrink-0 mx-auto md:mx-0"
   531→    >
   532→      <div className="absolute -inset-1 rounded-3xl bg-gradient-to-br from-primary/40 to-success/40 blur-md" />
   533→      <Avatar className="relative h-24 w-24 md:h-28 md:w-28 rounded-3xl">
   534→        {avatarUrl && <AvatarImage src={avatarUrl} alt={name} />}
   535→        <AvatarFallback
   536→          className={cn(
   537→            "rounded-3xl bg-gradient-to-br text-white font-bold text-2xl md:text-3xl",
   538→            gradientFor(name)
   539→          )}
   540→        >
   541→          {initials(name) || "U"}
   542→        </AvatarFallback>
   543→      </Avatar>
   544→      <input
   545→        ref={inputRef}
   546→        type="file"
   547→        accept="image/jpeg,image/png,image/webp,image/gif"
   548→        className="hidden"
   549→        onChange={(e) => {
   550→          const file = e.target.files?.[0];
   551→          if (file) handleFile(file);
   552→          e.target.value = "";
   553→        }}
   554→      />
   555→      <button
   556→        onClick={() => inputRef.current?.click()}
   557→        disabled={loading}
   558→        className="absolute -bottom-1 -right-1 h-9 w-9 rounded-full glass-strong grid place-items-center ring-2 ring-background hover:scale-110 transition-transform disabled:opacity-50"
   559→        aria-label="Upload avatar"
   560→      >
   561→        {loading ? (
   562→          <Loader2 className="h-4 w-4 animate-spin text-primary" />
   563→        ) : (
   564→          <Camera className="h-4 w-4 text-primary" />
   565→        )}
   566→      </button>
   567→    </motion.div>
   568→  );

   879→function ChangePasswordDialog({

   881→  onOpenChange,

   892→  const [loading, setLoading] = useState(false);
   893→
   894→  const passwordStrength = (() => {
   895→    const p = form.newPassword;
   896→    if (!p) return { score: 0, label: "", color: "" };
   897→    let score = 0;
   898→    if (p.length >= 8) score++;
   899→    if (/[A-Z]/.test(p)) score++;
   900→    if (/[a-z]/.test(p)) score++;
   901→    if (/[0-9]/.test(p)) score++;
   902→    if (/[^A-Za-z0-9]/.test(p)) score++;
   903→    const labels = ["", "Weak", "Fair", "Good", "Strong", "Excellent"];
   904→    const colors = ["", "bg-destructive", "bg-warning", "bg-warning", "bg-success", "bg-success"];

   909→    if (form.newPassword !== form.confirmPassword) {
   910→      toast.error("Passwords don't match");

   914→      toast.error("Password must be at least 8 characters");
   915→      return;
   916→    }

   919→      await api.post("/auth/change-password", {
   920→        currentPassword: form.currentPassword,
   921→        newPassword: form.newPassword,

  1037→function TwoFactorDialog({

  1050→  const [secret, setSecret] = useState("");
  1051→  const [qrCode, setQrCode] = useState("");
  1052→  const [code, setCode] = useState("");
  1053→  const [backupCodes, setBackupCodes] = useState<string[]>([]);

  1058→    if (open) {
  1059→      setStep("main");
  1060→      setSecret("");
  1061→      setQrCode("");
  1062→      setCode("");
  1063→      setBackupCodes([]);
  1064→      setDisablePassword("");

  1066→  }, [open]);

  1069→    setLoading(true);

  1071→      const res = await api.post<{ success: boolean; data: { secret: string; qrCode: string } }>("/auth/2fa/setup");

  1075→    } catch (e: unknown) {
  1076→      toast.error((e as Error).message || "Failed to start 2FA setup");
  1077→    } finally {
  1078→      setLoading(false);
  1079→    }

  1090→        secret,
  1091→        code,
  1092→      });

  1094→      setStep("backup");

  1104→    setLoading(true);

  1108→      toast.success("Two-factor authentication disabled");

  1110→    } catch (e: unknown) {

  1123→    const blob = new Blob([`BoardOps Backup Codes\n\n${backupCodes.join("\n")}\n\nKeep these safe. Each code can only be used once.`], { type: "text/plain" });

  1126→    a.href = url;
  1127→    a.download = "boardops-backup-codes.txt";
  1128→    a.click();
  1129→    URL.revokeObjectURL(url);
  1130→    toast.success("Backup codes downloaded");
  1131→  };
  1132→
  1133→  const content = (

  1297→  // For backup code regeneration (when already enabled)

  1342→
  1343→function BackupRegenerator({
  1344→  onDone,
  1345→  onCancel,
  1346→}: {
  1347→  onDone: (codes: string[]) => void;
  1348→  onCancel: () => void;
  1349→}) {
  1350→  const [code, setCode] = useState("");
  1351→  const [loading, setLoading] = useState(false);
  1352→
  1353→  const regenerate = async () => {
  1354→    if (!/^\d{6}$/.test(code)) {
  1355→      toast.error("Enter the 6-digit code from your authenticator");
  1356→      return;
  1357→    }
  1358→    setLoading(true);
  1359→    try {
  1360→      const res = await api.post<{ success: boolean; data: { backupCodes: string[] } }>("/auth/2fa/backup-codes", { code });
  1361→      onDone(res.data.backupCodes);
  1362→      toast.success("New backup codes generated");
  1363→    } catch (e: unknown) {
  1364→      toast.error((e as Error).message || "Invalid code");
  1365→    } finally {
  1366→      setLoading(false);
  1367→    }
  1368→  };
  1369→
  1370→  return (
  1371→    <div className="space-y-4">
  1372→      <div className="text-center">
  1373→        <div className="grid place-items-center h-14 w-14 rounded-3xl bg-primary/15 text-primary mx-auto mb-3">
  1374→          <RefreshCw className="h-7 w-7" />
  1375→        </div>
  1376→        <h3 className="font-semibold text-lg">Regenerate Backup Codes</h3>
  1377→        <p className="text-xs text-muted-foreground mt-1">
  1378→          Enter a code from your authenticator app to generate fresh backup codes.
  1379→        </p>
  1380→      </div>
  1381→      <GlassInput
  1382→        label="6-digit code"
  1383→        value={code}
  1384→        onChange={(e) => setCode(e.target.value.replace(/\D/g, "").slice(0, 6))}
  1385→        placeholder="000000"
  1386→        icon={<KeyRound />}
  1387→        className="text-center text-lg tracking-[0.5em] font-mono"
  1388→      />
  1389→      <div className="flex gap-2">
  1390→        <GlassButton variant="ghost" className="flex-1" onClick={onCancel}>
  1391→          Cancel
  1392→        </GlassButton>
  1393→        <GlassButton className="flex-1" loading={loading} onClick={regenerate} disabled={code.length !== 6}>
  1394→          <RefreshCw className="h-4 w-4" />
  1395→          Generate
  1396→        </GlassButton>
  1397→      </div>
  1398→    </div>
  1399→  );
  1400→}
