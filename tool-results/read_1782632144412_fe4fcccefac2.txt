   149→   149→function BillStatusBadge({ status }: { status: BillStatus }) {
   150→   150→  const s = BILL_STATUS_STYLES[status];

   152→   152→    <Badge variant="outline" className={cn("rounded-full", s.className)}>

   157→   157→
   158→   158→function formatINR(n: number) {
   159→   159→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   160→   160→}
   161→   161→
   162→   162→function formatMonthYear(month: number, year: number) {
   163→   163→  return `${MONTHS[month] ?? "—"} ${year}`;
   164→   164→}
   165→   165→
   166→   166→function formatDate(iso: string | null) {
   167→   167→  if (!iso) return "—";
   168→   168→  return new Date(iso).toLocaleDateString("en-IN", {
   169→   169→    day: "numeric",
   170→   170→    month: "short",
   171→   171→    year: "numeric",
   172→   172→  });
   173→   173→}

   180→   180→  const user = useAuthStore((s) => s.user);
   181→   181→  const setView = useAppStore((s) => s.setView);
   182→   182→  const isAdmin =
   183→   183→    user?.role === "ADMIN";
   184→   184→
   185→   185→  const qc = useQueryClient();
   186→   186→  const [search, setSearch] = useState("");

   189→   189→  const [genMonth, setGenMonth] = useState<number>(new Date().getMonth());

   191→   191→  const [selectedBill, setSelectedBill] = useState<Bill | null>(null);
   192→   192→  const [voidTarget, setVoidTarget] = useState<Bill | null>(null);

   195→   195→    queryKey: ["bills"],
   196→   196→    queryFn: async () => {
   197→   197→      const r = await api.get<ApiResponse<Bill[]>>("/bills");
   198→   198→      return r.data;
   199→   199→    },
   200→   200→  });

   203→   203→    mutationFn: () =>
   204→   204→      api.post<ApiResponse<{ generated: number; month: number; year: number }>>(
   205→   205→        "/bills",
   206→   206→        { month: genMonth, year: genYear }
   207→   207→      ),

   209→   209→      toast.success(`Generated ${r.data.generated} bills for ${MONTHS[genMonth]} ${genYear}`);

   211→   211→      qc.invalidateQueries({ queryKey: ["bills"] });

   213→   213→    onError: (e: Error) => toast.error(e.message || "Failed to generate bills"),

   218→   218→      api.delete<ApiResponse<{ success: boolean }>>(`/bills/${id}`),

   222→   222→      qc.invalidateQueries({ queryKey: ["bills"] });

   225→   225→  });

   227→   227→  // ── Derived KPIs ──

   229→   229→    const active = bills.filter((b) => b.status !== "VOID");

   231→   231→    const totalCollected = active.reduce((s, b) => s + b.paidAmount, 0);
   232→   232→    const totalOutstanding = active.reduce((s, b) => s + b.dueAmount, 0);
   233→   233→    const overdueCount = active.filter((b) => b.status === "OVERDUE").length;
   234→   234→    return { totalBilled, totalCollected, totalOutstanding, overdueCount };
   235→   235→  }, [bills]);

   237→   237→  // ── Filtered list ──

   239→   239→    const q = search.trim().toLowerCase();

   241→   241→      if (statusFilter !== "ALL" && b.status !== statusFilter) return false;
   242→   242→      if (!q) return true;
   243→   243→      return (
   244→   244→        b.user.name?.toLowerCase().includes(q) ||
   245→   245→        b.user.email?.toLowerCase().includes(q) ||
   246→   246→        (b.user.room || "").toLowerCase().includes(q)

   248→   248→    });

   261→   261→          {Array.from({ length: 5 }).map((_, i) => (
   262→   262→            <ShimmerSkeleton key={i} className="h-20" />
   263→   263→          ))}
   264→   264→        </div>
   265→   265→      </div>
   266→   266→    );
   267→   267→  }

   276→   276→              Generate and track resident bills
   277→   277→            </p>
   278→   278→            <GlassButton
   279→   279→              onClick={() => setGenerateOpen(true)}
   280→   280→              className="shrink-0"

   282→   282→              <Plus className="h-4 w-4" />
   283→   283→              Generate Bills
   284→   284→            </GlassButton>

   292→   292→          <KpiCard

   294→   294→            value={kpis.totalBilled}
   295→   295→            icon={<Wallet className="h-5 w-5" />}
   296→   296→            color="primary"
   297→   297→            prefix="₹"
   298→   298→          />

   300→   300→            label="Total Collected"

   303→   303→            color="success"
   304→   304→            prefix="₹"
   305→   305→          />
   306→   306→          <KpiCard
   307→   307→            label="Outstanding"
   308→   308→            value={kpis.totalOutstanding}
   309→   309→            icon={<TrendingUp className="h-5 w-5" />}
   310→   310→            color="warning"
   311→   311→            prefix="₹"
   312→   312→          />
   313→   313→          <KpiCard
   314→   314→            label="Overdue Count"
   315→   315→            value={kpis.overdueCount}
   316→   316→            icon={<AlertCircle className="h-5 w-5" />}
