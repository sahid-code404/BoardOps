   190→function formatINR(n: number) {
   191→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   192→}
   193→
   194→function formatDate(iso: string) {
   195→  const d = new Date(iso);
   196→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" });
   197→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   198→  return `${datePart}, ${timePart}`;
   199→}
   200→
   201→/** Full date-time format: "29 June 2026, 05:28 am" */
   202→function formatDateTime(iso: string) {
   203→  const d = new Date(iso);
   204→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric" });
   205→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   206→  return `${datePart}, ${timePart}`;
   207→}

   209→const AVATAR_GRADIENTS = [
   210→  "from-violet-500 to-fuchsia-500",
   211→  "from-emerald-500 to-teal-500",
   212→  "from-amber-500 to-orange-500",
   213→  "from-rose-500 to-pink-500",

   218→function gradientFor(name: string) {
   219→  const idx = name.split("").reduce((acc, c) => acc + c.charCodeAt(0), 0) % AVATAR_GRADIENTS.length;
   220→  return AVATAR_GRADIENTS[idx];
   221→}
   222→
   223→function initials(name: string) {
   224→  return name.split(" ").filter(Boolean).slice(0, 2).map((p) => p[0]?.toUpperCase() ?? "").join("");
   225→}

   245→  // filtering. Prevents re-filtering large lists on every keystroke.

   247→    const t = setTimeout(() => setSearch(searchInput), 200);

   249→  }, [searchInput]);

   251→

   253→  const [actionTarget, setActionTarget] = useState<{

   264→
   265→  // Fetch ALL payments for KPIs (not affected by month picker)

   267→    queryKey: ["payments", "all"],

   269→      const r = await api.get<ApiResponse<Payment[]>>("/payments");

   271→    },
   272→    placeholderData: (prev) => prev,
   273→  });
   274→
   275→  // Fetch month-filtered payments for the list
   276→  const { data: payments = [], isLoading, isFetching } = useQuery({
   277→    queryKey: ["payments", { month: selectedMonth, year: selectedYear }],
   278→    queryFn: async () => {
   279→      const r = await api.get<ApiResponse<Payment[]>>("/payments", {
   280→        params: { month: selectedMonth, year: selectedYear, limit: 500 },
   281→      });
   282→      return r.data;
   283→    },
   284→    // Keep previous data visible while a new month/year loads (stale-while-revalidate).
   285→    // Eliminates the flash of empty content when switching months.
   286→    placeholderData: (prev) => prev,
   287→  });
   288→
   289→  // Fetch soft-deleted payments (deletion queue) — admin only
   290→  const { data: deletedPayments = [] } = useQuery({
   291→    queryKey: ["payments", "deleted", { month: selectedMonth, year: selectedYear }],
   292→    queryFn: async () => {
   293→      const r = await api.get<ApiResponse<Payment[]>>("/payments", {
   294→        params: { month: selectedMonth, year: selectedYear, includeDeleted: "true", limit: 500 },
   295→      });
   296→      return r.data;
   297→    },
   298→    enabled: isAdmin,
   299→    placeholderData: (prev) => prev,
   300→  });
   301→
   302→  const submitMutation = useMutation({
   303→    mutationFn: (payload: {
   304→      amount: number;
   305→      method: PaymentMethod;
   306→      billId?: string;
   307→      reference?: string;
   308→      notes?: string;
   309→    }) =>
   310→      api.post<ApiResponse<Payment>>("/payments", payload),
   311→    onSuccess: () => {
   312→      toast.success("Payment submitted — pending admin approval");
   313→      setSubmitOpen(false);
   314→      qc.invalidateQueries({ queryKey: ["payments"] });
   315→    },
   316→    onError: (e: Error) => toast.error(e.message || "Failed to submit payment"),
   317→  });
   318→
   319→  const actionMutation = useMutation({
   320→    mutationFn: ({

   326→    }) => api.patch<ApiResponse<Payment>>(`/payments/${id}`, { action }),
   327→    onSuccess: (_data, vars) => {
   328→      toast.success(
   329→        vars.action === "APPROVE"
   330→          ? "Payment approved — bill updated"
   331→          : "Payment rejected"
   332→      );
   333→      setActionTarget(null);
   334→      qc.invalidateQueries({ queryKey: ["payments"] });
   335→      qc.invalidateQueries({ queryKey: ["bills"] });
   336→    },
   337→    onError: (e: Error) => toast.error(e.message || "Action failed"),
   338→  });
   339→
   340→  // Edit payment (PUT /api/payments/[id] { action: "EDIT", ... })

   342→    mutationFn: async ({
   343→      id,
   344→      payload,
   345→    }: {
   346→      id: string;

   348→        amount?: number;

   350→        reference?: string | null;
   351→        notes?: string | null;
   352→      };

   354→      api.put<ApiResponse<Payment>>(`/payments/${id}`, {
   355→        action: "EDIT",
   356→        ...payload,
   357→      }),
   358→    onSuccess: () => {
   359→      toast.success("Payment updated");
   360→      setEditOpen(false);
   361→      setEditTarget(null);
   362→      qc.invalidateQueries({ queryKey: ["payments"] });
   363→    },
   364→    onError: (e: Error) => toast.error(e.message || "Failed to update payment"),
   365→  });
   366→
   367→  // Void payment (PUT /api/payments/[id] { action: "VOID" })
   368→  const voidMutation = useMutation({
   369→    mutationFn: (id: string) =>
   370→      api.put<ApiResponse<Payment>>(`/payments/${id}`, { action: "VOID" }),
   371→    onSuccess: () => {
   372→      toast.success("Payment voided");
   373→      setVoidTarget(null);
   374→      qc.invalidateQueries({ queryKey: ["payments"] });
   375→      qc.invalidateQueries({ queryKey: ["bills"] });

   377→    onError: (e: Error) => toast.error(e.message || "Failed to void payment"),
   378→  });
   379→
   380→  // Soft-delete payment (DELETE /api/payments/[id] { reason? })
   381→  const deleteMutation = useMutation({
   382→    mutationFn: async ({ id, reason }: { id: string; reason: string }) => {
   383→      await api.delete(`/payments/${id}`, {
   384→        body: JSON.stringify({ reason: reason || undefined }),
   385→      });
   386→    },
   387→    onSuccess: () => {
   388→      toast.success("Payment scheduled for deletion — permanently removed in 7 days");
   389→      setDeleteTarget(null);
   390→      setDeleteReason("");
   391→      qc.invalidateQueries({ queryKey: ["payments"] });
   392→    },
   393→    onError: (e: Error) => toast.error(e.message || "Failed to delete payment"),
   394→  });
   395→
   396→  // Restore soft-deleted payment (POST /api/payments/[id]/restore)
   397→  const restoreMutation = useMutation({
   398→    mutationFn: async (id: string) => {
   399→      const r = await api.post<ApiResponse<Payment>>(`/payments/${id}/restore`);
   400→      return r.data;
   401→    },
   402→    onSuccess: () => {
   403→      toast.success("Payment restored successfully");
   404→      qc.invalidateQueries({ queryKey: ["payments"] });
   405→    },
   406→    onError: (e: Error) => toast.error(e.message || "Failed to restore payment"),
   407→  });
   408→
   409→  const openEditForm = useCallback((p: Payment) => {
   410→    setEditTarget(p);
   411→    setEditOpen(true);
   412→  }, []);
   413→
   414→  const closeEditForm = useCallback(() => {
   415→    setEditOpen(false);
   416→    setEditTarget(null);
   417→  }, []);
   418→
   419→  // KPIs — computed from the month-filtered payments list

   424→    const rejected = payments.filter((p) => p.status === "REJECTED").length;
   425→    const refunded = payments.filter((p) => p.status === "REFUNDED").length;

   436→      if (!q) return true;

   438→        p.user.name?.toLowerCase().includes(q) ||
   439→        p.user.email?.toLowerCase().includes(q) ||
   440→        (p.reference || "").toLowerCase().includes(q)

   442→    });
   443→  }, [sourcePayments, search, statusFilter]);

   445→  // Pending payments for the admin approval card — from ALL payments (not month-scoped)
   446→  const pendingPayments = allPayments.filter((p) => p.status === "PENDING");
   447→
   448→  if (isLoading) {
   449→    return (
   450→      <div className="space-y-4">
   451→        <div className="grid-kpi gap-3">
   452→          {Array.from({ length: 4 }).map((_, i) => (

   459→            <ShimmerSkeleton key={i} className="h-20" />

   461→        </div>

   482→        )}
   483→      </AnimatePresence>

   485→      <StaggerItem>
   486→        <div className="flex items-center justify-center gap-4">
   487→          <motion.button
   488→            whileTap={{ scale: 0.9 }}
   489→            onClick={() => {
   490→              const d = new Date(selectedYear, selectedMonth - 1, 1);

   496→          >
   497→            <ChevronLeft className="h-5 w-5" />

   503→          >
   504→            <Calendar className="h-4 w-4 text-primary shrink-0" />
   505→            <div className="leading-tight text-center">
   506→              <p className="text-sm font-bold text-primary">
   507→                {new Date(selectedYear, selectedMonth).toLocaleDateString("en-US", { month: "long" })}
   508→              </p>
   509→              <p className="text-[11px] text-muted-foreground">

   522→              setSelectedMonth(d.getMonth());
   523→              setSelectedYear(d.getFullYear());

   527→          >
   528→            <ChevronRight className="h-5 w-5" />

   530→        </div>

   532→
   533→      {/* Action bar — only for non-admins (who can submit payments) */}
   534→      {!isAdmin && (

   536→          <div className="flex items-center justify-end gap-3">

   558→          />
   559→          <KpiCard

   561→            value={kpis.pending}
   562→            icon={<Clock className="h-5 w-5" />}

  1052→const PaymentRow = memo(function PaymentRow({
  1053→  payment,
  1054→  isAdmin,
  1055→  onApprove,
  1056→  onReject,
  1057→  onEdit,
  1058→  onDelete,
  1059→  onVoid,
  1060→  onRestore,
  1061→}: {
  1062→  payment: Payment;
  1063→  isAdmin: boolean;
  1064→  onApprove: () => void;
  1065→  onReject: () => void;
  1066→  onEdit: () => void;
  1067→  onDelete: () => void;
  1068→  onVoid: () => void;
  1069→  onRestore: () => void;
  1070→}) {
  1071→  const isDeleted = !!payment.deletedAt;
  1072→
  1073→  // Build the actions list — admin-only. Deleted rows show Restore; otherwise
  1074→  // PENDING rows get Approve/Reject, non-VOID rows get Edit/Void, all get Delete.
  1075→  const actions: {
  1076→    label: string;
  1077→    icon: typeof CheckCircle2;
  1078→    onClick: () => void;
  1079→    variant?: "destructive";
  1080→  }[] = [];
  1081→
  1082→  if (isDeleted) {
  1083→    if (isAdmin) {
  1084→      actions.push({ label: "Restore Payment", icon: RotateCcw, onClick: onRestore });
  1085→    }
  1086→  } else {
  1087→    if (isAdmin && payment.status === "PENDING") {
  1088→      actions.push({ label: "Approve Payment", icon: CheckCircle2, onClick: onApprove });
  1089→      actions.push({
  1090→        label: "Reject Payment",
  1091→        icon: XCircle,
  1092→        onClick: onReject,
  1093→        variant: "destructive",
  1094→      });
  1095→    }
  1096→    if (isAdmin && payment.status !== "VOID") {
  1097→      actions.push({ label: "Edit Payment", icon: PencilLine, onClick: onEdit });
  1098→      actions.push({
  1099→        label: "Void Payment",
  1100→        icon: Ban,
  1101→        onClick: onVoid,
  1102→        variant: "destructive",
  1103→      });
  1104→    }
  1105→    if (isAdmin) {
  1106→      actions.push({
  1107→        label: "Delete Payment",
  1108→        icon: Trash2,
  1109→        onClick: onDelete,
  1110→        variant: "destructive",
  1111→      });
  1112→    }
  1113→  }
  1114→
  1115→  const methodMeta = METHOD_META[payment.method];
  1116→  const statusMeta = STATUS_STYLES[payment.status];
  1117→
  1118→  return (
  1119→    <GlassCard className="p-4" hover={false}>
  1120→      <div className="flex items-start gap-3">
  1121→        <Avatar className="h-10 w-10 rounded-xl shrink-0">
  1122→          <AvatarFallback
  1123→            className={cn(
  1124→              "rounded-xl bg-gradient-to-br text-white font-semibold text-xs",
  1125→              gradientFor(payment.user.name)
  1126→            )}
  1127→          >
  1128→            {initials(payment.user.name) || "U"}
  1129→          </AvatarFallback>
  1130→        </Avatar>
  1131→
  1132→        <div className="flex-1 min-w-0">
  1133→          <div className="flex items-start justify-between gap-2">
  1134→            <div className="min-w-0">
  1135→              <div className="flex items-center gap-2 flex-wrap">
  1136→                <h3
  1137→                  className={cn(
  1138→                    "font-medium text-sm truncate",
  1139→                    isDeleted && "text-muted-foreground line-through"
  1140→                  )}
  1141→                >
  1142→                  {isAdmin ? payment.user.name : methodMeta.label}
  1143→                </h3>
  1144→                {isDeleted ? (
  1145→                  <Badge
  1146→                    variant="outline"
  1147→                    className="text-[10px] bg-destructive/15 text-destructive border-destructive/30"
  1148→                  >
  1149→                    <Clock className="h-2.5 w-2.5" />{" "}
  1150→                    {formatDeletionCountdown(new Date(payment.deletedAt!))}
  1151→                  </Badge>
  1152→                ) : (
  1153→                  <>
  1154→                    <Badge
  1155→                      variant="outline"
  1156→                      className={cn("text-[10px]", statusMeta.className)}
  1157→                    >
  1158→                      {statusMeta.label}
  1159→                    </Badge>
  1160→                    <Badge
  1161→                      variant="outline"
  1162→                      className={cn("text-[10px]", methodMeta.className)}
  1163→                    >
  1164→                      {methodMeta.label}
  1165→                    </Badge>
  1166→                  </>
  1167→                )}
  1168→              </div>
  1169→              {/* Transaction strip — Amount with label, same size as billing */}

  1471→function PaymentEditBody({
  1472→  payment,
  1473→  onSubmit,
  1474→  loading,
  1475→  onCancel,
  1476→}: {
  1477→  payment: Payment | null;
  1478→  onSubmit: (
  1479→    id: string,
  1480→    payload: {
  1481→      amount?: number;
  1482→      method?: PaymentMethod;
  1483→      reference?: string | null;
  1484→      notes?: string | null;
  1485→    }
  1486→  ) => void;
  1487→  loading: boolean;
  1488→  onCancel: () => void;

  1491→  const [method, setMethod] = useState<PaymentMethod>(payment?.method ?? "UPI");
  1492→  const [reference, setReference] = useState(payment?.reference ?? "");
  1493→  const [notes, setNotes] = useState(payment?.notes ?? "");
  1494→  const [errors, setErrors] = useState<Record<string, string>>({});
  1495→
  1496→  // Cannot edit amount on APPROVED + bill-linked payments — backend will
  1497→  // reject with 422. Lock the field and explain why; admin must void + resubmit.
  1498→  const amountLocked = payment?.status === "APPROVED" && !!payment?.billId;
  1499→
  1500→  function handleSubmit() {
  1501→    if (!payment) return;
  1502→    const next: Record<string, string> = {};
  1503→    const amt = parseFloat(amount);
  1504→    if (!amountLocked) {
  1505→      if (!amt || amt <= 0) next.amount = "Enter a valid amount";
  1506→    }
  1507→    setErrors(next);
  1508→    if (Object.keys(next).length > 0) return;
  1509→
  1510→    const payload: {
  1511→      amount?: number;
  1512→      method?: PaymentMethod;
  1513→      reference?: string | null;
  1514→      notes?: string | null;
  1515→    } = {
  1516→      method,
  1517→      reference: reference.trim() || null,
  1518→      notes: notes.trim() || null,
  1519→    };
  1520→    if (!amountLocked) payload.amount = amt;
  1521→    onSubmit(payment.id, payload);
  1522→  }
  1523→
  1524→  return (
  1525→    <>
  1526→      <SheetHeader className="px-6 pt-6 pb-2">
  1527→        <SheetTitle className="flex items-center gap-2 text-xl">

  1533→        </SheetDescription>
  1534→      </SheetHeader>
  1535→
  1536→      <div className="flex-1 overflow-y-auto px-6 py-4 space-y-3 no-scrollbar">
  1537→        <GlassInput
  1538→          label="Amount (₹)"
  1539→          type="number"
  1540→          inputMode="decimal"
  1541→          placeholder="0.00"
  1542→          value={amount}
  1543→          onChange={(e) => setAmount(e.target.value)}
  1544→          error={errors.amount}
  1545→          icon={<IndianRupee />}
  1546→          disabled={amountLocked}
  1547→          hint={
  1548→            amountLocked
  1549→              ? "Amount locked — this approved payment is linked to a bill. Void it and submit a new payment to change the amount."
  1550→              : undefined
  1551→          }
  1552→        />
  1553→
  1554→        <div className="space-y-1.5">

  1559→            value={method}
  1560→            onValueChange={(v) => setMethod(v as PaymentMethod)}
  1561→          >
  1562→            <SelectTrigger className="w-full h-11 rounded-2xl">
