   200→   200→function formatDate(iso: string | null) {
   201→   201→  if (!iso) return "—";
   202→   202→  return new Date(iso).toLocaleDateString("en-IN", {
   203→   203→    day: "numeric",
   204→   204→    month: "short",
   205→   205→    year: "numeric",
   206→   206→  });
   207→   207→}
   208→   208→
   209→   209→/** Full date-time format: "29 June 2026, 05:28 am" */
   210→   210→function formatDateTime(iso: string | null) {
   211→   211→  if (!iso) return "—";
   212→   212→  const d = new Date(iso);
   213→   213→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric" });
   214→   214→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   215→   215→  return `${datePart}, ${timePart}`;
   216→   216→}
   217→   217→
   218→   218→/** A bill is overdue if it has a due date in the past AND there's still an outstanding due amount. */
   219→   219→function isOverdue(bill: Bill): boolean {
   220→   220→  if (!bill.dueDate || bill.deletedAt) return false;
   221→   221→  if (bill.status === "PAID" || bill.status === "VOID") return false;
   222→   222→  return new Date(bill.dueDate) < new Date() && bill.dueAmount > 0;
   223→   223→}

   888→   888→function KpiCard({
   889→   889→  label,
   890→   890→  value,
   891→   891→  icon,
   892→   892→  color,
   893→   893→  prefix,
   894→   894→}: {
   895→   895→  label: string;
   896→   896→  value: number;
   897→   897→  icon: React.ReactNode;
   898→   898→  color: "primary" | "success" | "warning" | "danger" | "info";
   899→   899→  prefix?: string;
   900→   900→}) {
   901→   901→  const colorVar =
   902→   902→    color === "primary"
   903→   903→      ? "var(--primary)"
   904→   904→      : color === "success"
   905→   905→        ? "var(--success)"
   906→   906→        : color === "warning"
   907→   907→          ? "var(--warning)"
   908→   908→          : color === "danger"
   909→   909→            ? "var(--destructive)"
   910→   910→            : "var(--info)";
   911→   911→  return (
   912→   912→    <GlassCard className="p-4" glow={color === "danger" ? "danger" : color === "warning" ? "warning" : color === "success" ? "success" : "primary"}>
   913→   913→      <div className="flex items-start justify-between mb-3">
   914→   914→        <div
   915→   915→          className="grid place-items-center h-10 w-10 rounded-2xl"
   916→   916→          style={{
   917→   917→            background: `color-mix(in oklch, ${colorVar} 15%, transparent)`,
   918→   918→            color: colorVar,
   919→   919→          }}
   920→   920→        >
   921→   921→          {icon}
   922→   922→        </div>
   923→   923→      </div>
   924→   924→      <p className="text-xs text-muted-foreground">{label}</p>
   925→   925→      <div className="text-2xl font-bold tracking-tight tabular-nums">
   926→   926→        <AnimatedCounter value={value} prefix={prefix || ""} />
   927→   927→      </div>
   928→   928→    </GlassCard>
   929→   929→  );
   930→   930→}
