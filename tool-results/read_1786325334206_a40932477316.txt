    56→const loginSchema = z.object({
    57→  email: z.string().email("Enter a valid email"),
    58→  password: z.string().min(1, "Password is required"),
    59→});
    60→
    61→const registerSchema = z
    62→  .object({
    63→    name: z.string().min(2, "Name too short"),
    64→    institutionName: z.string().min(2, "Institution name is required"),
    65→    institutionUserId: z.string().min(1, "Institution User ID is required"),
    66→    email: z.string().email("Enter a valid email"),
    67→    phone: z.string().min(8, "Enter a valid phone"),
    68→    room: z.string().min(1, "Room number is required"),
    69→    password: z.string().min(8, "At least 8 characters"),
    70→    confirmPassword: z.string(),
    71→    // Allow empty string (user didn't select) OR a valid enum value. Normalize "" to undefined.
    72→    gender: z.union([z.literal(""), z.enum(["MALE", "FEMALE", "OTHER"])]).optional(),
    73→    consents: z.object({
    74→      rules: z.boolean().refine((v) => v, "You must accept the Institution Rules"),
    75→      privacy: z.boolean().refine((v) => v, "You must accept the Privacy Policy"),
    76→      terms: z.boolean().refine((v) => v, "You must accept the Terms & Conditions"),
    77→    }),
    78→  })
    79→  .refine((d) => d.password === d.confirmPassword, {
    80→    message: "Passwords do not match",
    81→    path: ["confirmPassword"],
    82→  });

   412→        <GlassCard strong className="p-6" hover={false}>

  1034→function ResubmitDialog({
  1035→  status,
  1036→  onClose,
  1037→  onDone,
  1038→}: {
  1039→  status: RegistrationStatus;
  1040→  onClose: () => void;
  1041→  onDone: () => void;
  1042→}) {
  1043→  const fields = status.changesRequested ?? [];
  1044→  const [form, setForm] = useState({
  1045→    name: status.name ?? "",
  1046→    institutionUserId: status.institutionUserId ?? "",
  1047→    phone: status.phone ?? "",
  1048→    room: status.room ?? "",
  1049→    gender: (status.gender as "" | "MALE" | "FEMALE" | "OTHER") || "",
  1050→  });
  1051→  const [errors, setErrors] = useState<Record<string, string>>({});
  1052→  const [loading, setLoading] = useState(false);
  1053→
  1054→  const submit = async () => {
  1055→    setErrors({});
  1056→    if (!form.name || form.name.length < 2) {
  1057→      setErrors({ name: "Name too short" });
  1058→      return;
  1059→    }
  1060→    if (!form.institutionUserId) {
  1061→      setErrors({ institutionUserId: "Institution User ID is required" });
  1062→      return;
  1063→    }
  1064→    if (!form.phone || form.phone.length < 8) {
  1065→      setErrors({ phone: "Enter a valid phone" });
  1066→      return;
  1067→    }
  1068→    if (!form.room) {
  1069→      setErrors({ room: "Room number is required" });
  1070→      return;
  1071→    }
  1072→    try {
  1073→      setLoading(true);
  1074→      await api.post("/auth/resubmit", {
  1075→        email: status.email,
  1076→        name: form.name,
  1077→        institutionUserId: form.institutionUserId,
  1078→        phone: form.phone,
  1079→        room: form.room,
  1080→        gender: form.gender || undefined,
  1081→      });
  1082→      toast.success("Resubmitted for review!");
  1083→      onDone();
  1084→    } catch (err: any) {
  1085→      toast.error(err.message || "Could not resubmit");
  1086→    } finally {
  1087→      setLoading(false);
  1088→    }
  1089→  };
  1090→
  1091→  return (
  1092→    <div className="mt-4 rounded-2xl border-2 border-primary/30 bg-primary/5 p-4 space-y-3">
  1093→      <div className="flex items-center justify-between gap-2">
  1094→        <p className="font-semibold text-sm">Update fields & resubmit</p>
  1095→        <button
  1096→          type="button"
  1097→          onClick={onClose}
  1098→          className="text-muted-foreground hover:text-foreground"
  1099→          aria-label="Close"
  1100→        >
  1101→          ✕
  1102→        </button>
  1103→      </div>
  1104→      <GlassInput
  1105→        label="Full Name"
  1106→        value={form.name}
  1107→        onChange={(e) => setForm({ ...form, name: e.target.value })}
  1108→        error={errors.name}
  1109→        icon={<User className="h-4 w-4" />}
  1110→      />
  1111→      <GlassInput
  1112→        label="Institution User ID"
  1113→        value={form.institutionUserId}
  1114→        onChange={(e) => setForm({ ...form, institutionUserId: e.target.value })}
  1115→        error={errors.institutionUserId}
  1116→        icon={<BadgeCheck className="h-4 w-4" />}
  1117→      />
  1118→      <GlassInput
