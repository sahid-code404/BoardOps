   169→function formatDate(iso: string) {

   186→/** Returns { top, bottom } for the date picker.
   187→ *  - Today/Yesterday/Tomorrow: top = relative label, bottom = "EEE, d MMM"
   188→ *  - Other dates: top = "d MMM" (date only), bottom = "EEE" (day name only, no duplicate) */
   189→function getDatePickerLabels(d: Date): { top: string; bottom: string } {
   190→  const today = new Date();
   191→  today.setHours(0, 0, 0, 0);
   192→  const target = new Date(d);
   193→  target.setHours(0, 0, 0, 0);
   194→  const diffMs = target.getTime() - today.getTime();
   195→  const diffDays = Math.round(diffMs / (24 * 60 * 60 * 1000));
   196→
   197→  if (diffDays === 0) return { top: "Today", bottom: format(target, "EEE, d MMM") };
   198→  if (diffDays === -1) return { top: "Yesterday", bottom: format(target, "EEE, d MMM") };
   199→  if (diffDays === 1) return { top: "Tomorrow", bottom: format(target, "EEE, d MMM") };
   200→  // Far dates: top = date only, bottom = day name only (no duplicate)
   201→  return { top: format(target, "d MMM"), bottom: format(target, "EEE") };
   202→}

   588→                        {p.reference || "—"}

   590→                      <TableCell className="text-sm text-muted-foreground">
   591→                        {formatDate(p.createdAt)}
   592→                      </TableCell>
   593→                      {!isAdmin && (
   594→                        <TableCell className="text-right pr-4 text-xs text-muted-foreground max-w-[200px] truncate">
   595→                          {p.notes || "—"}

   607→      {/* Submit Payment Dialog */}
   608→      <SubmitPaymentDialog

   612→        loading={submitMutation.isPending}
   613→      />
   614→
   615→      {/* Approve/Reject confirm */}
   616→      <AlertDialog
   617→        open={!!actionTarget}
   618→        onOpenChange={(o) => !o && setActionTarget(null)}

   620→        <AlertDialogContent className="rounded-3xl">

   623→              {actionTarget?.action === "APPROVE"
   624→                ? "Approve payment?"

   626→            </AlertDialogTitle>
   627→            <AlertDialogDescription>
   628→              {actionTarget && (
   629→                <>
   630→                  {actionTarget.action === "APPROVE" ? (
   631→                    <>
   632→                      Approve{" "}

   635→                      </span>{" "}
   636→                      from{" "}
   637→                      <span className="font-medium text-foreground">
   638→                        {actionTarget.payment.user.name}
   639→                      </span>
   640→                      ? The linked bill will be updated.
   641→                    </>
   642→                  ) : (
   643→                    <>
   644→                      Reject the{" "}
   645→                      <span className="font-medium text-foreground">
   646→                        {formatINR(actionTarget.payment.amount)}
   647→                      </span>{" "}
   648→                      payment from{" "}
   649→                      <span className="font-medium text-foreground">
   650→                        {actionTarget.payment.user.name}
   651→                      </span>
   652→                      ?
   653→                    </>
   654→                  )}
   655→                </>
   656→              )}
   657→            </AlertDialogDescription>
   658→          </AlertDialogHeader>
   659→          <AlertDialogFooter>
   660→            <AlertDialogCancel className="rounded-2xl">Cancel</AlertDialogCancel>
   661→            <AlertDialogAction
   662→              className={cn(
   663→                "rounded-2xl",
   664→                actionTarget?.action === "APPROVE"
   665→                  ? "bg-success text-white hover:bg-success/90"
   666→                  : "bg-destructive text-white hover:bg-destructive/90"
   667→              )}
   668→              onClick={() =>
   669→                actionTarget &&
   670→                actionMutation.mutate({
   671→                  id: actionTarget.payment.id,
   672→                  action: actionTarget.action,
   673→                })
   674→              }
   675→              disabled={actionMutation.isPending}
   676→            >
   677→              {actionMutation.isPending
   678→                ? "Processing…"
   679→                : actionTarget?.action === "APPROVE"
   680→                  ? "Approve"
   681→                  : "Reject"}
   682→            </AlertDialogAction>
   683→          </AlertDialogFooter>
   684→        </AlertDialogContent>
   685→      </AlertDialog>
   686→    </StaggerGroup>
   687→  );
   688→}

   872→function SubmitPaymentDialog({
   873→  open,
   874→  onOpenChange,
   875→  onSubmit,
   876→  loading,
   877→}: {
   878→  open: boolean;
   879→  onOpenChange: (o: boolean) => void;
   880→  onSubmit: (payload: {
   881→    amount: number;
   882→    method: PaymentMethod;
   883→    billId?: string;
   884→    reference?: string;
   885→    notes?: string;
   886→  }) => void;
   887→  loading: boolean;
   888→}) {
   889→  const [amount, setAmount] = useState("");
   890→  const [method, setMethod] = useState<PaymentMethod>("UPI");
   891→  const [billId, setBillId] = useState<string>("NONE");
   892→  const [reference, setReference] = useState("");
   893→  const [notes, setNotes] = useState("");
   894→  const [error, setError] = useState<string>("");
   895→
   896→  // Load user's outstanding bills for selection
   897→  const { data: bills = [] } = useQuery({
   898→    queryKey: ["bills"],
   899→    queryFn: async () => {
   900→      const r = await api.get<ApiResponse<BillListItem[]>>("/bills");
   901→      return r.data;
   902→    },
   903→    enabled: open,
   904→  });
   905→
   906→  const outstanding = bills.filter(
   907→    (b) => b.status !== "PAID" && b.status !== "VOID" && b.dueAmount > 0
   908→  );
   909→
   910→  function reset() {
   911→    setAmount("");
   912→    setMethod("UPI");
   913→    setBillId("NONE");
   914→    setReference("");
   915→    setNotes("");
   916→    setError("");
   917→  }
   918→
   919→  function handleClose(o: boolean) {
   920→    if (!o) reset();
   921→    onOpenChange(o);
   922→  }
   923→
   924→  function handleSubmit() {
   925→    const amt = parseFloat(amount);
   926→    if (!amt || amt <= 0) {
   927→      setError("Enter a valid amount greater than 0");
   928→      return;
   929→    }
   930→    onSubmit({
   931→      amount: amt,
   932→      method,
   933→      billId: billId === "NONE" ? undefined : billId,
   934→      reference: reference.trim() || undefined,
   935→      notes: notes.trim() || undefined,
   936→    });
   937→  }
   938→
   939→  return (
   940→    <Dialog open={open} onOpenChange={handleClose}>
   941→      <DialogContent className="rounded-3xl max-w-md">
   942→        <DialogHeader>
   943→          <DialogTitle className="flex items-center gap-2">
   944→            <Wallet className="h-5 w-5 text-primary" />
   945→            Submit a Payment
   946→          </DialogTitle>
   947→          <DialogDescription>
   948→            Your payment will be reviewed by an administrator before being
   949→            applied to your bill.
   950→          </DialogDescription>
   951→        </DialogHeader>
   952→
   953→        <div className="space-y-3">
   954→          <GlassInput
   955→            label="Amount (₹)"
   956→            type="number"
   957→            inputMode="decimal"
   958→            placeholder="0.00"
   959→            value={amount}
   960→            onChange={(e) => {
   961→              setAmount(e.target.value);
   962→              setError("");
   963→            }}
   964→            error={error}
   965→            icon={<IndianRupee />}
   966→          />
   967→
   968→          <div className="space-y-1.5">
   969→            <label className="text-xs font-medium text-muted-foreground ml-1">
   970→              Method
   971→            </label>
   972→            <Select
   973→              value={method}
   974→              onValueChange={(v) => setMethod(v as PaymentMethod)}
   975→            >
   976→              <SelectTrigger className="w-full h-11 rounded-2xl">
   977→                <SelectValue />
   978→              </SelectTrigger>
   979→              <SelectContent>
   980→                {Object.entries(METHOD_META).map(([k, m]) => (
   981→                  <SelectItem key={k} value={k}>
   982→                    {m.label}
   983→                  </SelectItem>
   984→                ))}
   985→              </SelectContent>
   986→            </Select>
   987→          </div>
   988→
   989→          {outstanding.length > 0 && (
   990→            <div className="space-y-1.5">
   991→              <label className="text-xs font-medium text-muted-foreground ml-1">
   992→                Apply to bill (optional)
   993→              </label>
   994→              <Select value={billId} onValueChange={setBillId}>
   995→                <SelectTrigger className="w-full h-11 rounded-2xl">
   996→                  <SelectValue placeholder="Select a bill" />
   997→                </SelectTrigger>
   998→                <SelectContent>
   999→                  <SelectItem value="NONE">No specific bill</SelectItem>
  1000→                  {outstanding.map((b) => (
  1001→                    <SelectItem key={b.id} value={b.id}>
  1002→                      {b.user.name} — {formatMonthLabel(b.periodMonth, b.periodYear)} · ₹
  1003→                      {Math.round(b.dueAmount).toLocaleString("en-IN")} due
  1004→                    </SelectItem>
  1005→                  ))}
  1006→                </SelectContent>
  1007→              </Select>
  1008→            </div>
  1009→          )}
  1010→
  1011→          <GlassInput
  1012→            label="Reference (optional)"
  1013→            placeholder="UTR / Txn ID"
  1014→            value={reference}
  1015→            onChange={(e) => setReference(e.target.value)}
  1016→            icon={<ArrowUpRight />}
  1017→          />
  1018→
  1019→          <GlassTextarea
  1020→            label="Notes (optional)"
  1021→            placeholder="Any note for the admin…"
  1022→            value={notes}
  1023→            onChange={(e) => setNotes(e.target.value)}
  1024→            rows={2}
  1025→          />
  1026→        </div>
  1027→
  1028→        <DialogFooter>
  1029→          <GlassButton variant="ghost" onClick={() => handleClose(false)}>
  1030→            Cancel
  1031→          </GlassButton>
  1032→          <GlassButton onClick={handleSubmit} loading={loading}>
  1033→            <ArrowDownRight className="h-4 w-4" />
  1034→            Submit
  1035→          </GlassButton>
  1036→        </DialogFooter>
  1037→      </DialogContent>
  1038→    </Dialog>
  1039→  );
  1040→}

  1046→  });
