   126→async function unwrap<T>(promise: Promise<unknown>): Promise<T> {
   127→  const res = await promise;
   128→  if (res && typeof res === "object" && "success" in res && "data" in (res as Record<string, unknown>)) {
   129→    return (res as { data: T }).data;
   130→  }
   131→  return res as T;
   132→}

   134→function initials(name: string) {
   135→  return name
   136→    .split(" ")
   137→    .filter(Boolean)
   138→    .slice(0, 2)
   139→    .map((p) => p[0]?.toUpperCase() ?? "")
   140→    .join("");
   141→}

   732→function actionResultStatus(action: Action): string {
   733→  switch (action) {
   734→    case "APPROVE":
   735→    case "ACTIVATE":
   736→    case "RESTORE":
   737→      return "Active";
   738→    case "SUSPEND":
   739→      return "Suspended";
   740→    case "DEACTIVATE":
   741→      return "Inactive";
   742→    case "ARCHIVE":
   743→      return "Archived";
   744→    default:
   745→      return "";
   746→  }

   779→function UserRow({
   780→  user,
   781→  onAction,
   782→  canEditRole,
   783→}: {
   784→  user: ManagedUser;
   785→  onAction: (a: Action) => void;
   786→  canEditRole: boolean;
   787→}) {
   788→  const sMeta = STATUS_META[user.status];
   789→  const rMeta = ROLE_META[user.role];
   790→
   791→  const actions: { action: Action; label: string; icon: typeof CheckCircle2; variant?: "destructive" }[] = [];
   792→
   793→  // If user is in deletion queue, only show Restore
   794→  if (user.deletedAt) {
   795→    actions.push({ action: "RESTORE_DELETED", label: "Restore User", icon: RotateCcw });
   796→  } else {
   797→    // Edit User is always available for all statuses
   798→    actions.push({ action: "EDIT_USER", label: "Edit User", icon: Pencil });
   799→    switch (user.status) {
   800→      case "PENDING":
   801→        actions.push({ action: "APPROVE", label: "Approve", icon: CheckCircle2 });
   802→        actions.push({ action: "DEACTIVATE", label: "Reject", icon: Ban, variant: "destructive" });
   803→        break;
   804→      case "ACTIVE":
   805→        actions.push({ action: "SUSPEND", label: "Suspend", icon: Ban, variant: "destructive" });
   806→        actions.push({ action: "DEACTIVATE", label: "Deactivate", icon: Power });
   807→        actions.push({ action: "ARCHIVE", label: "Archive", icon: Archive });
   808→        if (canEditRole) actions.push({ action: "ASSIGN_ROLE", label: "Assign Role", icon: Shield });
   809→        break;
   810→      case "SUSPENDED":
   811→        actions.push({ action: "ACTIVATE", label: "Activate", icon: Power });
   812→        actions.push({ action: "ARCHIVE", label: "Archive", icon: Archive });
   813→        break;
   814→      case "INACTIVE":
   815→        actions.push({ action: "ACTIVATE", label: "Activate", icon: Power });
   816→        actions.push({ action: "ARCHIVE", label: "Archive", icon: Archive });
   817→        break;
   818→      case "ARCHIVED":
   819→        actions.push({ action: "RESTORE", label: "Restore", icon: RotateCcw });
   820→        break;
   821→      case "APPROVED":
   822→        actions.push({ action: "ACTIVATE", label: "Activate", icon: Power });
   823→        break;
   824→    }
   825→    // Delete is available for all non-deleted users (except other admins)
   826→    if (user.role !== "ADMIN") {
   827→      actions.push({ action: "DELETE_USER", label: "Delete", icon: Trash2, variant: "destructive" });
   828→    }
   829→  }
   830→
   831→  return (
   832→    <GlassCard className="p-4 md:p-5" hover={false}>
   833→      <div className="flex items-start gap-3 md:gap-4">
   834→        <Avatar className="h-12 w-12 md:h-14 md:w-14 rounded-2xl">
   835→          {user.avatarUrl && <AvatarImage src={user.avatarUrl} alt={user.name} />}
   836→          <AvatarFallback className={cn("rounded-2xl bg-gradient-to-br text-white font-semibold", gradientFor(user.name))}>
   837→            {initials(user.name) || "U"}
   838→          </AvatarFallback>
   839→        </Avatar>
   840→
   841→        <div className="flex-1 min-w-0">
   842→          <div className="flex items-start justify-between gap-2">
   843→            <div className="min-w-0">
   844→              <div className="flex items-center gap-2 flex-wrap">
   845→                <h3 className={cn("font-semibold truncate", user.deletedAt && "text-muted-foreground line-through")}>
   846→                  {user.name}
   847→                </h3>
   848→                {user.deletedAt ? (
   849→                  <Badge variant="outline" className="text-[10px] bg-destructive/15 text-destructive border-destructive/30">
   850→                    <Clock className="h-2.5 w-2.5" /> {formatDeletionCountdown(new Date(user.deletedAt))}
   851→                  </Badge>
   852→                ) : (
   853→                  <>
   854→                    <Badge variant="outline" className={cn("text-[10px]", rMeta.className)}>
   855→                      {rMeta.label}
   856→                    </Badge>
   857→                    <Badge variant="outline" className={cn("text-[10px]", sMeta.className)}>
   858→                      {sMeta.label}
   859→                    </Badge>
   860→                  </>
   861→                )}
   862→              </div>
   863→              <div className="flex flex-col sm:flex-row sm:items-center gap-x-3 gap-y-0.5 mt-1.5 text-xs text-muted-foreground">
   864→                <span className="inline-flex items-center gap-1 truncate">
   865→                  <Mail className="h-3 w-3" /> {user.email}
   866→                </span>
   867→                {user.phone && (
   868→                  <span className="inline-flex items-center gap-1">
   869→                    <Phone className="h-3 w-3" /> {user.phone}
   870→                  </span>
   871→                )}
   872→                {user.room && (
   873→                  <span className="inline-flex items-center gap-1">
   874→                    <DoorOpen className="h-3 w-3" /> {user.room}
   875→                  </span>
   876→                )}
   877→              </div>
   878→              <div className="flex flex-wrap items-center gap-x-3 gap-y-0.5 mt-1.5 text-[11px] text-muted-foreground">
   879→                <span>Joined {format(new Date(user.createdAt), "MMM d, yyyy")}</span>
   880→                {user.lastLoginAt && !user.deletedAt && (
   881→                  <span>· Last login {formatDistanceToNow(new Date(user.lastLoginAt), { addSuffix: true })}</span>
   882→                )}
   883→              </div>
   884→              {user.deletedAt && user.deletionReason && (
   885→                <div className="mt-1.5 text-[11px] text-destructive/80 flex items-start gap-1">
   886→                  <AlertTriangle className="h-3 w-3 shrink-0 mt-0.5" />
   887→                  <span>Reason: {user.deletionReason}</span>
   888→                </div>
   889→              )}
   890→            </div>
   891→            <DropdownMenu>
   892→              <DropdownMenuTrigger asChild>
   893→                <GlassButton variant="ghost" size="icon" className="shrink-0" aria-label="User actions">
   894→                  <MoreVertical className="h-4 w-4" />
   895→                </GlassButton>
   896→              </DropdownMenuTrigger>
   897→              <DropdownMenuContent align="end" className="w-44 rounded-2xl">
   898→                <DropdownMenuLabel>Actions</DropdownMenuLabel>
   899→                <DropdownMenuSeparator />
   900→                {actions.map((a) => {
   901→                  const Icon = a.icon;
   902→                  return (
   903→                    <DropdownMenuItem
   904→                      key={a.action}
   905→                      onClick={() => onAction(a.action)}
   906→                      variant={a.variant}
   907→                      className="rounded-xl cursor-pointer"
   908→                    >
   909→                      <Icon className="h-4 w-4" />
   910→                      {a.label}
   911→                    </DropdownMenuItem>
   912→                  );
   913→                })}
   914→              </DropdownMenuContent>
   915→            </DropdownMenu>
   916→          </div>
   917→        </div>
   918→      </div>
   919→    </GlassCard>
   920→  );
   921→}
