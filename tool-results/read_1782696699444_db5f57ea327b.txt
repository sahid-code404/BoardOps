   192→function formatINR(n: number) {
   193→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   194→}
   195→
   196→function formatMonthYear(month: number, year: number) {
   197→  return `${MONTHS[month] ?? "—"} ${year}`;
   198→}
   199→
   200→function formatDate(iso: string | null) {
   201→  if (!iso) return "—";
   202→  return new Date(iso).toLocaleDateString("en-IN", {
   203→    day: "numeric",
   204→    month: "short",
   205→    year: "numeric",
   206→  });
   207→}
   208→
   209→/** Full date-time format: "29 June 2026, 05:28 am" */
   210→function formatDateTime(iso: string | null) {
   211→  if (!iso) return "—";
   212→  const d = new Date(iso);
   213→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric" });
   214→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   215→  return `${datePart}, ${timePart}`;
   216→}
   217→
   218→/** A bill is overdue if it has a due date in the past AND there's still an outstanding due amount. */
   219→function isOverdue(bill: Bill): boolean {
   220→  if (!bill.dueDate || bill.deletedAt) return false;
   221→  if (bill.status === "PAID" || bill.status === "VOID") return false;
   222→  return new Date(bill.dueDate) < new Date() && bill.dueAmount > 0;
   223→}

   932→function BillRow({
   933→  bill,
   934→  isAdmin,
   935→  onView,
   936→  onVoid,
   937→  onDelete,
   938→  onRestore,
   939→}: {
   940→  bill: Bill;
   941→  isAdmin: boolean;
   942→  onView: () => void;
   943→  onVoid: () => void;
   944→  onDelete: () => void;
   945→  onRestore: () => void;
   946→}) {
   947→  const isDeleted = !!bill.deletedAt;
   948→
   949→  // Build the actions list — same shape as UserRow's `actions` array.
   950→  const actions: { label: string; icon: typeof Eye; onClick: () => void; variant?: "destructive" }[] = [];

   952→    if (isAdmin) {
   953→      actions.push({ label: "Restore Bill", icon: RotateCcw, onClick: onRestore });
   954→    }
   955→  } else {
   956→    actions.push({ label: "View Details", icon: Eye, onClick: onView });
   957→    if (isAdmin && bill.status !== "VOID") {
   958→      actions.push({ label: "Void Bill", icon: Ban, onClick: onVoid, variant: "destructive" });
   959→    }
   960→    if (isAdmin) {
   961→      actions.push({ label: "Delete Bill", icon: Trash2, onClick: onDelete, variant: "destructive" });
   962→    }
   963→  }
   964→
   965→  return (
   966→    <GlassCard className="p-4" hover={false}>
   967→      <div className="flex items-start gap-3">
   968→        <Avatar className="h-10 w-10 rounded-xl shrink-0">
   969→          {bill.user.avatarUrl && <AvatarImage src={bill.user.avatarUrl} alt={bill.user.name} />}
   970→          <AvatarFallback className={cn("rounded-xl bg-gradient-to-br text-white font-semibold text-xs", gradientFor(bill.user.name))}>
   971→            {initials(bill.user.name) || "U"}
   972→          </AvatarFallback>
   973→        </Avatar>
   974→
   975→        <div className="flex-1 min-w-0">
   976→          <div className="flex items-start justify-between gap-2">
   977→            <div className="min-w-0">
   978→              <div className="flex items-center gap-2 flex-wrap">
   979→                <h3 className={cn("font-medium text-sm truncate", isDeleted && "text-muted-foreground line-through")}>
   980→                  {bill.user.name}
   981→                </h3>
   982→                {isDeleted ? (
   983→                  <Badge variant="outline" className="text-[10px] bg-destructive/15 text-destructive border-destructive/30">

   985→                  </Badge>
   986→                ) : (
   987→                  <>
   988→                    <Badge variant="outline" className={cn("text-[10px]", BILL_STATUS_STYLES[bill.status].className)}>
   989→                      {BILL_STATUS_STYLES[bill.status].label}
   990→                    </Badge>
   991→                    <Badge variant="outline" className="text-[10px] bg-muted text-muted-foreground border-border">
   992→                      <Calendar className="h-2.5 w-2.5" /> {formatDateTime(bill.generatedAt)}
   993→                    </Badge>
   994→                  </>
   995→                )}
   996→              </div>
   997→              {/* Transaction strip — Total / Paid / Due all the same size + dates */}
   998→              <div className="flex flex-wrap items-baseline gap-x-4 gap-y-1 mt-2">
   999→                <div className="flex items-baseline gap-1">
  1000→                  <span className="text-[10px] uppercase tracking-wide text-muted-foreground">Total</span>
  1001→                  <span className="text-base font-bold tabular-nums">{formatINR(bill.totalAmount)}</span>

  1005→                    <div className="flex items-baseline gap-1">

  1016→                  <span className="inline-flex items-start gap-1 text-[11px] text-destructive/80">
  1017→                    <AlertTriangle className="h-3 w-3 shrink-0 mt-0.5" />

  1019→                  </span>
  1020→                )}
  1021→              </div>
  1022→              {/* Due date — red if overdue, with OVERDUE badge */}
  1023→              {!isDeleted && bill.dueDate && (

  1044→                  <GlassButton variant="ghost" size="icon" className="shrink-0" aria-label="Bill actions">

  1050→                  <DropdownMenuSeparator />
  1051→                  {actions.map((a) => {
  1052→                    const Icon = a.icon;
  1053→                    return (
  1054→                      <DropdownMenuItem
  1055→                        key={a.label}
  1056→                        onClick={a.onClick}
  1057→                        variant={a.variant}
  1058→                        className="rounded-xl cursor-pointer"
  1059→                      >
  1060→                        <Icon className="h-4 w-4" />
  1061→                        {a.label}
  1062→                      </DropdownMenuItem>
  1063→                    );
  1064→                  })}
  1065→                </DropdownMenuContent>
  1066→              </DropdownMenu>
  1067→            )}
  1068→          </div>
  1069→        </div>
  1070→      </div>
  1071→    </GlassCard>
  1072→  );
  1073→}
