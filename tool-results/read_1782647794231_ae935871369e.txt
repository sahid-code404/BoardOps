   122→const AVATAR_GRADIENTS = [
   123→  "from-violet-500 to-fuchsia-500",
   124→  "from-emerald-500 to-teal-500",
   125→  "from-amber-500 to-orange-500",

   131→function gradientFor(name: string) {
   132→  const idx = name.split("").reduce((acc, c) => acc + c.charCodeAt(0), 0) % AVATAR_GRADIENTS.length;
   133→  return AVATAR_GRADIENTS[idx];
   134→}

   136→function initials(name: string) {
   137→  return name.split(" ").filter(Boolean).slice(0, 2).map((p) => p[0]?.toUpperCase() ?? "").join("");
   138→}

   194→function formatINR(n: number) {
   195→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   196→}
   197→
   198→function formatMonthYear(month: number, year: number) {
   199→  return `${MONTHS[month] ?? "—"} ${year}`;
   200→}
   201→
   202→function formatDate(iso: string | null) {
   203→  if (!iso) return "—";
   204→  return new Date(iso).toLocaleDateString("en-IN", {
   205→    day: "numeric",
   206→    month: "short",
   207→    year: "numeric",
   208→  });
   209→}

   215→export function BillingView() {
   216→  const user = useAuthStore((s) => s.user);
   217→  const setView = useAppStore((s) => s.setView);
   218→  const isAdmin =
   219→    user?.role === "ADMIN";
   220→
   221→  const qc = useQueryClient();
   222→  const [search, setSearch] = useState("");
   223→  const [statusFilter, setStatusFilter] = useState<BillStatus | "ALL" | "DELETED">("ALL");
   224→  const [generateOpen, setGenerateOpen] = useState(false);
   225→  const [genMonth, setGenMonth] = useState<number>(new Date().getMonth());
   226→  const [genYear, setGenYear] = useState<number>(new Date().getFullYear());
   227→  const [selectedBill, setSelectedBill] = useState<Bill | null>(null);
   228→  const [voidTarget, setVoidTarget] = useState<Bill | null>(null);
   229→  const now = new Date();
   230→  const [selectedMonth, setSelectedMonth] = useState(now.getMonth());
   231→  const [selectedYear, setSelectedYear] = useState(now.getFullYear());
   232→
   233→  const { data: bills = [], isLoading } = useQuery({
   234→    queryKey: ["bills", { month: selectedMonth, year: selectedYear }],
   235→    queryFn: async () => {
   236→      const r = await api.get<ApiResponse<Bill[]>>("/bills", {
   237→        params: { month: selectedMonth, year: selectedYear },
   238→      });
   239→      return r.data;
   240→    },
   241→  });
   242→
   243→  // Fetch soft-deleted bills (deletion queue) — admin only
   244→  const { data: deletedBills = [] } = useQuery({
   245→    queryKey: ["bills", "deleted", { month: selectedMonth, year: selectedYear }],
   246→    queryFn: async () => {
   247→      const r = await api.get<ApiResponse<Bill[]>>("/bills", {
   248→        params: { month: selectedMonth, year: selectedYear, includeDeleted: "true" },
   249→      });
   250→      return r.data;
   251→    },
   252→    enabled: isAdmin,
   253→  });
   254→
   255→  const generateMutation = useMutation({
   256→    mutationFn: () =>
   257→      api.post<ApiResponse<{ generated: number; month: number; year: number }>>(
   258→        "/bills",
   259→        { month: genMonth, year: genYear }
   260→      ),
   261→    onSuccess: (r) => {
   262→      toast.success(`Generated ${r.data.generated} bills for ${MONTHS[genMonth]} ${genYear}`);
   263→      setGenerateOpen(false);
   264→      qc.invalidateQueries({ queryKey: ["bills"] });
   265→    },
   266→    onError: (e: Error) => toast.error(e.message || "Failed to generate bills"),
   267→  });
   268→
   269→  const voidMutation = useMutation({
   270→    mutationFn: (id: string) =>
   271→      api.delete<ApiResponse<{ success: boolean }>>(`/bills/${id}`),
   272→    onSuccess: () => {
   273→      toast.success("Bill voided successfully");
   274→      setVoidTarget(null);
   275→      qc.invalidateQueries({ queryKey: ["bills"] });
   276→    },
   277→    onError: (e: Error) => toast.error(e.message || "Failed to void bill"),
   278→  });
   279→
   280→  const [deleteTarget, setDeleteTarget] = useState<Bill | null>(null);
   281→  const [deleteAllOpen, setDeleteAllOpen] = useState(false);
   282→  const [deleteReason, setDeleteReason] = useState("");
   283→  const [deleteAllReason, setDeleteAllReason] = useState("");
   284→
   285→  const deleteMutation = useMutation({
   286→    mutationFn: async (id: string) => {
   287→      await api.delete(`/bills/${id}`, { body: JSON.stringify({ reason: deleteReason || undefined }) });
   288→    },
   289→    onSuccess: () => {
   290→      toast.success("Bill scheduled for deletion — permanently removed in 7 days");
   291→      setDeleteTarget(null);
   292→      setDeleteReason("");
   293→      qc.invalidateQueries({ queryKey: ["bills"] });
   294→    },
   295→    onError: (e: Error) => toast.error(e.message || "Failed to delete bill"),
   296→  });
   297→
   298→  const deleteAllMutation = useMutation({
   299→    mutationFn: async () => {
   300→      const r = await api.delete<ApiResponse<{ deleted: number }>>("/bills", {
   301→        params: { month: selectedMonth, year: selectedYear },
   302→        body: JSON.stringify({ reason: deleteAllReason || undefined }),
   303→      });
   304→      return r.data;
   305→    },
   306→    onSuccess: (data) => {
   307→      toast.success(`${data.deleted} bills scheduled for deletion — permanently removed in 7 days`);
   308→      setDeleteAllOpen(false);
   309→      setDeleteAllReason("");
   310→      qc.invalidateQueries({ queryKey: ["bills"] });
   311→    },
   312→    onError: (e: Error) => toast.error(e.message || "Failed to delete bills"),
   313→  });
   314→
   315→  const restoreMutation = useMutation({
   316→    mutationFn: async (id: string) => {
   317→      const r = await api.post<ApiResponse<Bill>>(`/bills/${id}/restore`);
   318→      return r.data;
   319→    },
   320→    onSuccess: () => {
   321→      toast.success("Bill restored successfully");
   322→      qc.invalidateQueries({ queryKey: ["bills"] });
   323→    },
   324→    onError: (e: Error) => toast.error(e.message || "Failed to restore bill"),
   325→  });
   326→
   327→  const restoreAllMutation = useMutation({
   328→    mutationFn: async () => {
   329→      const results = await Promise.all(
   330→        deletedBills.map((b) => api.post(`/bills/${b.id}/restore`).catch(() => null))
   331→      );
   332→      return results.filter((r) => r !== null).length;
   333→    },
   334→    onSuccess: (count) => {
   335→      toast.success(`${count} bills restored successfully`);
   336→      qc.invalidateQueries({ queryKey: ["bills"] });
   337→    },
   338→    onError: (e: Error) => toast.error(e.message || "Failed to restore bills"),
   339→  });
   340→
   341→  // ── Derived KPIs ──
   342→  const kpis = useMemo(() => {
   343→    const active = bills.filter((b) => b.status !== "VOID");
   344→    const totalBilled = active.reduce((s, b) => s + b.totalAmount, 0);
   345→    const totalCollected = active.reduce((s, b) => s + b.paidAmount, 0);
   346→    const totalOutstanding = active.reduce((s, b) => s + b.dueAmount, 0);
   347→    const overdueCount = active.filter((b) => b.status === "OVERDUE").length;
   348→    return { totalBilled, totalCollected, totalOutstanding, overdueCount };
   349→  }, [bills]);
   350→
   351→  // ── Filtered list ──
   352→  // Use deleted bills when filter is DELETED, otherwise use active bills
   353→  const sourceBills = statusFilter === "DELETED" ? deletedBills : bills;
   354→
   355→  const filtered = useMemo(() => {
   356→    const q = search.trim().toLowerCase();
   357→    return sourceBills.filter((b) => {
   358→      if (statusFilter !== "ALL" && statusFilter !== "DELETED" && b.status !== statusFilter) return false;
   359→      if (!q) return true;
   360→      return (
   361→        b.user.name?.toLowerCase().includes(q) ||
   362→        b.user.email?.toLowerCase().includes(q) ||
   363→        (b.user.room || "").toLowerCase().includes(q)
   364→      );
   365→    });
   366→  }, [sourceBills, search, statusFilter]);
   367→
   368→  if (isLoading) {
   369→    return (
   370→      <div className="space-y-4">
   371→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   372→          {Array.from({ length: 4 }).map((_, i) => (
   373→            <ShimmerSkeleton key={i} className="h-32" />
   374→          ))}
   375→        </div>
   376→        <ShimmerSkeleton className="h-14 w-full" />
   377→        <div className="grid gap-3">
   378→          {Array.from({ length: 5 }).map((_, i) => (
   379→            <ShimmerSkeleton key={i} className="h-20" />
   380→          ))}
   381→        </div>
   382→      </div>
   383→    );
   384→  }
   385→
   386→  return (
   387→    <StaggerGroup className="space-y-4 md:space-y-5">
   388→      {/* Month picker — centered, capsule design (matches expenses page) */}
   389→      <StaggerItem>
   390→        <div className="flex items-center justify-center gap-4">
   391→          <motion.button
   392→            whileTap={{ scale: 0.9 }}
   393→            onClick={() => {
   394→              const d = new Date(selectedYear, selectedMonth - 1, 1);
   395→              setSelectedMonth(d.getMonth());
   396→              setSelectedYear(d.getFullYear());
   397→            }}
   398→            aria-label="Previous month"
   399→            className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   400→          >
   401→            <ChevronLeft className="h-5 w-5" />
   402→          </motion.button>
   403→
   404→          <div className="flex-1 max-w-[280px] flex items-center justify-center gap-2.5 glass-soft rounded-full px-6 py-2.5">
   405→            <Calendar className="h-4 w-4 text-primary shrink-0" />
   406→            <div className="leading-tight text-center">
   407→              <p className="text-sm font-bold text-primary">
   408→                {new Date(selectedYear, selectedMonth).toLocaleDateString("en-US", { month: "long" })}
   409→              </p>
   410→              <p className="text-[11px] text-muted-foreground">
   411→                {selectedYear}
   412→              </p>
   413→            </div>
   414→          </div>
   415→
   416→          <motion.button
   417→            whileTap={{ scale: 0.9 }}
   418→            onClick={() => {
   419→              const d = new Date(selectedYear, selectedMonth + 1, 1);
   420→              setSelectedMonth(d.getMonth());
   421→              setSelectedYear(d.getFullYear());
   422→            }}
   423→            aria-label="Next month"
   424→            className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   425→          >
   426→            <ChevronRight className="h-5 w-5" />
   427→          </motion.button>
   428→        </div>
   429→      </StaggerItem>
   430→
   431→      {/* Compact action bar */}
   432→      {isAdmin && (
   433→        <StaggerItem>
   434→          <div className="flex items-center justify-end gap-3">
   435→            <GlassButton
   436→              onClick={() => setGenerateOpen(true)}
   437→              className="shrink-0"
   438→            >
   439→              <Plus className="h-4 w-4" />
   440→              Generate Bills
   441→            </GlassButton>
   442→          </div>
   443→        </StaggerItem>
   444→      )}
   445→
   446→      {/* KPIs */}
   447→      <StaggerItem>
   448→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   449→          <KpiCard
   450→            label="Total Billed"
   451→            value={kpis.totalBilled}
   452→            icon={<Wallet className="h-5 w-5" />}
   453→            color="primary"
   454→            prefix="₹"
   455→          />
   456→          <KpiCard
   457→            label="Total Collected"
   458→            value={kpis.totalCollected}
   459→            icon={<CheckCircle2 className="h-5 w-5" />}
   460→            color="success"
   461→            prefix="₹"
   462→          />
   463→          <KpiCard
   464→            label="Outstanding"
   465→            value={kpis.totalOutstanding}
   466→            icon={<TrendingUp className="h-5 w-5" />}
   467→            color="warning"
   468→            prefix="₹"
   469→          />
   470→          <KpiCard
   471→            label="Overdue Count"
   472→            value={kpis.overdueCount}
   473→            icon={<AlertCircle className="h-5 w-5" />}
   474→            color="danger"
   475→          />
   476→        </div>
   477→      </StaggerItem>
   478→
   479→      {/* Search + Filter pills — all in a single horizontal scrollable line */}
   480→      <StaggerItem>
   481→        <div className="space-y-3">
   482→          <GlassInput
   483→            placeholder="Search by name, email, room…"
   484→            value={search}
   485→            onChange={(e) => setSearch(e.target.value)}
   486→            icon={<Search />}
   487→          />
   488→          <div className="flex items-center gap-1 overflow-x-auto no-scrollbar pb-1">
   489→            {(
   490→              [
   491→                "ALL",
   492→                "GENERATED",
   493→                "PARTIALLY_PAID",
   494→                "PAID",
   495→                "OVERDUE",
   496→                "VOID",
   497→                ...(isAdmin ? ["DELETED" as const] : []),
   498→              ] as const
   499→            ).map((s) => {
   500→              const active = statusFilter === s;
   501→              const label = s === "DELETED" ? "Deletion Queue" : s === "ALL" ? "All" : BILL_STATUS_STYLES[s as BillStatus]?.label || s;
   502→              const badge = s === "DELETED" && deletedBills.length > 0 ? deletedBills.length : null;
   503→              return (
   504→                <button
   505→                  key={s}
   506→                  onClick={() => setStatusFilter(s)}
   507→                  className={cn(
   508→                    "inline-flex items-center gap-1 h-8 px-2.5 rounded-xl text-[11px] font-medium whitespace-nowrap shrink-0 transition-all",
   509→                    active
   510→                      ? "bg-primary text-primary-foreground shadow-md shadow-primary/30"
   511→                      : "glass-soft text-muted-foreground hover:text-foreground"
   512→                  )}
   513→                >
   514→                  {label}
   515→                  {badge !== null && (
   516→                    <span className={cn(
   517→                      "text-[9px] rounded-full px-1.5 py-0.5 leading-none font-bold min-w-[16px] text-center",
   518→                      active && s !== "DELETED"
   519→                        ? "bg-primary-foreground/20 text-primary-foreground"
   520→                        : active && s === "DELETED"
   521→                          ? "bg-primary-foreground/20 text-primary-foreground"
   522→                          : "bg-destructive text-white"
   523→                    )}>
   524→                      {badge}
   525→                    </span>
   526→                  )}
   527→                </button>
   528→              );
   529→            })}
   530→            {/* Delete All — inline at the end of the same scrollable row */}
   531→            {isAdmin && statusFilter !== "DELETED" && bills.length > 0 && (
   532→              <button
   533→                onClick={() => setDeleteAllOpen(true)}
   534→                className="inline-flex items-center gap-1 h-8 px-2.5 rounded-xl text-[11px] font-medium whitespace-nowrap shrink-0 text-destructive/70 hover:text-destructive transition-colors"
   535→              >
   536→                <Trash2 className="h-3 w-3" />
   537→                Delete All ({bills.length})
   538→              </button>
   539→            )}
   540→            {/* Restore All — inline at the end of the same scrollable row (deletion queue only) */}
   541→            {isAdmin && statusFilter === "DELETED" && deletedBills.length > 0 && (
   542→              <button
   543→                onClick={() => restoreAllMutation.mutate()}
   544→                disabled={restoreAllMutation.isPending}
   545→                className="inline-flex items-center gap-1 h-8 px-2.5 rounded-xl text-[11px] font-medium whitespace-nowrap shrink-0 text-success/70 hover:text-success transition-colors"
   546→              >
   547→                <RotateCcw className="h-3 w-3" />
   548→                {restoreAllMutation.isPending ? "Restoring…" : `Restore All (${deletedBills.length})`}
   549→              </button>
   550→            )}
   551→          </div>
   552→        </div>
   553→      </StaggerItem>
   554→
   555→      {/* List */}
   556→      <StaggerItem>
   557→        {filtered.length === 0 ? (
   558→          <GlassCard className="p-10 md:p-16" hover={false}>
   559→            <div className="flex flex-col items-center justify-center text-center gap-3">
   560→              <div className="grid place-items-center h-14 w-14 rounded-3xl bg-muted/40">
   561→                <Receipt className="h-6 w-6 text-muted-foreground" />
   562→              </div>
   563→              <div>
   564→                <p className="font-medium">No bills found</p>

   827→function KpiCard({

   832→  prefix,

   871→function BillRow({
   872→  bill,
   873→  isAdmin,
   874→  onView,
   875→  onVoid,
   876→  onDelete,
   877→  onRestore,

   891→    if (isAdmin) {
   892→      actions.push({ label: "Restore Bill", icon: RotateCcw, onClick: onRestore });
   893→    }
   894→  } else {
   895→    actions.push({ label: "View Details", icon: Eye, onClick: onView });
   896→    if (isAdmin && bill.status !== "VOID") {
   897→      actions.push({ label: "Void Bill", icon: Ban, onClick: onVoid, variant: "destructive" });
   898→    }
   899→    if (isAdmin) {
   900→      actions.push({ label: "Delete Bill", icon: Trash2, onClick: onDelete, variant: "destructive" });
   901→    }
   902→  }
   903→
   904→  return (
   905→    <GlassCard className="p-4 md:p-5" hover={false}>
   906→      <div className="flex items-start gap-3 md:gap-4">
   907→        <Avatar className="h-12 w-12 md:h-14 md:w-14 rounded-2xl shrink-0">
   908→          {bill.user.avatarUrl && <AvatarImage src={bill.user.avatarUrl} alt={bill.user.name} />}
   909→          <AvatarFallback className={cn("rounded-2xl bg-gradient-to-br text-white font-semibold", gradientFor(bill.user.name))}>
   910→            {initials(bill.user.name) || "U"}

   912→        </Avatar>
   913→
   914→        <div className="flex-1 min-w-0">
   915→          <div className="flex items-start justify-between gap-2">
   916→            <div className="min-w-0">
   917→              <div className="flex items-center gap-2 flex-wrap">
   918→                <h3 className={cn("font-semibold truncate", isDeleted && "text-muted-foreground line-through")}>
   919→                  {bill.user.name}
   920→                </h3>
   921→                {isDeleted ? (
   922→                  <Badge variant="outline" className="text-[10px] bg-destructive/15 text-destructive border-destructive/30">
   923→                    <Clock className="h-2.5 w-2.5" /> {formatDeletionCountdown(new Date(bill.deletedAt!))}
   924→                  </Badge>
   925→                ) : (
   926→                  <>
   927→                    <Badge variant="outline" className={cn("text-[10px]", BILL_STATUS_STYLES[bill.status].className)}>
   928→                      {BILL_STATUS_STYLES[bill.status].label}
   929→                    </Badge>
   930→                    <Badge variant="outline" className="text-[10px] bg-muted text-muted-foreground border-border">
   931→                      <Calendar className="h-2.5 w-2.5" /> {formatMonthYear(bill.periodMonth, bill.periodYear)}
   932→                    </Badge>
   933→                  </>
   934→                )}
   935→              </div>
   936→              <div className="flex flex-col sm:flex-row sm:items-center gap-x-3 gap-y-0.5 mt-1.5 text-xs text-muted-foreground">
   937→                <span className="inline-flex items-center gap-1 truncate">
   938→                  <Mail className="h-3 w-3" /> {bill.user.email}
   939→                </span>
   940→                {bill.user.room && (
   941→                  <span className="inline-flex items-center gap-1">
   942→                    <DoorOpen className="h-3 w-3" /> Room {bill.user.room}
   943→                  </span>
   944→                )}
   945→                {!isDeleted && (
   946→                  <span className="inline-flex items-center gap-1">
   947→                    <Clock className="h-3 w-3" /> Due {formatDate(bill.dueDate)}
   948→                  </span>
   949→                )}
   950→              </div>

   952→              <div className="flex flex-wrap items-center gap-x-3 gap-y-0.5 mt-1.5 text-[11px] text-muted-foreground">
   953→                <span>
   954→                  Total <span className="font-semibold text-foreground tabular-nums">{formatINR(bill.totalAmount)}</span>
   955→                </span>
   956→                <span>
   957→                  Paid <span className="font-semibold text-success tabular-nums">{formatINR(bill.paidAmount)}</span>
   958→                </span>

   960→                  Due <span className="font-semibold text-warning tabular-nums">{formatINR(bill.dueAmount)}</span>

   962→                {bill.deletedAt && bill.deletionReason && (
   963→                  <span className="inline-flex items-start gap-1 text-destructive/80">
   964→                    <AlertTriangle className="h-3 w-3 shrink-0 mt-0.5" />

   966→                  </span>
   967→                )}
   968→              </div>
   969→            </div>
   970→
   971→            {/* Dropdown — only render if there are actions */}

   996→                </DropdownMenuContent>
   997→              </DropdownMenu>
   998→            )}
   999→          </div>
  1000→        </div>
  1001→      </div>
