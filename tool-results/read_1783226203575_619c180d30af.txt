   583→// PRD DEC-029: Partial refunds supported. A Refund tracks the full refund amount,
   584→// how much has been paid so far, and the remaining amount. Multiple partial
   585→// payments can be recorded until the refund is fully completed.
   586→model Refund {
   587→  id              String   @id @default(cuid())
   588→  refundNumber    String?  // Human-readable reference (e.g. REF-2026-0001)

   591→  billingCycleId  String?  // the cycle that generated this refund
   592→  // Full refund amount (the total the resident is owed)
   593→  amount          Float
   594→  // How much has been paid out so far (sum of all partial refund payments)
   595→  paidAmount      Float    @default(0)
   596→  // Remaining = amount - paidAmount (computed, but stored for query convenience)
   597→  remainingAmount Float    @default(0)
   598→  status          String   @default("PENDING") // PENDING | PARTIALLY_PAID | COMPLETED | CANCELLED
   599→  method          String?  // UPI | CASH | BANK_TRANSFER | CHEQUE
   600→  reference       String?  // transaction reference number
   601→  notes           String?
   602→  processedBy     String?
   603→  processedAt     DateTime?
   604→  completedAt     DateTime?
   605→  createdAt       DateTime @default(now())
   606→  updatedAt       DateTime @updatedAt
   607→  user            User     @relation(fields: [userId], references: [id], onDelete: Cascade)
   608→  bill            Bill?    @relation(fields: [billId], references: [id], onDelete: SetNull)
   609→  transactions    RefundTransaction[]
   610→
   611→  @@index([userId, status])
   612→  @@index([billingCycleId])
   613→}

   656→// ─────────────────────────────────────────────────────────────
   657→// FINANCIAL LEDGER (PRD: Resident Ledger — accounting backbone)
   658→// Every financial event creates a ledger entry. Balances are derived
   659→// from the sum of entries, never stored directly. This is the single
   660→// source of truth for a resident's financial position.
   661→// ─────────────────────────────────────────────────────────────
   662→
   663→model LedgerEntry {
   664→  id              String   @id @default(cuid())
   665→  userId          String
   666→  // What type of event
   667→  type            String   // DEPOSIT | BILL_SETTLEMENT | REFUND | ADJUSTMENT | PURCHASE | EXPENSE
   668→  // Credit/debit (from the resident's perspective)
   669→  // DEPOSIT: credit (+) — money in
   670→  // BILL_SETTLEMENT: debit (-) — money out (bill deducted)
   671→  // REFUND: debit (-) — money returned to resident
   672→  // ADJUSTMENT: credit or debit depending on sign
   673→  amount          Float    // positive = credit (adds to balance), negative = debit
   674→  // Running balance after this entry (for quick display — the authoritative
   675→  // balance is always the sum of all entries, but storing the running balance
   676→  // avoids re-summing on every query)
   677→  runningBalance  Float    @default(0)
   678→  // Reference to the source entity
   679→  entityType      String   // Payment | Bill | Refund | Adjustment | Expense | Purchase
   680→  entityId        String?
   681→  // Description for the human-readable ledger
   682→  description     String
   683→  // Billing cycle this entry applies to
   684→  billingMonth    Int?
   685→  billingYear     Int?
   686→  createdAt       DateTime @default(now())
   687→  user            User     @relation(fields: [userId], references: [id], onDelete: Cascade)
   688→  payment         Payment? @relation(fields: [entityId], references: [id], onDelete: SetNull)
   689→
   690→  @@index([userId, createdAt])
   691→  @@index([entityType, entityId])
   692→  @@index([billingMonth, billingYear])
   693→}
