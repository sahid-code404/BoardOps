     1→/**
     2→ * Monthly Closing Engine (PRD Module 13)
     3→ *
     4→ * Orchestrates the monthly financial closing workflow:
     5→ *   1. Validate readiness (meals locked, expenses approved, variables valid, formula valid)
     6→ *   2. Create immutable MonthlySnapshot (freeze meals, expenses, variables, formula)
     7→ *   3. Execute Formula Engine → calculate meal charge
     8→ *   4. Generate bills from the snapshot (not live data)
     9→ *   5. Settle resident fund accounts (deduct bill, determine refund/due)
    10→ *   6. Close the billing cycle
    11→ *
    12→ * Rollback is only allowed before bills are published (status < BILLS_GENERATED).
    13→ * After publication, corrections require adjustment entries (PRD DEC-033).
    14→ */
    15→

    20→const MONTHS = [
    21→  "January", "February", "March", "April", "May", "June",
    22→  "July", "August", "September", "October", "November", "December",
    23→];
    24→
    25→export function periodLabel(month: number, year: number): string {
    26→  return `${MONTHS[month] ?? `Month ${month + 1}`} ${year}`;
    27→}

    29→// ─────────────────────────────────────────────────────────────
    30→// 1. Readiness Checklist
    31→// ─────────────────────────────────────────────────────────────
    32→
    33→export type ReadinessItem = {
    34→  key: string;
    35→  label: string;
    36→  status: "ready" | "warning" | "error";
    37→  detail: string;
    38→  count?: number;
    39→};
    40→
    41→export type ReadinessResult = {
    42→  month: number;
    43→  year: number;
    44→  periodLabel: string;
    45→  items: ReadinessItem[];
    46→  canClose: boolean;
    47→  existingCycle?: {
    48→    id: string;
    49→    status: string;
    50→  } | null;
    51→};
    52→
    53→export async function getReadiness(month: number, year: number): Promise<ReadinessResult> {
    54→  const label = periodLabel(month, year);
    55→  const start = new Date(year, month, 1);
    56→  const end = new Date(year, month + 1, 0, 23, 59, 59, 999);
    57→
    58→  const items: ReadinessItem[] = [];
    59→
    60→  // Check 1: Existing billing cycle
    61→  const existingCycle = await db.billingCycle.findUnique({
    62→    where: { periodMonth_periodYear: { periodMonth: month, periodYear: year } },
    63→    select: { id: true, status: true },
    64→  });
    65→  if (existingCycle) {
    66→    if (existingCycle.status === "CLOSED") {
    67→      items.push({
    68→        key: "cycle",
    69→        label: "Billing Cycle",
    70→        status: "error",
    71→        detail: `This period is already CLOSED. Corrections require adjustment entries.`,
    72→      });
    73→    } else if (existingCycle.status === "BILLS_GENERATED" || existingCycle.status === "SETTLED") {
    74→      items.push({
    75→        key: "cycle",
    76→        label: "Billing Cycle",
    77→        status: "warning",
    78→        detail: `Cycle is in progress (status: ${existingCycle.status}). Bills already generated — rollback to restart.`,
    79→      });
    80→    } else {
    81→      items.push({
    82→        key: "cycle",
    83→        label: "Billing Cycle",
    84→        status: "ready",
    85→        detail: `Cycle started (status: ${existingCycle.status}).`,
    86→      });
    87→    }
    88→  } else {
    89→    items.push({
    90→      key: "cycle",
    91→      label: "Billing Cycle",
    92→      status: "ready",
    93→      detail: "No existing cycle — ready to start.",
    94→    });
    95→  }
    96→
    97→  // Check 2: Active residents
    98→  const activeUsers = await db.user.count({
    99→    where: { status: "ACTIVE", role: "USER", deletedAt: null },
   100→  });
   101→  items.push({
   102→    key: "residents",
   103→    label: "Active Residents",
   104→    status: activeUsers > 0 ? "ready" : "error",
   105→    detail: activeUsers > 0
   106→      ? `${activeUsers} resident(s) will be billed.`
   107→      : "No active residents — cannot generate bills.",
   108→    count: activeUsers,
   109→  });
   110→
   111→  // Check 3: Meal entries for this period (exclude admin users)
   112→  const mealEntries = await db.mealEntry.count({
   113→    where: {
   114→      serviceDate: { gte: start, lte: end },
   115→      status: { in: ["ON", "LOCKED"] },
   116→      user: { role: "USER" },
   117→    },
   118→  });
   119→  const mealConfigs = await db.mealConfiguration.count({ where: { status: "ACTIVE" } });
   120→  if (mealConfigs === 0) {
   121→    items.push({
   122→      key: "meals",
   123→      label: "Meal Configuration",
   124→      status: "error",
   125→      detail: "No active meals configured. Configure meals first.",
   126→    });
   127→  } else if (mealEntries === 0) {
   128→    items.push({
   129→      key: "meals",
   130→      label: "Meal Entries",
   131→      status: "warning",
   132→      detail: `No ON/LOCKED meal entries for ${label}. Bills will have ₹0 meal charges.`,
   133→    });
   134→  } else {
   135→    items.push({
   136→      key: "meals",
   137→      label: "Meal Entries",
   138→      status: "ready",
   139→      detail: `${mealEntries} meal entry(ies) for ${label}.`,
   140→      count: mealEntries,
   141→    });
   142→  }
   143→
   144→  // Check 4: Expenses for this period
   145→  const expenses = await db.expense.count({
   146→    where: {
   147→      expenseDate: { gte: start, lte: end },
   148→      deletedAt: null,
   149→      status: { not: "DELETED" },
   150→    },
   151→  });
   152→  const totalExpenseAmount = await db.expense.aggregate({
   153→    where: {
   154→      expenseDate: { gte: start, lte: end },
   155→      deletedAt: null,
   156→      status: { not: "DELETED" },
   157→    },
   158→    _sum: { amount: true },
   159→  });
   160→  const expenseTotal = totalExpenseAmount._sum.amount ?? 0;
   161→  if (expenses === 0) {
   162→    items.push({
   163→      key: "expenses",
   164→      label: "Expenses",
   165→      status: "warning",
   166→      detail: `No expenses recorded for ${label}. Meal charge may be ₹0.`,
   167→    });
   168→  } else {
   169→    items.push({
   170→      key: "expenses",
   171→      label: "Expenses",
   172→      status: "ready",
   173→      detail: `${expenses} expense(s) totaling ₹${Math.round(expenseTotal).toLocaleString("en-IN")}.`,
   174→      count: expenses,
   175→    });
   176→  }
   177→
   178→  // Check 5: Variables
   179→  const activeVars = await db.variable.count({
   180→    where: { status: "ACTIVE" },
   181→  });
   182→  const systemVars = await db.variable.count({
   183→    where: { status: "ACTIVE", isSystem: true },
   184→  });
   185→  if (activeVars === 0) {
   186→    items.push({
   187→      key: "variables",
   188→      label: "Variables",
   189→      status: "error",
   190→      detail: "No active variables. The formula engine needs variables to calculate charges.",
   191→    });
   192→  } else {
   193→    items.push({
   194→      key: "variables",
   195→      label: "Variables",
   196→      status: "ready",
   197→      detail: `${activeVars} active variable(s) (${systemVars} system, ${activeVars - systemVars} custom).`,
   198→      count: activeVars,
   199→    });
   200→  }
   201→
   202→  // Check 6: Active formula (meal charge)
   203→  const mealChargeFormula = await db.formula.findFirst({
   204→    where: { key: "formula.mealCharges", status: "ACTIVE" },
   205→  });
   206→  if (!mealChargeFormula) {
   207→    items.push({
   208→      key: "formula",
   209→      label: "Meal Charge Formula",
   210→      status: "warning",
   211→      detail: "No active formula.mealCharges formula. Bills will use the legacy rate×count calculation.",
   212→    });
   213→  } else {
   214→    // Validate the formula expression
   215→    const validation = validateFormula(mealChargeFormula.expression);
   216→    if (!validation.valid) {
   217→      items.push({
   218→        key: "formula",
   219→        label: "Meal Charge Formula",
   220→        status: "warning",
   221→        detail: `formula.mealCharges (v${mealChargeFormula.version}) is invalid: ${validation.error}. Bills will use the legacy rate×count calculation.`,
   222→      });
   223→    } else {
   224→      items.push({
   225→        key: "formula",
   226→        label: "Meal Charge Formula",
   227→        status: "ready",
   228→        detail: `formula.mealCharges v${mealChargeFormula.version} is valid.`,
   229→      });
   230→    }
   231→  }
   232→
   233→  // Check 7: Pending payments (informational — not a blocker)
   234→  const pendingPayments = await db.payment.count({
   235→    where: {
   236→      status: "PENDING",
   237→      createdAt: { gte: start, lte: end },
   238→    },
   239→  });
   240→  items.push({
   241→    key: "payments",
   242→    label: "Pending Payments",
   243→    status: pendingPayments > 0 ? "warning" : "ready",
   244→    detail: pendingPayments > 0
   245→      ? `${pendingPayments} payment(s) pending approval. They will apply to the NEXT billing cycle if approved after closing.`
   246→      : "No pending payments.",
   247→    count: pendingPayments,
   248→  });
   249→
   250→  // Bill generation requires ALL items to be "ready" — no errors AND no warnings.
   251→  // Admins must resolve all warnings (e.g. add missing expenses, fix invalid formula,
   252→  // approve pending payments) before they can close the billing cycle.
   253→  // PRD: Bills can only be generated for PAST months — the current month hasn't ended yet,
   254→  // so its meal data is still incomplete.
   255→  const now = new Date();
   256→  const currentPeriod = now.getFullYear() * 12 + now.getMonth();
   257→  const selectedPeriod = year * 12 + month;
   258→  const isCurrentOrFutureMonth = selectedPeriod >= currentPeriod;
   259→
   260→  if (isCurrentOrFutureMonth) {
   261→    items.unshift({
   262→      key: "period",
   263→      label: "Billing Period",
   264→      status: "error",
   265→      detail: `Cannot generate bills for ${label} — this month has not ended yet. Bills can only be generated for past months after the month is complete.`,
   266→    });
   267→  }
   268→
   269→  const canClose = items.every((i) => i.status === "ready") && (!existingCycle || existingCycle.status !== "CLOSED");
   270→
   271→  return {
   272→    month,
   273→    year,
   274→    periodLabel: label,
   275→    items,
   276→    canClose,
   277→    existingCycle,
   278→  };
   279→}

   281→// ─────────────────────────────────────────────────────────────
   282→// 2. Create Snapshot
   283→// ─────────────────────────────────────────────────────────────
   284→
   285→type SnapshotData = {
   286→  mealsData: string;
   287→  expensesData: string;
   288→  variablesData: string;
   289→  formulaData: string;
   290→  totalExpenses: number;
   291→  totalResidentMeals: number;
   292→  totalGuestMeals: number;
   293→  guestRevenue: number;
   294→  mealCharge: number;
   295→};
   296→
   297→async function createSnapshot(month: number, year: number): Promise<SnapshotData> {
   298→  const start = new Date(year, month, 1);
   299→  const end = new Date(year, month + 1, 0, 23, 59, 59, 999);
   300→
   301→  // ── Meals data: per-resident meal counts ──
   302→  const activeUsers = await db.user.findMany({
   303→    where: { status: "ACTIVE", role: "USER", deletedAt: null },
   304→    select: { id: true, name: true, room: true },
   305→  });
   306→  const meals = await db.mealConfiguration.findMany({ where: { status: "ACTIVE" } });
   307→  const mealNameById: Record<string, string> = {};
   308→  meals.forEach((m) => (mealNameById[m.id] = m.name));
   309→
   310→  const mealEntries = await db.mealEntry.findMany({
   311→    where: { serviceDate: { gte: start, lte: end }, user: { role: "USER" } },
   312→  });
   313→  const guestMeals = await db.guestMeal.findMany({
   314→    where: { serviceDate: { gte: start, lte: end } },
   315→  });
   316→
   317→  // Per-resident meal counts
   318→  const residentMeals: Record<string, Record<string, number>> = {};
   319→  let totalResidentMeals = 0;
   320→  for (const u of activeUsers) {
   321→    residentMeals[u.id] = {};
   322→  }
   323→  for (const e of mealEntries) {
   324→    if (e.status === "ON" || e.status === "LOCKED") {
   325→      const name = mealNameById[e.mealId] || "unknown";
   326→      if (!residentMeals[e.userId]) residentMeals[e.userId] = {};
   327→      residentMeals[e.userId][name] = (residentMeals[e.userId][name] || 0) + 1;
   328→      totalResidentMeals++;
   329→    }
   330→  }
   331→
   332→  // Guest meals per meal type
   333→  const guestMealCounts: Record<string, number> = {};
   334→  let totalGuestMeals = 0;
   335→  for (const g of guestMeals) {
   336→    const name = mealNameById[g.mealId] || "unknown";
   337→    guestMealCounts[name] = (guestMealCounts[name] || 0) + (g.guestCount || 1);
   338→    totalGuestMeals += g.guestCount || 1;
   339→  }
   340→
   341→  const mealsData = JSON.stringify({
   342→    residentMeals,
   343→    guestMealCounts,
   344→    totalResidentMeals,
   345→    totalGuestMeals,
   346→    mealConfigs: meals.map((m) => ({ id: m.id, name: m.name, displayName: m.displayName })),
   347→  });
   348→
   349→  // ── Expenses data ──
   350→  const expenses = await db.expense.findMany({
   351→    where: {
   352→      expenseDate: { gte: start, lte: end },
   353→      deletedAt: null,
   354→      status: { not: "DELETED" },
   355→    },
   356→    include: {
   357→      purchase: { include: { items: true } },
   358→    },
   359→  });
   360→  const totalExpenses = expenses.reduce((s, e) => s + e.amount, 0);
   361→  const expensesData = JSON.stringify({
   362→    expenses: expenses.map((e) => ({
   363→      id: e.id,
   364→      title: e.title,
   365→      category: e.category,
   366→      amount: e.amount,
   367→      expenseDate: e.expenseDate,
   368→      purchase: e.purchase
   369→        ? {
   370→            vendor: e.purchase.vendor,
   371→            items: e.purchase.items.map((it) => ({
   372→              name: it.productName,
   373→              qty: it.quantity,
   374→              unit: it.unit,
   375→              rate: it.rate,
   376→              total: it.total,
   377→            })),
   378→          }
   379→        : null,
   380→    })),
   381→    totalExpenses,
   382→  });
   383→
   384→  // ── Variables data ──
   385→  const variables = await db.variable.findMany({
   386→    where: { status: "ACTIVE" },
   387→  });
   388→  const variablesData = JSON.stringify(
   389→    Object.fromEntries(variables.map((v) => [v.key, { value: v.value, type: v.type, name: v.name }]))
   390→  );
   391→
   392→  // ── Formula data ──
   393→  const formulas = await db.formula.findMany({
   394→    where: { status: "ACTIVE" },
   395→    select: { key: true, name: true, expression: true, version: true, returnType: true },
   396→  });
   397→  const formulaData = JSON.stringify(
   398→    Object.fromEntries(formulas.map((f) => [f.key, { name: f.name, expression: f.expression, version: f.version, returnType: f.returnType }]))
   399→  );
   400→
   401→  // ── Calculate meal charge using the Formula Engine ──
   402→  // Build the variable resolver from the snapshot (not live DB — for consistency)
   403→  const varMap = new Map(variables.map((v) => [v.key, parseFloat(v.value) || 0]));
   404→  const resolver: FormulaVarResolver = (slug: string) => varMap.get(slug) ?? 0;
   405→
   406→  // Also inject computed totals as variables the formula can reference
   407→  const computedVars: Record<string, number> = {
   408→    total_expense: totalExpenses,
   409→    total_resident_meals: totalResidentMeals,
   410→    total_guest_meals: totalGuestMeals,
   411→    guest_revenue: 0, // computed below
   412→  };
   413→
   414→  // Calculate guest revenue: sum of guest meals × their meal rate
   415→  let guestRevenue = 0;
   416→  for (const [mealName, count] of Object.entries(guestMealCounts)) {
   417→    const rateVar = varMap.get(`meal.rate.${mealName.toLowerCase()}`);
   418→    if (rateVar !== undefined) {
   419→      guestRevenue += rateVar * count;
   420→    }
   421→  }
   422→  computedVars.guest_revenue = guestRevenue;
   423→
   424→  // Merge computed vars into the resolver
   425→  const fullResolver: FormulaVarResolver = (slug: string) => {
   426→    if (slug in computedVars) return computedVars[slug];
   427→    return varMap.get(slug) ?? 0;
   428→  };
   429→
   430→  // Try to evaluate the meal charge formula
   431→  const mealChargeFormula = formulas.find((f) => f.key === "formula.mealCharges");
   432→  let mealCharge = 0;
   433→  if (mealChargeFormula && totalResidentMeals > 0) {
   434→    const result = evaluateFormula(mealChargeFormula.expression, fullResolver);
   435→    if (!result.error) {
   436→      mealCharge = result.value;
   437→    }
   438→  }
   439→
   440→  return {
   441→    mealsData,
   442→    expensesData,
   443→    variablesData,
   444→    formulaData,
   445→    totalExpenses,
   446→    totalResidentMeals,
   447→    totalGuestMeals,
   448→    guestRevenue,
   449→    mealCharge,
   450→  };
   451→}

   457→export type ClosingResult = {
   458→  success: boolean;
   459→  cycleId: string;
   460→  status: string;
   461→  summary: {
   462→    totalExpenses: number;
   463→    totalResidentMeals: number;
   464→    totalGuestMeals: number;
   465→    guestRevenue: number;
   466→    mealCharge: number;
   467→    billsGenerated: number;
   468→    refundQueueTotal: number;
   469→    outstandingDue: number;
   470→  };
   471→  error?: string;
   472→};
   473→
   474→export async function executeClosing(
   475→  month: number,
   476→  year: number,
   477→  adminId: string,
   478→  dueDate?: Date
   479→): Promise<ClosingResult> {
   480→  const label = periodLabel(month, year);
   481→  const start = new Date(year, month, 1);
   482→  const end = new Date(year, month + 1, 0, 23, 59, 59, 999);
   483→
   484→  // 1. Check readiness
   485→  const readiness = await getReadiness(month, year);
   486→  if (!readiness.canClose) {
   487→    const errors = readiness.items.filter((i) => i.status === "error").map((i) => `${i.label}: ${i.detail}`);
   488→    return {
   489→      success: false,
   490→      cycleId: "",
   491→      status: "OPEN",
   492→      summary: { totalExpenses: 0, totalResidentMeals: 0, totalGuestMeals: 0, guestRevenue: 0, mealCharge: 0, billsGenerated: 0, refundQueueTotal: 0, outstandingDue: 0 },
   493→      error: `Cannot close: ${errors.join("; ")}`,
   494→    };
   495→  }
   496→
   497→  // 2. Get or create the billing cycle
   498→  let cycle = readiness.existingCycle
   499→    ? await db.billingCycle.findUnique({ where: { id: readiness.existingCycle.id } })
   500→    : null;
   501→
   502→  if (cycle && cycle.status === "CLOSED") {
   503→    return {
   504→      success: false,
   505→      cycleId: cycle.id,
   506→      status: cycle.status,
   507→      summary: { totalExpenses: 0, totalResidentMeals: 0, totalGuestMeals: 0, guestRevenue: 0, mealCharge: 0, billsGenerated: 0, refundQueueTotal: 0, outstandingDue: 0 },
   508→      error: "This billing cycle is already CLOSED. Corrections require adjustment entries.",
   509→    };
   510→  }
   511→
   512→  // Create or update the cycle to PREPARING status
   513→  if (!cycle) {
   514→    cycle = await db.billingCycle.create({
   515→      data: {
   516→        periodMonth: month,
   517→        periodYear: year,
   518→        status: "PREPARING",
   519→        startedBy: adminId,
   520→        startedAt: new Date(),
   521→        readiness: JSON.stringify(readiness.items),
   522→      },
   523→    });
   524→  } else {
   525→    cycle = await db.billingCycle.update({
   526→      where: { id: cycle.id },
   527→      data: {
   528→        status: "PREPARING",
   529→        startedBy: adminId,
   530→        startedAt: new Date(),
   531→        readiness: JSON.stringify(readiness.items),
   532→        errorMessage: null,
   533→      },
   534→    });
   535→  }
   536→
   537→  try {
   538→    // 3. Create the snapshot
   539→    const snapshotData = await createSnapshot(month, year);
   540→
   541→    // Delete any existing snapshot (from a previous failed attempt)
   542→    if (cycle.snapshotId) {
   543→      await db.monthlySnapshot.deleteMany({ where: { billingCycleId: cycle.id } });
   544→    }
   545→
   546→    const snapshot = await db.monthlySnapshot.create({
   547→      data: {
   548→        billingCycleId: cycle.id,
   549→        mealsData: snapshotData.mealsData,
   550→        expensesData: snapshotData.expensesData,
   551→        variablesData: snapshotData.variablesData,
   552→        formulaData: snapshotData.formulaData,
   553→        totalExpenses: snapshotData.totalExpenses,
   554→        totalResidentMeals: snapshotData.totalResidentMeals,
   555→        totalGuestMeals: snapshotData.totalGuestMeals,
   556→        guestRevenue: snapshotData.guestRevenue,
   557→        mealCharge: snapshotData.mealCharge,
   558→      },
   559→    });
   560→
   561→    cycle = await db.billingCycle.update({
   562→      where: { id: cycle.id },
   563→      data: {
   564→        status: "SNAPSHOT_CREATED",
   565→        snapshotId: snapshot.id,
   566→        totalExpenses: snapshotData.totalExpenses,
   567→        totalMeals: snapshotData.totalResidentMeals,
   568→        totalGuestMeals: snapshotData.totalGuestMeals,
   569→        mealCharge: snapshotData.mealCharge,
   570→      },
   571→    });
   572→
   573→    // 4. Generate bills from the snapshot
   574→    const activeUsers = await db.user.findMany({
   575→      where: { status: "ACTIVE", role: "USER", deletedAt: null },
   576→    });
   577→
   578→    // Load room rent + cleaning from the snapshot variables (not live DB)
   579→    const snapshotVars = JSON.parse(snapshotData.variablesData) as Record<string, { value: string; type: string }>;
   580→    const roomRent = parseFloat(snapshotVars["billing.roomRent"]?.value || "0") || 0;
   581→    const cleaning = parseFloat(snapshotVars["billing.cleaningCharges"]?.value || "0") || 0;
   582→
   583→    // PRD: Meal charge is a SINGLE dynamic value = (Total Expenses - Guest Revenue) / Total Resident Meals
   584→    // NOT fixed per-meal rates. The snapshot already calculated this.
   585→    const perMealCharge = snapshotData.mealCharge;
   586→
   587→    // PRD DEC-012: store the formula version + expression on each bill for reproducibility
   588→    const snapshotFormulas = JSON.parse(snapshotData.formulaData) as Record<string, { name: string; expression: string; version: number }>;
   589→    const mealChargeFormulaInfo = snapshotFormulas["formula.mealCharges"] || null;
   590→
   591→    const snapshotMeals = JSON.parse(snapshotData.mealsData) as {
   592→      residentMeals: Record<string, Record<string, number>>;
   593→    };
   594→
   595→    // Read due date day from policy variable (default 10th)
   596→    const dueDateDayVar = await db.variable.findUnique({ where: { key: "policy.billing.dueDateDay" } });
   597→    const dueDateDay = dueDateDayVar ? parseInt(dueDateDayVar.value) || 10 : 10;
   598→    const effectiveDueDate = dueDate ?? new Date(year, month + 1, dueDateDay);
   599→    let billsGenerated = 0;
   600→    let refundQueueTotal = 0;
   601→    let outstandingDue = 0;
   602→
   603→    for (const u of activeUsers) {
   604→      const counts = snapshotMeals.residentMeals[u.id] || {};
   605→      // PRD: meal charges = resident's total meal count × per-meal charge (dynamic, not fixed rates)
   606→      const residentMealCount = Object.values(counts).reduce((s, c) => s + c, 0);
   607→      const mealCharges = Math.round(residentMealCount * perMealCharge);
   608→      const otherCharges = roomRent + cleaning;
   609→      const totalAmount = mealCharges + otherCharges;
   610→
   611→      // PRD DEC-027: previous due tracked separately (not added to totalAmount)
   612→      const previousDue = await getPreviousDue(u.id, month, year);
   613→
   614→      // Check existing bill (for re-runs)
   615→      const existing = await db.bill.findUnique({
   616→        where: { userId_periodMonth_periodYear: { userId: u.id, periodMonth: month, periodYear: year } },
   617→      });
   618→
   619→      if (existing && (existing.status === "VOID" || existing.deletedAt)) {
   620→        continue; // skip VOID/deleted
   621→      }
   622→
   623→      const billSnapshot = JSON.stringify({
   624→        counts,
   625→        residentMealCount,
   626→        perMealCharge,
   627→        mealCharges,
   628→        roomRent,
   629→        cleaning,
   630→        totalExpenses: snapshotData.totalExpenses,
   631→        guestRevenue: snapshotData.guestRevenue,
   632→        totalResidentMeals: snapshotData.totalResidentMeals,
   633→        snapshotId: snapshot.id,
   634→        formulaVersion: mealChargeFormulaInfo?.version ?? null,
   635→        formulaExpression: mealChargeFormulaInfo?.expression ?? null,
   636→        closedAt: new Date().toISOString(),
   637→      });
   638→
   639→      if (existing) {
   640→        // Preserve paidAmount, recompute due + status
   641→        const paidAmount = existing.paidAmount;
   642→        const dueAmount = Math.max(0, totalAmount - paidAmount);
   643→        let newStatus: string;
   644→        if (totalAmount > 0 && paidAmount >= totalAmount) {
   645→          newStatus = "PAID";
   646→        } else if (paidAmount > 0) {
   647→          newStatus = "PARTIALLY_PAID";
   648→        } else {
   649→          newStatus = "GENERATED";
   650→        }
   651→        await db.bill.update({
   652→          where: { id: existing.id },
   653→          data: {
   654→            mealCharges,
   655→            otherCharges,
   656→            totalAmount,
   657→            dueAmount,
   658→            previousDue,
   659→            status: newStatus,
   660→            generatedAt: new Date(),
   661→            dueDate: effectiveDueDate,
   662→            snapshot: billSnapshot,
   663→            billingCycleId: cycle.id,
   664→            // Formula snapshot (only set if not already set — preserves the original generation's formula)
   665→            ...(existing.formulaKey ? {} : {
   666→              formulaKey: mealChargeFormulaInfo ? "formula.mealCharges" : null,
   667→              formulaVersion: mealChargeFormulaInfo?.version ?? null,
   668→              formulaExpression: mealChargeFormulaInfo?.expression ?? null,
   669→            }),
   670→            // Bill number (only set if not already assigned)
   671→            ...(existing.billNumber ? {} : { billNumber: await generateBillNumber() }),
   672→          },
   673→        });
   674→      } else {
   675→        await db.bill.create({
   676→          data: {
   677→            billNumber: await generateBillNumber(),
   678→            userId: u.id,
   679→            periodMonth: month,
   680→            periodYear: year,
   681→            mealCharges,
   682→            otherCharges,
   683→            totalAmount,
   684→            paidAmount: 0,
   685→            dueAmount: totalAmount,
   686→            previousDue,
   687→            status: "GENERATED",
   688→            generatedAt: new Date(),
   689→            dueDate: effectiveDueDate,
   690→            snapshot: billSnapshot,
   691→            billingCycleId: cycle.id,
   692→            formulaKey: mealChargeFormulaInfo ? "formula.mealCharges" : null,
   693→            formulaVersion: mealChargeFormulaInfo?.version ?? null,
   694→            formulaExpression: mealChargeFormulaInfo?.expression ?? null,
   695→          },
   696→        });
   697→      }
   698→      billsGenerated++;
   699→
   700→      // Settlement: if paid > total, create a Refund record for the overpaid user
   701→      const paidAmount = existing?.paidAmount ?? 0;
   702→      if (paidAmount > totalAmount) {
   703→        const refundAmount = paidAmount - totalAmount;
   704→        refundQueueTotal += refundAmount;
   705→        // Create a refund record for this overpaid user
   706→        const existingRefund = await db.refund.findFirst({
   707→          where: { userId: u.id, billingCycleId: cycle.id, status: { in: ["PENDING", "PARTIALLY_PAID"] } },
   708→        });
   709→        if (!existingRefund) {
   710→          const { generateRefundNumber } = await import("@/lib/reference-numbers");
   711→          await db.refund.create({
   712→            data: {
   713→              refundNumber: await generateRefundNumber(),
   714→              userId: u.id,
   715→              billId: existing?.id ?? null,
   716→              billingCycleId: cycle.id,
   717→              amount: refundAmount,
   718→              paidAmount: 0,
   719→              remainingAmount: refundAmount,
   720→              status: "PENDING",
   721→              notes: `Auto-created refund: overpayment after bill generation for ${periodLabel(month, year)}`,
   722→              processedBy: adminId,
   723→              processedAt: new Date(),
   724→            },
   725→          });
   726→        }
   727→      } else if (totalAmount > paidAmount) {
   728→        outstandingDue += (totalAmount - paidAmount);
   729→      }
   730→    }
   731→
   732→    // 5. Lock expenses for this period (PRD DEC-030 — expenses become immutable after snapshot)
   733→    const lockedCount = await lockExpensesForPeriod(month, year, cycle.id);
   734→
   735→    // 6. Update cycle to BILLS_GENERATED
   736→    cycle = await db.billingCycle.update({
   737→      where: { id: cycle.id },
   738→      data: {
   739→        status: "BILLS_GENERATED",
   740→        billsGenerated,
   741→        refundQueueTotal,
   742→        outstandingDue,
   743→      },
   744→    });
   745→
   746→    // 6. Settle: mark as SETTLED then CLOSED
   747→    cycle = await db.billingCycle.update({
   748→      where: { id: cycle.id },
   749→      data: {
   750→        status: "SETTLED",
   751→      },
   752→    });
   753→
   754→    cycle = await db.billingCycle.update({
   755→      where: { id: cycle.id },
   756→      data: {
   757→        status: "CLOSED",
   758→        closedBy: adminId,
   759→        closedAt: new Date(),
   760→      },
   761→    });
   762→
   763→    return {
   764→      success: true,
   765→      cycleId: cycle.id,
   766→      status: "CLOSED",
   767→      summary: {
   768→        totalExpenses: snapshotData.totalExpenses,
   769→        totalResidentMeals: snapshotData.totalResidentMeals,
   770→        totalGuestMeals: snapshotData.totalGuestMeals,
   771→        guestRevenue: snapshotData.guestRevenue,
   772→        mealCharge: snapshotData.mealCharge,
   773→        billsGenerated,
   774→        refundQueueTotal,
   775→        outstandingDue,
   776→      },
   777→    };
   778→  } catch (e) {
   779→    // Mark cycle as FAILED
   780→    await db.billingCycle.update({
   781→      where: { id: cycle.id },
   782→      data: {
   783→        status: "FAILED",
   784→        errorMessage: e instanceof Error ? e.message : String(e),
   785→      },
   786→    });
   787→    return {
   788→      success: false,
   789→      cycleId: cycle.id,
   790→      status: "FAILED",
   791→      summary: { totalExpenses: 0, totalResidentMeals: 0, totalGuestMeals: 0, guestRevenue: 0, mealCharge: 0, billsGenerated: 0, refundQueueTotal: 0, outstandingDue: 0 },
   792→      error: e instanceof Error ? e.message : String(e),
   793→    };
   794→  }
   795→}

   797→// ─────────────────────────────────────────────────────────────
   798→// 4. Rollback (only before bills are published)
   799→// ─────────────────────────────────────────────────────────────
   800→
   801→export async function rollbackCycle(cycleId: string): Promise<{ success: boolean; error?: string }> {
   802→  const cycle = await db.billingCycle.findUnique({ where: { id: cycleId } });
   803→  if (!cycle) return { success: false, error: "Billing cycle not found" };
   804→
   805→  // Can only rollback before bills are generated
   806→  if (cycle.status === "BILLS_GENERATED" || cycle.status === "SETTLED" || cycle.status === "CLOSED") {
   807→    return {
   808→      success: false,
   809→      error: "Cannot rollback after bills have been generated. Corrections require adjustment entries (PRD DEC-033).",
   810→    };
   811→  }
   812→
   813→  // Delete the snapshot
   814→  if (cycle.snapshotId) {
   815→    await db.monthlySnapshot.deleteMany({ where: { billingCycleId: cycleId } });
   816→  }
   817→
   818→  // Reset the cycle to OPEN
   819→  await db.billingCycle.update({
   820→    where: { id: cycleId },
   821→    data: {
   822→      status: "OPEN",
   823→      snapshotId: null,
   824→      readiness: null,
   825→      startedBy: null,
   826→      startedAt: null,
   827→      errorMessage: null,
   828→      totalExpenses: 0,
   829→      totalMeals: 0,
   830→      totalGuestMeals: 0,
   831→      mealCharge: 0,
   832→      billsGenerated: 0,
   833→      refundQueueTotal: 0,
   834→      outstandingDue: 0,
   835→    },
   836→  });
   837→
   838→  return { success: true };
   839→}
