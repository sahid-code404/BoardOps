   220→function formatDate(iso: string) {
   221→  return new Date(iso).toLocaleDateString("en-IN", {

   224→    year: "numeric",

   228→/** Format quantity + unit for display (e.g. "5 kg", "2 piece"). */
   229→function formatQuantity(qty: number, unit: string): string {
   230→  if (!qty && !unit) return "";
   231→  if (!unit) return String(qty);
   232→  if (!qty) return unit;
   233→  return `${qty} ${unit}`;
   234→}

   236→/**
   237→ * An expense is locked when its status is LOCKED, it's in the deletion queue,
   238→ * or it belongs to a past month (bills may have been generated against it).
   239→ */
   240→function isExpenseLocked(expense: Expense): boolean {
   241→  if (expense.status === "LOCKED") return true;
   242→  if (expense.deletedAt) return true; // in deletion queue — can't edit
   243→  const expDate = new Date(expense.expenseDate);
   244→  const now = new Date();
   245→  const expYM = expDate.getFullYear() * 12 + expDate.getMonth();
   246→  const todayYM = now.getFullYear() * 12 + now.getMonth();
   247→  return expYM < todayYM;
   248→}

  1009→function ExpenseFormBody({
  1010→  expense,
  1011→  onSubmit,
  1012→  loading,
  1013→  onCancel,
  1014→}: {
  1015→  expense: Expense | null;
  1016→  onSubmit: (payload: ExpensePayload, id?: string) => void;
  1017→  loading: boolean;
  1018→  onCancel: () => void;
  1019→}) {
  1020→  const today = new Date().toISOString().slice(0, 10);
  1021→  const isEdit = !!expense;
  1022→
  1023→  // Derive initial category (predefined or CUSTOM) from the expense.
  1024→  const initialCategory: ExpenseCategory = (() => {
  1025→    if (!expense?.category) return "GROCERY";
  1026→    if (
  1027→      CATEGORY_ORDER.includes(expense.category as ExpenseCategory) &&
  1028→      expense.category !== "CUSTOM"
  1029→    ) {
  1030→      return expense.category as ExpenseCategory;
  1031→    }
  1032→    return "CUSTOM";
  1033→  })();
  1034→  const initialCustomCategory =
  1035→    initialCategory === "CUSTOM" ? expense?.category ?? "" : "";
  1036→
  1037→  // Derive initial unit (predefined or CUSTOM) from the expense.
  1038→  const initialUnit: string = (() => {
  1039→    if (!expense?.unit) return "piece";
  1040→    if (UNIT_OPTIONS.includes(expense.unit)) return expense.unit;
  1041→    return "CUSTOM";
  1042→  })();
  1043→  const initialCustomUnit = initialUnit === "CUSTOM" ? expense?.unit ?? "" : "";
  1044→
  1045→  const [title, setTitle] = useState(expense?.title ?? "");
  1046→  const [amount, setAmount] = useState(
  1047→    expense?.amount ? String(expense.amount) : ""
  1048→  );
  1049→  const [quantity, setQuantity] = useState(
  1050→    expense?.quantity ? String(expense.quantity) : ""
  1051→  );
  1052→  const [category, setCategory] = useState<ExpenseCategory>(initialCategory);
  1053→  const [customCategory, setCustomCategory] = useState(initialCustomCategory);
  1054→  const [unit, setUnit] = useState<string>(initialUnit);
  1055→  const [customUnit, setCustomUnit] = useState(initialCustomUnit);
  1056→  const [date, setDate] = useState(
  1057→    expense
  1058→      ? new Date(expense.expenseDate).toISOString().slice(0, 10)
  1059→      : today
  1060→  );
  1061→  const [description, setDescription] = useState(expense?.description ?? "");
  1062→  const [errors, setErrors] = useState<Record<string, string>>({});
  1063→
  1064→  function handleSubmit() {
  1065→    const next: Record<string, string> = {};
  1066→    if (!title.trim() || title.trim().length < 2) next.title = "Item name is required";
  1067→    const amt = parseFloat(amount);
  1068→    if (!amt || amt <= 0) next.amount = "Enter a valid cost";
  1069→    if (!date) next.date = "Date is required";
  1070→    if (category === "CUSTOM" && customCategory.trim().length < 2) {
  1071→      next.customCategory = "Enter a custom category name";
  1072→    }
  1073→    const qty = quantity ? parseFloat(quantity) : 0;
  1074→    if (quantity && (!qty || qty <= 0)) next.quantity = "Enter a valid quantity";
  1075→    if (unit === "CUSTOM" && customUnit.trim().length < 1) {
  1076→      next.customUnit = "Enter a custom unit";
  1077→    }
  1078→    setErrors(next);
  1079→    if (Object.keys(next).length > 0) return;
  1080→
  1081→    const finalCategory =
  1082→      category === "CUSTOM"
  1083→        ? customCategory.trim().toUpperCase().replace(/\s+/g, "_")
  1084→        : category;
  1085→    const finalUnit = unit === "CUSTOM" ? customUnit.trim() : unit;
  1086→
  1087→    onSubmit(
  1088→      {
  1089→        title: title.trim(),
  1090→        amount: amt,
  1091→        category: finalCategory,
  1092→        quantity: qty || 0,
  1093→        unit: finalUnit,
  1094→        expenseDate: new Date(date).toISOString(),
  1095→        description: description.trim() || undefined,
  1096→      },
  1097→      isEdit ? expense!.id : undefined
  1098→    );
  1099→  }
  1100→
  1101→  return (
  1102→    <>
  1103→      <SheetHeader className="px-6 pt-6 pb-2">
  1104→        <SheetTitle className="flex items-center gap-2 text-xl">
  1105→          <PencilLine className="h-5 w-5 text-primary" />
  1106→          {isEdit ? "Edit Expense" : "Add Expense"}
  1107→        </SheetTitle>
  1108→        <SheetDescription>
  1109→          {isEdit
  1110→            ? "Update the details of this expense."
  1111→            : "Record a new operational expense. It will be visible immediately."}
  1112→        </SheetDescription>
  1113→      </SheetHeader>
  1114→
  1115→      <div className="flex-1 overflow-y-auto px-6 py-4 space-y-3 no-scrollbar">
  1116→        <GlassInput
  1117→          label="Item"
  1118→          placeholder="e.g. Monthly groceries"
  1119→          value={title}
  1120→          onChange={(e) => setTitle(e.target.value)}
  1121→          error={errors.title}
  1122→          icon={<Receipt />}
  1123→        />
  1124→
  1125→        <div className="space-y-1.5">
  1126→          <label className="text-xs font-medium text-muted-foreground ml-1">
  1127→            Category
  1128→          </label>
  1129→          <Select
  1130→            value={category}
  1131→            onValueChange={(v) => setCategory(v as ExpenseCategory)}
  1132→          >
  1133→            <SelectTrigger className="w-full h-11 rounded-2xl">
  1134→              <SelectValue />
  1135→            </SelectTrigger>
  1136→            <SelectContent>
  1137→              {CATEGORY_ORDER.map((c) => (
  1138→                <SelectItem key={c} value={c}>
  1139→                  {CATEGORY_META[c].label}
  1140→                </SelectItem>
  1141→              ))}
  1142→            </SelectContent>
  1143→          </Select>
  1144→          {category === "CUSTOM" && (
  1145→            <GlassInput
  1146→              placeholder="Enter custom category name…"
  1147→              value={customCategory}
  1148→              onChange={(e) => setCustomCategory(e.target.value)}
  1149→              error={errors.customCategory}
  1150→              icon={<PencilLine className="h-4 w-4" />}
  1151→            />
  1152→          )}
  1153→        </div>
