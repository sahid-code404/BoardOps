   292→   292→          {Array.from({ length: 5 }).map((_, i) => (

   307→   307→              Track & manage operational expenses

   336→   336→

   458→   458→                  )}

   460→   460→                  {meta?.icon}

   462→   462→                </button>
   463→   463→              );
   464→   464→              });
   465→   465→            })()}
   466→   466→          </div>

   468→   468→      </StaggerItem>
   469→   469→
   470→   470→      {/* List */}
   471→   471→      <StaggerItem>

   474→   474→            <div className="flex flex-col items-center justify-center text-center gap-3">
   475→   475→              <div className="grid place-items-center h-14 w-14 rounded-3xl bg-muted/40">
   476→   476→                <Receipt className="h-6 w-6 text-muted-foreground" />

   750→   750→function AddExpenseSheet({
   751→   751→  open,
   752→   752→  onOpenChange,
   753→   753→  onSubmit,
   754→   754→  loading,
   755→   755→}: {
   756→   756→  open: boolean;
   757→   757→  onOpenChange: (o: boolean) => void;
   758→   758→  onSubmit: (payload: {
   759→   759→    title: string;
   760→   760→    amount: number;
   761→   761→    category: ExpenseCategory;
   762→   762→    expenseDate: string;
   763→   763→    paidTo?: string;
   764→   764→    description?: string;
   765→   765→  }) => void;
   766→   766→  loading: boolean;
   767→   767→}) {

   774→   774→  const [paidTo, setPaidTo] = useState("");

   779→   779→    setTitle("");
   780→   780→    setAmount("");
   781→   781→    setCategory("GROCERY");

   810→   810→      amount: amt,

   832→   832→        </SheetHeader>
   833→   833→

   842→   842→          />
