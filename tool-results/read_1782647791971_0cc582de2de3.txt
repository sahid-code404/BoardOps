    85→  category: string;
    86→  isSystem: boolean;
    87→  isProtected: boolean;
    88→  status: string;

   166→// ─────────────────────────────────────────────────────────────

   169→
   170→export function VariablesView() {
   171→  const user = useAuthStore((s) => s.user);
   172→  const queryClient = useQueryClient();
   173→  const isAdmin = user?.role === "ADMIN";
   174→
   175→  const [search, setSearch] = useState("");
   176→  const [typeFilter, setTypeFilter] = useState<string>("ALL");
   177→  const [catFilter, setCatFilter] = useState<string>("ALL");
   178→  const [createOpen, setCreateOpen] = useState(false);
   179→
   180→  const { data: resp, isLoading } = useQuery({
   181→    queryKey: ["variables"],
   182→    queryFn: () => api.get<ApiResponse<Variable[]>>("/variables"),
   183→  });
   184→
   185→  const variables = resp?.data ?? [];
   186→
   187→  const createMutation = useMutation({
   188→    mutationFn: (data: CreateForm) =>
   189→      api.post<ApiResponse<Variable>>("/variables", data),
   190→    onSuccess: () => {
   191→      toast.success("Variable created successfully");
   192→      queryClient.invalidateQueries({ queryKey: ["variables"] });
   193→      setCreateOpen(false);
   194→    },
   195→    onError: (e: { message?: string }) =>
   196→      toast.error(e.message || "Failed to create variable"),
   197→  });
   198→
   199→  const updateMutation = useMutation({
   200→    mutationFn: ({ id, value }: { id: string; value: string }) =>
   201→      api.put<ApiResponse<Variable>>(`/variables/${id}`, { value }),
   202→    onSuccess: () => {
   203→      toast.success("Variable updated");
   204→      queryClient.invalidateQueries({ queryKey: ["variables"] });
   205→    },
   206→    onError: (e: { message?: string }) =>
   207→      toast.error(e.message || "Failed to update variable"),
   208→  });
   209→
   210→  const deleteMutation = useMutation({
   211→    mutationFn: (id: string) =>
   212→      api.delete<ApiResponse<{ success: boolean }>>(`/variables/${id}`),
   213→    onSuccess: () => {
   214→      toast.success("Variable archived");
   215→      queryClient.invalidateQueries({ queryKey: ["variables"] });
   216→    },
   217→    onError: (e: { message?: string }) =>
   218→      toast.error(e.message || "Failed to delete variable"),
   219→  });
   220→
   221→  // Categories — derive from existing variables plus presets
   222→  const categories = useMemo(() => {
   223→    const set = new Set<string>(PRESET_CATEGORIES);
   224→    variables.forEach((v) => set.add(v.category));
   225→    return Array.from(set).sort();
   226→  }, [variables]);
   227→
   228→  // Filter + group
   229→  const grouped = useMemo(() => {
   230→    const filtered = variables.filter((v) => {
   231→      if (search) {
   232→        const q = search.toLowerCase();
   233→        const hit =
   234→          v.name.toLowerCase().includes(q) ||
   235→          v.key.toLowerCase().includes(q) ||
   236→          (v.description ?? "").toLowerCase().includes(q) ||
   237→          v.category.toLowerCase().includes(q);
   238→        if (!hit) return false;
   239→      }
   240→      if (typeFilter !== "ALL" && v.type !== typeFilter) return false;
   241→      if (catFilter !== "ALL" && v.category !== catFilter) return false;
   242→      return true;
   243→    });
   244→
   245→    const groups: Record<string, Variable[]> = {};
   246→    filtered.forEach((v) => {
   247→      if (!groups[v.category]) groups[v.category] = [];
   248→      groups[v.category].push(v);
   249→    });
   250→    return Object.entries(groups).sort((a, b) => a[0].localeCompare(b[0]));
   251→  }, [variables, search, typeFilter, catFilter]);
   252→
   253→  const stats = {
   254→    total: variables.length,
   255→    system: variables.filter((v) => v.isSystem).length,
   256→    custom: variables.filter((v) => !v.isSystem).length,
   257→    categories: new Set(variables.map((v) => v.category)).size,
   258→  };
   259→
   260→  if (isLoading) return <VariablesSkeleton />;
   261→
   262→  return (
   263→    <StaggerGroup className="space-y-4 md:space-y-6">
   264→      {/* Action bar */}
   265→      {isAdmin && (
   266→        <StaggerItem>
   267→          <div className="flex items-center justify-end gap-3">
   268→            <GlassButton onClick={() => setCreateOpen(true)} className="shrink-0">
   269→              <Plus className="h-4 w-4" />
   270→              Create Variable
   271→            </GlassButton>
   272→          </div>
   273→        </StaggerItem>
   274→      )}
   275→
   276→      {/* Stats bar */}
   277→      <StaggerItem>
   278→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   279→          <StatCard
   280→            icon={Database}
   281→            label="Total"
   282→            value={stats.total}
   283→            color="primary"
   284→          />
   285→          <StatCard
   286→            icon={Shield}
   287→            label="System"
   288→            value={stats.system}
   289→            color="info"
   290→          />
   291→          <StatCard
   292→            icon={Sparkles}
   293→            label="Custom"
   294→            value={stats.custom}
   295→            color="success"
   296→          />
   297→          <StatCard
   298→            icon={Filter}
   299→            label="Categories"
   300→            value={stats.categories}
   301→            color="warning"
   302→          />
   303→        </div>
   304→      </StaggerItem>
   305→
   306→      {/* Search + filters */}
   307→      <StaggerItem>
   308→        <GlassCard className="p-4" hover={false}>
   309→          <div className="flex flex-col md:flex-row gap-3">
   310→            <div className="flex-1">
   311→              <GlassInput
   312→                value={search}
   313→                onChange={(e) => setSearch(e.target.value)}
   314→                placeholder="Search by name, key, or description..."
   315→                icon={<Search />}
   316→              />
   317→            </div>
   318→            <Select value={typeFilter} onValueChange={setTypeFilter}>
   319→              <SelectTrigger className="w-full md:w-44 h-12 glass-soft rounded-2xl border-2 border-transparent">
   320→                <Filter className="h-4 w-4 mr-1 text-muted-foreground" />
   321→                <SelectValue placeholder="Type" />
   322→              </SelectTrigger>
   323→              <SelectContent>
   324→                <SelectItem value="ALL">All types</SelectItem>
   325→                <SelectItem value="NUMBER">Number</SelectItem>
   326→                <SelectItem value="CURRENCY">Currency</SelectItem>
   327→                <SelectItem value="PERCENTAGE">Percentage</SelectItem>
   328→                <SelectItem value="TEXT">Text</SelectItem>
   329→                <SelectItem value="BOOLEAN">Boolean</SelectItem>
   330→              </SelectContent>
   331→            </Select>
   332→            <Select value={catFilter} onValueChange={setCatFilter}>
   333→              <SelectTrigger className="w-full md:w-44 h-12 glass-soft rounded-2xl border-2 border-transparent">
   334→                <Tag className="h-4 w-4 mr-1 text-muted-foreground" />
   335→                <SelectValue placeholder="Category" />
   336→              </SelectTrigger>
   337→              <SelectContent>
   338→                <SelectItem value="ALL">All categories</SelectItem>
   339→                {categories.map((c) => (
   340→                  <SelectItem key={c} value={c}>
   341→                    {c}
   342→                  </SelectItem>
   343→                ))}
   344→              </SelectContent>
   345→            </Select>
   346→          </div>
   347→        </GlassCard>
   348→      </StaggerItem>
   349→
   350→      {/* Grouped list */}
   351→      {grouped.length === 0 ? (
   352→        <StaggerItem>
   353→          <GlassCard className="p-10 md:p-14 text-center" hover={false}>
   354→            <div className="grid place-items-center h-16 w-16 rounded-3xl bg-muted/40 mx-auto mb-4">
   355→              <Database className="h-8 w-8 text-muted-foreground/60" />
   356→            </div>
   357→            <h3 className="font-semibold text-lg mb-1">
   358→              {variables.length === 0
   359→                ? "No variables yet"
   360→                : "No matching variables"}
   361→            </h3>
   362→            <p className="text-sm text-muted-foreground max-w-sm mx-auto">
   363→              {variables.length === 0
   364→                ? "Create your first variable to start configuring system behaviour."
   365→                : "Try adjusting your search or filter criteria."}
   366→            </p>
   367→            {isAdmin && variables.length === 0 && (
   368→              <GlassButton
   369→                className="mt-4"
   370→                onClick={() => setCreateOpen(true)}
   371→              >
   372→                <Plus className="h-4 w-4" /> Create Variable
   373→              </GlassButton>
   374→            )}
   375→          </GlassCard>
   376→        </StaggerItem>
   377→      ) : (
   378→        <StaggerItem>
   379→          <Accordion
   380→            type="multiple"
   381→            defaultValue={grouped[0] ? [grouped[0][0]] : []}
   382→            className="space-y-3"
   383→          >
   384→            {grouped.map(([category, vars]) => (
   385→              <AccordionItem
   386→                key={category}
   387→                value={category}
   388→                className="glass rounded-3xl overflow-hidden border-b-0 px-4 md:px-5"
   389→              >
   390→                <AccordionTrigger className="hover:no-underline py-4">
   391→                  <div className="flex items-center gap-3">
   392→                    <div className="grid place-items-center h-9 w-9 rounded-2xl bg-primary/15">
   393→                      <Database className="h-4 w-4 text-primary" />
   394→                    </div>
   395→                    <div className="text-left">
   396→                      <div className="font-semibold text-sm tracking-wide">
   397→                        {category}
   398→                      </div>
   399→                      <div className="text-[11px] text-muted-foreground">
   400→                        {vars.length} variable
   401→                        {vars.length === 1 ? "" : "s"} ·{" "}
   402→                        {vars.filter((v) => v.isSystem).length} system
   403→                      </div>
   404→                    </div>
   405→                  </div>
   406→                </AccordionTrigger>
   407→                <AccordionContent>
   408→                  <div className="grid grid-cols-1 lg:grid-cols-2 gap-3 pb-3 pt-1">
   409→                    {vars.map((v) => (
   410→                      <VariableCard
   411→                        key={v.id}
   412→                        variable={v}
   413→                        isAdmin={isAdmin}
   414→                        onSave={(value) =>
   415→                          updateMutation.mutate({ id: v.id, value })
   416→                        }
   417→                        onDelete={() => deleteMutation.mutate(v.id)}
   418→                        saving={
   419→                          updateMutation.isPending &&
   420→                          updateMutation.variables?.id === v.id
   421→                        }
   422→                        deleting={
   423→                          deleteMutation.isPending &&
   424→                          deleteMutation.variables === v.id
   425→                        }
   426→                      />
   427→                    ))}
   428→                  </div>
   429→                </AccordionContent>
   430→              </AccordionItem>
   431→            ))}
   432→          </Accordion>
   433→        </StaggerItem>
   434→      )}
   435→
   436→      {/* Create dialog */}
   437→      <CreateVariableDialog
   438→        open={createOpen}
   439→        onOpenChange={setCreateOpen}
   440→        onSubmit={(d) => createMutation.mutate(d)}
   441→        loading={createMutation.isPending}
   442→        categories={categories}
   443→      />
   444→    </StaggerGroup>
   445→  );
   446→}

   452→function StatCard({
   453→  icon: Icon,
   454→  label,
   455→  value,
   456→  color,
   457→}: {

   459→  label: string;
   460→  value: number;

   466→      : color === "info"

   473→      className="p-4 relative overflow-hidden"

   480→      />

   501→// ─────────────────────────────────────────────────────────────
   502→// Variable card with inline edit
   503→// ─────────────────────────────────────────────────────────────
   504→
   505→function VariableCard({
   506→  variable,
   507→  isAdmin,
   508→  onSave,
   509→  onDelete,
   510→  saving,
   511→  deleting,
   512→}: {
   513→  variable: Variable;
   514→  isAdmin: boolean;
   515→  onSave: (value: string) => void;
   516→  onDelete: () => void;
   517→  saving: boolean;
   518→  deleting: boolean;
   519→}) {
   520→  const [editing, setEditing] = useState(false);
   521→  const [draft, setDraft] = useState(variable.value);
   522→
   523→  // NOTE: draft is re-synced from variable.value each time the user opens the
   524→  // editor (see startEdit below). External refetches that update variable.value
   525→  // while not editing don't affect the (hidden) draft, which is the desired
   526→  // behaviour.
   527→
   528→  const typeMeta = TYPE_META[variable.type] ?? TYPE_META.TEXT;
   529→  const TypeIcon = typeMeta.icon;
   530→  const isBool = variable.type === "BOOLEAN";
   531→  const boolValue = variable.value === "true";
   532→
   533→  const startEdit = () => {
   534→    setDraft(variable.value);
   535→    setEditing(true);
   536→  };
   537→  const cancelEdit = () => {
   538→    setDraft(variable.value);
   539→    setEditing(false);
   540→  };
   541→  const save = () => {
   542→    if (draft === variable.value) {
   543→      setEditing(false);
   544→      return;
   545→    }
   546→    onSave(draft);
   547→    setEditing(false);
   548→  };
   549→
   550→  const canEditValue = isAdmin;
   551→  const canDelete = isAdmin && !variable.isProtected && !variable.isSystem;
   552→
   553→  return (
   554→    <motion.div
   555→      whileHover={{ y: -2 }}
   556→      transition={{ type: "spring", stiffness: 300, damping: 22 }}
   557→      className="glass-soft rounded-3xl p-4 relative overflow-hidden"
   558→    >
   559→      <div className="flex items-start justify-between gap-3">
   560→        <div className="flex-1 min-w-0">
   561→          <div className="flex items-center gap-2 mb-1 flex-wrap">
   562→            <h4 className="font-semibold text-sm truncate">
   563→              {variable.name}
   564→            </h4>
   565→            <span
   566→              className={cn(
   567→                "inline-flex items-center gap-1 text-[10px] px-2 py-0.5 rounded-full font-medium",
   568→                typeMeta.tint
   569→              )}
   570→            >
   571→              <TypeIcon className="h-3 w-3" />
   572→              {variable.type}
   573→            </span>
   574→            {variable.isSystem && (
   575→              <Badge
   576→                variant="secondary"
   577→                className="h-5 text-[10px] gap-1 px-1.5"
   578→              >
   579→                <Shield className="h-3 w-3" /> System
   580→              </Badge>
   581→            )}
   582→            {variable.isProtected && (
   583→              <Badge
   584→                variant="outline"
   585→                className="h-5 text-[10px] gap-1 px-1.5 border-warning/40 text-warning"
   586→              >
   587→                <Lock className="h-3 w-3" /> Protected
   588→              </Badge>
   589→            )}
   590→          </div>
   591→          <div className="flex items-center gap-1.5 mb-2">
   592→            <code className="text-[11px] font-mono text-muted-foreground bg-muted/40 px-1.5 py-0.5 rounded">
   593→              {variable.key}
   594→            </code>
   595→            {variable.unit && (
   596→              <span className="text-[10px] text-muted-foreground">
   597→                · unit: <strong>{variable.unit}</strong>
   598→              </span>
   599→            )}
   600→          </div>
   601→          {variable.description && (
   602→            <p className="text-xs text-muted-foreground/80 line-clamp-2 mb-2">
   603→              {variable.description}
   604→            </p>
   605→          )}
   606→        </div>
   607→      </div>
   608→
   609→      {/* Value display / inline edit */}
   610→      <div className="mt-2">
   611→        {isBool ? (
   612→          // BOOLEAN — switch (only editable by admin)
   613→          <div className="flex items-center justify-between glass rounded-2xl px-3 py-2">
   614→            <span className="text-[11px] text-muted-foreground">
   615→              Current state
   616→            </span>
   617→            <Switch
   618→              checked={boolValue}
   619→              disabled={!canEditValue || saving}
   620→              onCheckedChange={(c) => {
   621→                if (canEditValue) onSave(c ? "true" : "false");
   622→              }}
   623→            />
   624→          </div>
   625→        ) : editing ? (
   626→          <div className="flex items-center gap-2">
   627→            <input
   628→              autoFocus
   629→              value={draft}
   630→              onChange={(e) => setDraft(e.target.value)}
   631→              onKeyDown={(e) => {
   632→                if (e.key === "Enter") save();
   633→                if (e.key === "Escape") cancelEdit();
   634→              }}
   635→              className="flex-1 bg-transparent glass rounded-2xl px-3 py-2 text-sm font-medium tabular-nums outline-none focus:ring-2 focus:ring-primary/40"
   636→            />
   637→            <GlassButton
   638→              size="icon"
   639→              variant="success"
   640→              onClick={save}
   641→              loading={saving}
   642→              aria-label="Save"
   643→            >
   644→              <Check className="h-4 w-4" />
   645→            </GlassButton>
   646→            <GlassButton
   647→              size="icon"
   648→              variant="ghost"
   649→              onClick={cancelEdit}
   650→              aria-label="Cancel"
   651→            >
   652→              <X className="h-4 w-4" />
   653→            </GlassButton>
   654→          </div>
   655→        ) : (
   656→          <div className="flex items-center justify-between glass rounded-2xl px-3 py-2">
   657→            <div className="flex items-baseline gap-1.5 min-w-0">
   658→              <span className="text-[11px] uppercase tracking-wider text-muted-foreground">
   659→                Value
   660→              </span>
   661→              <span className="text-base font-semibold tabular-nums truncate">
   662→                {variable.value}
   663→              </span>
   664→              {variable.unit && (
   665→                <span className="text-xs text-muted-foreground">
   666→                  {variable.unit}
   667→                </span>
   668→              )}
   669→            </div>
   670→            {canEditValue && (
   671→              <GlassButton
   672→                size="icon"
   673→                variant="ghost"
   674→                onClick={startEdit}
   675→                aria-label="Edit value"
   676→                className="h-8 w-8"
   677→              >
   678→                <Pencil className="h-3.5 w-3.5" />
   679→              </GlassButton>
   680→            )}
   681→          </div>
   682→        )}
   683→      </div>
   684→
   685→      {/* Footer actions */}
   686→      {canDelete && (
   687→        <div className="mt-2 flex justify-end">
   688→          <GlassButton
   689→            size="sm"
   690→            variant="ghost"
   691→            onClick={onDelete}
   692→            loading={deleting}
   693→            className="text-destructive hover:text-destructive"
   694→          >
   695→            <Trash2 className="h-3.5 w-3.5" />
   696→            Archive
   697→          </GlassButton>
   698→        </div>
   699→      )}
   700→    </motion.div>
   701→  );
   702→}

   708→function CreateVariableDialog({
   709→  open,
   710→  onOpenChange,
   711→  onSubmit,
   712→  loading,
   713→  categories,
   714→}: {
   715→  open: boolean;
   716→  onOpenChange: (v: boolean) => void;
   717→  onSubmit: (data: CreateForm) => void;
   718→  loading: boolean;
   719→  categories: string[];
   720→}) {
   721→  const form = useForm<CreateForm>({
   722→    resolver: zodResolver(createSchema),
   723→    defaultValues: {
   724→      key: "",
   725→      name: "",
   726→      type: "NUMBER",
   727→      value: "",
   728→      unit: "",
   729→      category: "GENERAL",
   730→      description: "",
   731→    },
   732→  });
   733→
   734→  const type = form.watch("type");
   735→  const isBool = type === "BOOLEAN";
   736→
   737→  useEffect(() => {
   738→    if (!open) {
   739→      form.reset();
   740→    }
   741→  }, [open, form]);
   742→
   743→  const submit = (data: CreateForm) => {
   744→    onSubmit(data);
   745→  };
   746→
   747→  return (
   748→    <Dialog open={open} onOpenChange={onOpenChange}>
   749→      <DialogContent className="glass-strong rounded-3xl max-w-md max-h-[90vh] overflow-y-auto no-scrollbar">
   750→        <DialogHeader>
   751→          <DialogTitle className="flex items-center gap-2">
   752→            <Plus className="h-5 w-5 text-primary" />
   753→            Create Variable
   754→          </DialogTitle>
   755→          <DialogDescription>
   756→            Add a new configurable variable to the system. Keys must be unique.
   757→          </DialogDescription>
   758→        </DialogHeader>
   759→
   760→        <form
   761→          onSubmit={form.handleSubmit(submit)}
   762→          className="space-y-4 pt-2"
   763→        >
   764→          <GlassInput
   765→            label="Name"
   766→            placeholder="e.g., Breakfast Rate"
   767→            error={form.formState.errors.name?.message}
   768→            {...form.register("name")}
   769→          />
   770→
   771→          <GlassInput
   772→            label="Key"
   773→            placeholder="e.g., breakfast_rate"
   774→            hint="Letters, numbers, dots, dashes, underscores only"
   775→            error={form.formState.errors.key?.message}
   776→            trailing={
   777→              <code className="text-[10px] font-mono text-muted-foreground">
   778→                /^[a-z0-9_.-]+$/i
   779→              </code>
   780→            }
   781→            {...form.register("key")}
   782→          />
   783→
   784→          <div>
   785→            <label className="mb-1.5 ml-1 block text-xs font-medium text-muted-foreground">
   786→              Type
   787→            </label>
   788→            <Controller
   789→              control={form.control}
   790→              name="type"
   791→              render={({ field }) => (
   792→                <Select
   793→                  value={field.value}
   794→                  onValueChange={field.onChange}
   795→                >
   796→                  <SelectTrigger className="w-full h-12 glass-soft rounded-2xl border-2 border-transparent">
   797→                    <SelectValue placeholder="Select type" />
   798→                  </SelectTrigger>
   799→                  <SelectContent>
   800→                    <SelectItem value="NUMBER">Number</SelectItem>
   801→                    <SelectItem value="CURRENCY">Currency</SelectItem>
   802→                    <SelectItem value="PERCENTAGE">Percentage</SelectItem>
   803→                    <SelectItem value="TEXT">Text</SelectItem>
   804→                    <SelectItem value="BOOLEAN">Boolean</SelectItem>
   805→                  </SelectContent>
   806→                </Select>
   807→              )}
   808→            />
   809→          </div>
   810→
   811→          {isBool ? (
   812→            <div>
   813→              <label className="mb-1.5 ml-1 block text-xs font-medium text-muted-foreground">
   814→                Value
   815→              </label>
   816→              <Controller
   817→                control={form.control}
   818→                name="value"
   819→                render={({ field }) => (
   820→                  <div className="flex items-center gap-3 glass-soft rounded-2xl px-4 py-3 h-12">
   821→                    <Switch
   822→                      checked={field.value === "true"}
   823→                      onCheckedChange={(c) =>
   824→                        field.onChange(c ? "true" : "false")
   825→                      }
   826→                    />
   827→                    <span className="text-sm font-medium">
   828→                      {field.value === "true" ? "Enabled" : "Disabled"}
   829→                    </span>
   830→                  </div>
   831→                )}
   832→              />
   833→            </div>
   834→          ) : (
   835→            <GlassInput
   836→              label="Value"
   837→              placeholder={
   838→                type === "CURRENCY"
   839→                  ? "e.g., 50.00"
   840→                  : type === "PERCENTAGE"
   841→                    ? "e.g., 12.5"
   842→                    : type === "NUMBER"
   843→                      ? "e.g., 4"
   844→                      : "Enter value"
   845→              }
   846→              error={form.formState.errors.value?.message}
   847→              {...form.register("value")}
   848→            />
   849→          )}
   850→
   851→          <div className="grid grid-cols-2 gap-3">
   852→            <GlassInput
   853→              label="Unit (optional)"
   854→              placeholder="e.g., ₹/meal"
   855→              {...form.register("unit")}
   856→            />
   857→            <div>
   858→              <label className="mb-1.5 ml-1 block text-xs font-medium text-muted-foreground">
   859→                Category
   860→              </label>
   861→              <Controller
   862→                control={form.control}
   863→                name="category"
   864→                render={({ field }) => (
   865→                  <Select
   866→                    value={field.value}
   867→                    onValueChange={field.onChange}
   868→                  >
   869→                    <SelectTrigger className="w-full h-12 glass-soft rounded-2xl border-2 border-transparent">
   870→                      <SelectValue placeholder="Select category" />
   871→                    </SelectTrigger>
   872→                    <SelectContent>
   873→                      {categories.map((c) => (
   874→                        <SelectItem key={c} value={c}>
   875→                          {c}
   876→                        </SelectItem>
   877→                      ))}
   878→                    </SelectContent>
   879→                  </Select>
   880→                )}
   881→              />
   882→            </div>
   883→          </div>
   884→
   885→          <div>
   886→            <label className="mb-1.5 ml-1 block text-xs font-medium text-muted-foreground">
   887→              Description (optional)
   888→            </label>
   889→            <GlassTextarea
   890→              placeholder="What does this variable control?"
   891→              className="min-h-[80px]"
   892→              {...form.register("description")}
   893→            />
   894→          </div>
   895→
   896→          <div className="flex items-center gap-2 p-3 rounded-2xl bg-info/10 text-info text-[11px]">
   897→            <AlertCircle className="h-3.5 w-3.5 shrink-0" />
   898→            Custom variables can be edited or archived later. System variables
   899→            are locked to preserve formula integrity.
   900→          </div>
   901→
   902→          <DialogFooter className="gap-2">
   903→            <GlassButton
   904→              type="button"
   905→              variant="ghost"
   906→              onClick={() => onOpenChange(false)}
   907→            >
   908→              Cancel
   909→            </GlassButton>
   910→            <GlassButton type="submit" loading={loading}>
   911→              <Save className="h-4 w-4" />
   912→              Create Variable
   913→            </GlassButton>
   914→          </DialogFooter>
   915→        </form>
   916→      </DialogContent>
   917→    </Dialog>
   918→  );
   919→}
