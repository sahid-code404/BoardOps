   206→   206→/** Format quantity + unit for display (e.g. "5 kg", "2 piece"). */
   207→   207→function formatQuantity(qty: number, unit: string): string {
   208→   208→  if (!qty && !unit) return "";
   209→   209→  if (!unit) return String(qty);
   210→   210→  if (!qty) return unit;
   211→   211→  return `${qty} ${unit}`;
   212→   212→}
   213→   213→
   214→   214→/** An expense is locked when its status is LOCKED or it belongs to a past month. */
   215→   215→function isExpenseLocked(expense: Expense): boolean {
   216→   216→  if (expense.status === "LOCKED") return true;
   217→   217→  const expDate = new Date(expense.expenseDate);
   218→   218→  const now = new Date();
   219→   219→  const expYM = expDate.getFullYear() * 12 + expDate.getMonth();
   220→   220→  const todayYM = now.getFullYear() * 12 + now.getMonth();
   221→   221→  return expYM < todayYM;
   222→   222→}

   886→   886→function ExpenseFormBody({
   887→   887→  expense,
   888→   888→  onSubmit,
   889→   889→  loading,
   890→   890→  onCancel,
   891→   891→}: {
   892→   892→  expense: Expense | null;
   893→   893→  onSubmit: (payload: ExpensePayload, id?: string) => void;
   894→   894→  loading: boolean;
   895→   895→  onCancel: () => void;
   896→   896→}) {
   897→   897→  const today = new Date().toISOString().slice(0, 10);
   898→   898→  const isEdit = !!expense;
   899→   899→
   900→   900→  // Derive initial category (predefined or CUSTOM) from the expense.
   901→   901→  const initialCategory: ExpenseCategory = (() => {
   902→   902→    if (!expense?.category) return "GROCERY";
   903→   903→    if (
   904→   904→      CATEGORY_ORDER.includes(expense.category as ExpenseCategory) &&
   905→   905→      expense.category !== "CUSTOM"
   906→   906→    ) {
   907→   907→      return expense.category as ExpenseCategory;
   908→   908→    }
   909→   909→    return "CUSTOM";
   910→   910→  })();
   911→   911→  const initialCustomCategory =
   912→   912→    initialCategory === "CUSTOM" ? expense?.category ?? "" : "";
   913→   913→
   914→   914→  // Derive initial unit (predefined or CUSTOM) from the expense.
   915→   915→  const initialUnit: string = (() => {
   916→   916→    if (!expense?.unit) return "piece";
   917→   917→    if (UNIT_OPTIONS.includes(expense.unit)) return expense.unit;
   918→   918→    return "CUSTOM";
   919→   919→  })();
   920→   920→  const initialCustomUnit = initialUnit === "CUSTOM" ? expense?.unit ?? "" : "";
   921→   921→
   922→   922→  const [title, setTitle] = useState(expense?.title ?? "");
   923→   923→  const [amount, setAmount] = useState(
   924→   924→    expense?.amount ? String(expense.amount) : ""
   925→   925→  );
   926→   926→  const [quantity, setQuantity] = useState(
   927→   927→    expense?.quantity ? String(expense.quantity) : ""
   928→   928→  );
   929→   929→  const [category, setCategory] = useState<ExpenseCategory>(initialCategory);
   930→   930→  const [customCategory, setCustomCategory] = useState(initialCustomCategory);
   931→   931→  const [unit, setUnit] = useState<string>(initialUnit);
   932→   932→  const [customUnit, setCustomUnit] = useState(initialCustomUnit);
   933→   933→  const [date, setDate] = useState(
   934→   934→    expense
   935→   935→      ? new Date(expense.expenseDate).toISOString().slice(0, 10)
   936→   936→      : today
   937→   937→  );
   938→   938→  const [description, setDescription] = useState(expense?.description ?? "");
   939→   939→  const [errors, setErrors] = useState<Record<string, string>>({});
   940→   940→
   941→   941→  function handleSubmit() {
   942→   942→    const next: Record<string, string> = {};
   943→   943→    if (!title.trim() || title.trim().length < 2) next.title = "Item name is required";
   944→   944→    const amt = parseFloat(amount);
   945→   945→    if (!amt || amt <= 0) next.amount = "Enter a valid cost";
   946→   946→    if (!date) next.date = "Date is required";
   947→   947→    if (category === "CUSTOM" && customCategory.trim().length < 2) {
   948→   948→      next.customCategory = "Enter a custom category name";
   949→   949→    }
   950→   950→    const qty = quantity ? parseFloat(quantity) : 0;
   951→   951→    if (quantity && (!qty || qty <= 0)) next.quantity = "Enter a valid quantity";
   952→   952→    if (unit === "CUSTOM" && customUnit.trim().length < 1) {
   953→   953→      next.customUnit = "Enter a custom unit";
   954→   954→    }
