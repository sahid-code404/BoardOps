   101→   101→function toDateString(d: Date): string {
   102→   102→  const y = d.getFullYear();
   103→   103→  const m = String(d.getMonth() + 1).padStart(2, "0");
   104→   104→  const day = String(d.getDate()).padStart(2, "0");
   105→   105→  return `${y}-${m}-${day}`;
   106→   106→}
   107→   107→
   108→   108→/** Returns { top, bottom } for the date picker.
   109→   109→ *  - Today/Yesterday/Tomorrow: top = relative label, bottom = "EEE, d MMM"
   110→   110→ *  - Other dates: top = "d MMM" (date only), bottom = "EEE" (day name only, no duplicate) */
   111→   111→function getDatePickerLabels(d: Date): { top: string; bottom: string } {
   112→   112→  const today = new Date();
   113→   113→  today.setHours(0, 0, 0, 0);
   114→   114→  const target = new Date(d);
   115→   115→  target.setHours(0, 0, 0, 0);
   116→   116→  const diffMs = target.getTime() - today.getTime();
   117→   117→  const diffDays = Math.round(diffMs / (24 * 60 * 60 * 1000));
   118→   118→
   119→   119→  if (diffDays === 0) return { top: "Today", bottom: format(target, "EEE, d MMM") };
   120→   120→  if (diffDays === -1) return { top: "Yesterday", bottom: format(target, "EEE, d MMM") };
   121→   121→  if (diffDays === 1) return { top: "Tomorrow", bottom: format(target, "EEE, d MMM") };
   122→   122→  // Far dates: top = date only, bottom = day name only (no duplicate)
   123→   123→  return { top: format(target, "d MMM"), bottom: format(target, "EEE") };
   124→   124→}
