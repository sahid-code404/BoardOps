   103→function toDateString(d: Date): string {
   104→  const y = d.getFullYear();
   105→  const m = String(d.getMonth() + 1).padStart(2, "0");
   106→  const day = String(d.getDate()).padStart(2, "0");
   107→  return `${y}-${m}-${day}`;
   108→}
   109→
   110→function to12h(t: string): string {
   111→  const [h, m] = t.split(":").map(Number);
   112→  const period = h >= 12 ? "PM" : "AM";
   113→  const hr = h % 12 || 12;
   114→  return `${hr}:${String(m).padStart(2, "0")} ${period}`;
   115→}
   116→
   117→function parseDateStr(s: string): Date {
   118→  const [y, m, d] = s.split("-").map(Number);
   119→  return new Date(y, m - 1, d);
   120→}

   137→

   140→    queryKey: ["user-meals", { month: selectedMonth, year: selectedYear }],
   141→    queryFn: async () => {

   143→        params: { month: selectedMonth, year: selectedYear },

   148→    enabled: viewMode !== "day",
   149→  });

   170→      qc.invalidateQueries({ queryKey: ["user-meals"] });
   171→      qc.invalidateQueries({ queryKey: ["user-meals-day"] });
   172→      qc.invalidateQueries({ queryKey: ["kitchen"] });

   176→
   177→  // Normalize month data into sorted days array
   178→  const days = useMemo(() => {
   179→    const src = monthData;

   182→    const result: { dateStr: string; date: Date; entries: MealEntry[] }[] = [];

   184→    for (const dateStr of dateKeys) {
   185→      const flatEntries = src.byDate[dateStr];
   186→      if (!flatEntries || flatEntries.length === 0) continue;
   187→      const entries: MealEntry[] = flatEntries.map((f) => {
   188→        const meal = mealMap.get(f.mealId);
   189→        return {
   190→          id: f.id, status: f.status, locked: f.locked,
   191→          editableUntil: f.editableUntil, serviceDate: f.serviceDate,
   192→          originalState: f.originalState,
   193→          overridden: f.overridden,
   194→          preRegistration: f.preRegistration,
   195→          meal: {
   196→            id: f.mealId, name: f.mealName, displayName: f.mealDisplayName,
   197→            icon: f.mealIcon, color: f.mealColor,
   198→            startTime: f.startTime, endTime: f.endTime,
   199→            cutoffTime: meal?.cutoffTime ?? "",
   200→          },
   201→        };
   202→      });
   203→      result.push({ dateStr, date: parseDateStr(dateStr), entries });
   204→    }
   205→    return result;
   206→  }, [monthData]);
   207→
   208→  // Map for quick lookup by dateStr
   209→  const dayMap = useMemo(() => {
   210→    const m = new Map<string, MealEntry[]>();
   211→    for (const d of days) m.set(d.dateStr, d.entries);
   212→    return m;
   213→  }, [days]);
   214→
   215→  // Day view entries
   216→  const dayEntries = useMemo(() => {
   217→    if (!dayData) return [];
   218→    const mealMap = new Map(dayData.meals.map((m) => [m.id, m]));
   219→    const flat = dayData.byDate[dayStr] ?? [];
   220→    return flat.map<MealEntry>((f) => {
   221→      const meal = mealMap.get(f.mealId);
   222→      return {
   223→        id: f.id, status: f.status, locked: f.locked,
   224→        editableUntil: f.editableUntil, serviceDate: f.serviceDate,
   225→        originalState: f.originalState,
   226→        overridden: f.overridden,
   227→        preRegistration: f.preRegistration,
   228→        meal: {
   229→          id: f.mealId, name: f.mealName, displayName: f.mealDisplayName,
   230→          icon: f.mealIcon, color: f.mealColor,
   231→          startTime: f.startTime, endTime: f.endTime,
   232→          cutoffTime: meal?.cutoffTime ?? "",
   233→        },
   234→      };
   235→    });
   236→  }, [dayData, dayStr]);
   237→
   238→  // Month stats
   239→  const stats = useMemo(() => {

   241→    for (const day of days) {
   242→      for (const e of day.entries) {
   243→        if (e.status === "ON" || e.status === "LOCKED") on++;
   244→        else if (e.status === "OFF") off++;
   245→        if (e.locked || e.status === "LOCKED") locked++;
   246→      }
   247→    }
   248→    return { on, off, locked };
   249→  }, [days]);
   250→
   251→  // Day stats
   252→  const dayStats = useMemo(() => {
   253→    const on = dayEntries.filter((e) => e.status === "ON" || e.status === "LOCKED").length;

   255→    const locked = dayEntries.filter((e) => e.locked || e.status === "LOCKED").length;
   256→    return { on, off, locked };

   259→  // Registration date — used for the pre-reg toast message + before-enrollment labels.
   260→  // Normalize to local midnight so the comparison is date-only (not affected by timezone).
   261→  const registrationDate = useMemo(() => {
   262→    const raw = monthData?.registrationDate ?? dayData?.registrationDate;
   263→    if (!raw) return null;
   264→    const d = new Date(raw);
   265→    d.setHours(0, 0, 0, 0);
   266→    return d;
   267→  }, [monthData, dayData]);
   268→
   269→  // Whether the selected day (day view) is before registration.
   270→  // Uses date-only comparison (both normalized to local midnight) so a user who
   271→  // registered on July 8 is NOT "before enrollment" on July 8.
   272→  const isDayBeforeRegistration = useMemo(() => {
   273→    if (!registrationDate) return false;
   274→    const sel = new Date(selectedDay);
   275→    sel.setHours(0, 0, 0, 0);
   276→    return sel.getTime() < registrationDate.getTime();
   277→  }, [registrationDate, selectedDay]);
   278→
   279→  // Short toast shown when a user tries to toggle a pre-registration meal
   280→  const showPreRegToast = useCallback(() => {
   281→    const dateLabel = registrationDate
   282→      ? format(registrationDate, "d MMM yyyy")

   284→    toast.error(
   285→      `You enrolled on ${dateLabel} — meals before this date can only be changed by an admin.`
   286→    );
   287→  }, [registrationDate]);
   288→
   289→  const handleToggleDay = useCallback((dateStr: string) => {
   290→    setExpandedDay((prev) => (prev === dateStr ? null : dateStr));
   291→  }, []);
   292→
   293→  // Auto-scroll to today's row when agenda view loads or month changes
   294→  const todayRef = useRef<HTMLDivElement>(null);
   295→  const todayDateStr = toDateString(now);
   296→  const isLoading = viewMode === "day" ? dayLoading : monthLoading;
   297→  useEffect(() => {
   298→    if (viewMode !== "agenda" || isLoading || days.length === 0) return;
   299→    // Only auto-scroll if today is in the current month's data
   300→    const hasToday = days.some((d) => d.dateStr === todayDateStr);
   301→    if (!hasToday) return;
   302→    // Small delay to let the DOM render after loading
   303→    const t = setTimeout(() => {
   304→      todayRef.current?.scrollIntoView({ behavior: "smooth", block: "center" });

   306→    return () => clearTimeout(t);
   307→  }, [viewMode, isLoading, days, todayDateStr]);
   308→
   309→  if (isLoading) {
   310→    return (
   311→      <div className="space-y-4">
   312→        <ShimmerSkeleton className="h-14 w-full" />
   313→        <div className="grid grid-cols-3 gap-3">
   314→          <ShimmerSkeleton className="h-28" />
   315→          <ShimmerSkeleton className="h-28" />
   316→          <ShimmerSkeleton className="h-28" />
   317→        </div>
   318→        <div className="space-y-3">
   319→          {Array.from({ length: 5 }).map((_, i) => (
   320→            <ShimmerSkeleton key={i} className="h-16" />
   321→          ))}
   322→        </div>
   323→      </div>
   324→    );
   325→  }
   326→
   327→  return (
   328→    <StaggerGroup className="space-y-4 pb-6">
   329→      {/* Picker — month picker for agenda/calendar, day picker for day view */}
   330→      {viewMode === "day" ? (
   331→        <StaggerItem>
   332→          <div className="flex items-center justify-center gap-4">
   333→            <motion.button
   334→              whileTap={{ scale: 0.9 }}
   335→              onClick={() => setSelectedDay((d) => addDays(d, -1))}
   336→              aria-label="Previous day"
   337→              className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   338→            >
   339→              <ChevronLeft className="h-5 w-5" />
   340→            </motion.button>
   341→            <button
   342→              onClick={() => !isSameDay(selectedDay, now) && setSelectedDay(new Date())}
   343→              className="flex-1 max-w-[280px] flex items-center justify-center gap-2.5 glass-soft rounded-full px-6 py-2.5 transition-all hover:ring-1 hover:ring-primary/30"
   344→            >
   345→              <Calendar className="h-4 w-4 text-primary shrink-0" />
   346→              <div className="leading-tight text-center">
   347→                <p className="text-sm font-bold text-primary">
   348→                  {isSameDay(selectedDay, now) ? "Today" : format(selectedDay, "d MMM")}
   349→                </p>
   350→                <p className="text-[11px] text-muted-foreground">
   351→                  {format(selectedDay, "EEE, d MMM yyyy")}
   352→                </p>
   353→              </div>
   354→              {!isSameDay(selectedDay, now) && (
   355→                <RotateCcw className="h-3 w-3 text-muted-foreground shrink-0" />
   356→              )}
   357→            </button>
   358→            <motion.button
   359→              whileTap={{ scale: 0.9 }}
   360→              onClick={() => setSelectedDay((d) => addDays(d, 1))}
   361→              aria-label="Next day"
   362→              className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   363→            >
   364→              <ChevronRight className="h-5 w-5" />
   365→            </motion.button>
   366→          </div>
   367→        </StaggerItem>
   368→      ) : (
   369→        <StaggerItem>
   370→          <div className="flex items-center justify-center gap-4">
   371→            <motion.button
   372→              whileTap={{ scale: 0.9 }}
   373→              onClick={() => {
   374→                const d = new Date(selectedYear, selectedMonth - 1, 1);
   375→                setSelectedMonth(d.getMonth());
   376→                setSelectedYear(d.getFullYear());
   377→              }}
   378→              aria-label="Previous month"
   379→              className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   380→            >
   381→              <ChevronLeft className="h-5 w-5" />
   382→            </motion.button>
   383→            <button
   384→              onClick={() => {
   385→                if (!isThisMonth) {
   386→                  setSelectedMonth(now.getMonth());
   387→                  setSelectedYear(now.getFullYear());
   388→                }
   389→              }}
   390→              className="flex-1 max-w-[280px] flex items-center justify-center gap-2.5 glass-soft rounded-full px-6 py-2.5 transition-all hover:ring-1 hover:ring-primary/30"
   391→            >
   392→              <Calendar className="h-4 w-4 text-primary shrink-0" />
   393→              <div className="leading-tight text-center">
   394→                <p className="text-sm font-bold text-primary">
   395→                  {new Date(selectedYear, selectedMonth).toLocaleDateString("en-US", { month: "long" })}
   396→                </p>
   397→                <p className="text-[11px] text-muted-foreground">{selectedYear}</p>
   398→              </div>

   400→                <RotateCcw className="h-3 w-3 text-muted-foreground shrink-0" />
   401→              )}
   402→            </button>
   403→            <motion.button
   404→              whileTap={{ scale: 0.9 }}
   405→              onClick={() => {
   406→                const d = new Date(selectedYear, selectedMonth + 1, 1);
   407→                setSelectedMonth(d.getMonth());
   408→                setSelectedYear(d.getFullYear());
   409→              }}
   410→              aria-label="Next month"
   411→              className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   412→            >
   413→              <ChevronRight className="h-5 w-5" />
   414→            </motion.button>
   415→          </div>
   416→        </StaggerItem>
   417→      )}
   418→
   419→      {/* KPI cards */}
   420→      <StaggerItem>
   421→        <div className="grid grid-cols-3 gap-3">
   422→          <GlassCard className="p-4" glow="success" hover={false}>
   423→            <div className="grid place-items-center h-10 w-10 rounded-2xl bg-success/15 text-success mb-3">
   424→              <Check className="h-5 w-5" />
   425→            </div>
   426→            <p className="text-xs text-muted-foreground">Meals ON</p>
   427→            <div className="text-2xl font-bold tracking-tight">
   428→              <AnimatedCounter value={viewMode === "day" ? dayStats.on : stats.on} />
   429→            </div>
   430→          </GlassCard>
   431→          <GlassCard className="p-4" glow="warning" hover={false}>
   432→            <div className="grid place-items-center h-10 w-10 rounded-2xl bg-warning/15 text-warning mb-3">
   433→              <X className="h-5 w-5" />
   434→            </div>
   435→            <p className="text-xs text-muted-foreground">Meals OFF</p>
   436→            <div className="text-2xl font-bold tracking-tight">

   440→          <GlassCard className="p-4" glow="danger" hover={false}>

   448→          </GlassCard>
   449→        </div>
   450→      </StaggerItem>
   451→
   452→      {/* View mode toggle — segmented control (below KPIs) */}

   456→            {([

   469→                    active
   470→                      ? "bg-primary text-primary-foreground shadow-md shadow-primary/30"
   471→                      : "text-muted-foreground hover:text-foreground"

   473→                >
   474→                  <Icon className="h-3.5 w-3.5" />
   475→                  {opt.label}

   478→            })}
   479→          </div>
   480→        </div>
   481→      </StaggerItem>
   482→
   483→      {/* Content — agenda / calendar / day */}
   484→      {viewMode === "agenda" && (
   485→        <StaggerItem>
   486→          {days.length === 0 ? (
   487→            <GlassCard className="p-10 text-center" hover={false}>
   488→              <Utensils className="h-8 w-8 mx-auto text-muted-foreground mb-2" />

   490→                No meals configured for {new Date(selectedYear, selectedMonth).toLocaleDateString("en-US", { month: "long", year: "numeric" })}.

   492→            </GlassCard>
   493→          ) : (
   494→            <div className="space-y-2">
   495→              <AnimatePresence mode="popLayout">
   496→                {days.map((day) => (
   497→                  <motion.div
   498→                    key={day.dateStr}
   499→                    ref={day.dateStr === todayDateStr ? todayRef : undefined}
   500→                    initial={{ opacity: 0, y: 8 }}
   501→                    animate={{ opacity: 1, y: 0 }}
   502→                    exit={{ opacity: 0, scale: 0.98 }}
   503→                    transition={{ type: "spring", stiffness: 280, damping: 26 }}
   504→                  >
   505→                    <DayRow
   506→                      dateStr={day.dateStr}
   507→                      date={day.date}
   508→                      entries={day.entries}
   509→                      isExpanded={expandedDay === day.dateStr}
   510→                      onToggleExpand={() => handleToggleDay(day.dateStr)}
   511→                      onToggleMeal={(entryId, status) =>
   512→                        toggleMutation.mutate({ entryId, status })
   513→                      }
   514→                      onPreRegToggle={showPreRegToast}
   515→                      registrationDate={registrationDate}
   516→                      loading={toggleMutation.isPending}
   517→                    />
   518→                  </motion.div>
   519→                ))}
   520→              </AnimatePresence>
   521→            </div>
   522→          )}
   523→        </StaggerItem>
   524→      )}
   525→
   526→      {viewMode === "calendar" && (
   527→        <StaggerItem>
   528→          <GlassCard className="p-4" hover={false}>
   529→            {/* Weekday headers */}
   530→            <div className="grid grid-cols-7 gap-1 mb-2">
   531→              {WEEKDAYS.map((wd) => (
