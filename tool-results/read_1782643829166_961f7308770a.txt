   214→function formatDate(iso: string) {
   215→  return new Date(iso).toLocaleDateString("en-IN", {
   216→    day: "numeric",
   217→    month: "short",
   218→    year: "numeric",
   219→  });
   220→}
   221→
   222→/** Format quantity + unit for display (e.g. "5 kg", "2 piece"). */
   223→function formatQuantity(qty: number, unit: string): string {
   224→  if (!qty && !unit) return "";
   225→  if (!unit) return String(qty);
   226→  if (!qty) return unit;
   227→  return `${qty} ${unit}`;
   228→}
   229→
   230→/** An expense is locked when its status is LOCKED or it belongs to a past month. */
   231→function isExpenseLocked(expense: Expense): boolean {
   232→  if (expense.status === "LOCKED") return true;
   233→  const expDate = new Date(expense.expenseDate);
   234→  const now = new Date();
   235→  const expYM = expDate.getFullYear() * 12 + expDate.getMonth();
   236→  const todayYM = now.getFullYear() * 12 + now.getMonth();
   237→  return expYM < todayYM;
   238→}

   583→          </div>
   584→        )}
   585→      </StaggerItem>
   586→
   587→      {/* Add/Edit Expense Sheet */}
   588→      <ExpenseFormSheet
   589→        open={formOpen}
   590→        onOpenChange={(o) => !o && closeForm()}
   591→        onSubmit={handleSubmit}
   592→        loading={addMutation.isPending || editMutation.isPending}
   593→        expense={editTarget}
   594→      />
   595→
   596→      {/* Delete confirm */}
   597→      <AlertDialog
   598→        open={!!deleteTarget}
   599→        onOpenChange={(o) => !o && setDeleteTarget(null)}

   601→        <AlertDialogContent className="rounded-3xl">
   602→          <AlertDialogHeader>
   603→            <AlertDialogTitle>Delete expense?</AlertDialogTitle>
   604→            <AlertDialogDescription>
   605→              {deleteTarget && (
   606→                <>
   607→                  Permanently delete{" "}
   608→                  <span className="font-medium text-foreground">
   609→                    {deleteTarget.title}
   610→                  </span>{" "}
   611→                  ({formatINR(deleteTarget.amount)})? This cannot be undone.
   612→                </>
   613→              )}
   614→            </AlertDialogDescription>
   615→          </AlertDialogHeader>
   616→          <AlertDialogFooter>
   617→            <AlertDialogCancel className="rounded-2xl">Cancel</AlertDialogCancel>
   618→            <AlertDialogAction
   619→              className="rounded-2xl bg-destructive text-white hover:bg-destructive/90"
   620→              onClick={() =>
   621→                deleteTarget && deleteMutation.mutate(deleteTarget.id)
   622→              }
   623→              disabled={deleteMutation.isPending}
   624→            >
   625→              {deleteMutation.isPending ? "Deleting…" : "Delete"}
   626→            </AlertDialogAction>
   627→          </AlertDialogFooter>
   628→        </AlertDialogContent>
   629→      </AlertDialog>
   630→    </StaggerGroup>
   631→  );
   632→}

   877→function ExpenseFormBody({
   878→  expense,
   879→  onSubmit,
   880→  loading,
   881→  onCancel,
   882→}: {
   883→  expense: Expense | null;
   884→  onSubmit: (payload: ExpensePayload, id?: string) => void;
   885→  loading: boolean;
   886→  onCancel: () => void;
   887→}) {
   888→  const today = new Date().toISOString().slice(0, 10);
   889→  const isEdit = !!expense;
   890→
   891→  // Derive initial category (predefined or CUSTOM) from the expense.
   892→  const initialCategory: ExpenseCategory = (() => {
   893→    if (!expense?.category) return "GROCERY";
   894→    if (
   895→      CATEGORY_ORDER.includes(expense.category as ExpenseCategory) &&
   896→      expense.category !== "CUSTOM"
   897→    ) {
   898→      return expense.category as ExpenseCategory;
   899→    }
   900→    return "CUSTOM";
   901→  })();
   902→  const initialCustomCategory =
   903→    initialCategory === "CUSTOM" ? expense?.category ?? "" : "";
   904→
   905→  // Derive initial unit (predefined or CUSTOM) from the expense.
   906→  const initialUnit: string = (() => {
   907→    if (!expense?.unit) return "piece";
   908→    if (UNIT_OPTIONS.includes(expense.unit)) return expense.unit;
   909→    return "CUSTOM";
   910→  })();
   911→  const initialCustomUnit = initialUnit === "CUSTOM" ? expense?.unit ?? "" : "";
   912→
   913→  const [title, setTitle] = useState(expense?.title ?? "");
   914→  const [amount, setAmount] = useState(
   915→    expense?.amount ? String(expense.amount) : ""
   916→  );
   917→  const [quantity, setQuantity] = useState(
   918→    expense?.quantity ? String(expense.quantity) : ""
   919→  );
   920→  const [category, setCategory] = useState<ExpenseCategory>(initialCategory);
   921→  const [customCategory, setCustomCategory] = useState(initialCustomCategory);
   922→  const [unit, setUnit] = useState<string>(initialUnit);
   923→  const [customUnit, setCustomUnit] = useState(initialCustomUnit);
   924→  const [date, setDate] = useState(
   925→    expense
   926→      ? new Date(expense.expenseDate).toISOString().slice(0, 10)
   927→      : today
   928→  );
   929→  const [description, setDescription] = useState(expense?.description ?? "");
   930→  const [errors, setErrors] = useState<Record<string, string>>({});
   931→
   932→  function handleSubmit() {
   933→    const next: Record<string, string> = {};
   934→    if (!title.trim() || title.trim().length < 2) next.title = "Item name is required";
   935→    const amt = parseFloat(amount);
   936→    if (!amt || amt <= 0) next.amount = "Enter a valid cost";
   937→    if (!date) next.date = "Date is required";
   938→    if (category === "CUSTOM" && customCategory.trim().length < 2) {
   939→      next.customCategory = "Enter a custom category name";
   940→    }
   941→    const qty = quantity ? parseFloat(quantity) : 0;
   942→    if (quantity && (!qty || qty <= 0)) next.quantity = "Enter a valid quantity";
   943→    if (unit === "CUSTOM" && customUnit.trim().length < 1) {
   944→      next.customUnit = "Enter a custom unit";
   945→    }
   946→    setErrors(next);
   947→    if (Object.keys(next).length > 0) return;
   948→
   949→    const finalCategory =
   950→      category === "CUSTOM"
   951→        ? customCategory.trim().toUpperCase().replace(/\s+/g, "_")
   952→        : category;
   953→    const finalUnit = unit === "CUSTOM" ? customUnit.trim() : unit;
   954→
   955→    onSubmit(
   956→      {
   957→        title: title.trim(),
   958→        amount: amt,
   959→        category: finalCategory,
   960→        quantity: qty || 0,
   961→        unit: finalUnit,
   962→        expenseDate: new Date(date).toISOString(),
   963→        description: description.trim() || undefined,
   964→      },
   965→      isEdit ? expense!.id : undefined
   966→    );
   967→  }
   968→
   969→  return (
   970→    <>
   971→      <SheetHeader className="px-6 pt-6 pb-2">
   972→        <SheetTitle className="flex items-center gap-2 text-xl">
   973→          <PencilLine className="h-5 w-5 text-primary" />
   974→          {isEdit ? "Edit Expense" : "Add Expense"}
   975→        </SheetTitle>
   976→        <SheetDescription>
   977→          {isEdit
   978→            ? "Update the details of this expense."
   979→            : "Record a new operational expense. It will be visible immediately."}
   980→        </SheetDescription>
   981→      </SheetHeader>
   982→
   983→      <div className="flex-1 overflow-y-auto px-6 py-4 space-y-3 no-scrollbar">
   984→        <GlassInput
   985→          label="Item"
   986→          placeholder="e.g. Monthly groceries"
   987→          value={title}
   988→          onChange={(e) => setTitle(e.target.value)}
   989→          error={errors.title}
   990→          icon={<Receipt />}
   991→        />
   992→
   993→        <div className="space-y-1.5">
   994→          <label className="text-xs font-medium text-muted-foreground ml-1">
   995→            Category
   996→          </label>
   997→          <Select
   998→            value={category}
   999→            onValueChange={(v) => setCategory(v as ExpenseCategory)}
  1000→          >
  1001→            <SelectTrigger className="w-full h-11 rounded-2xl">
  1002→              <SelectValue />
  1003→            </SelectTrigger>
  1004→            <SelectContent>
  1005→              {CATEGORY_ORDER.map((c) => (
  1006→                <SelectItem key={c} value={c}>
  1007→                  {CATEGORY_META[c].label}
  1008→                </SelectItem>
  1009→              ))}
  1010→            </SelectContent>
  1011→          </Select>
  1012→          {category === "CUSTOM" && (
  1013→            <GlassInput
  1014→              placeholder="Enter custom category name…"
  1015→              value={customCategory}
  1016→              onChange={(e) => setCustomCategory(e.target.value)}
  1017→              error={errors.customCategory}
  1018→              icon={<PencilLine className="h-4 w-4" />}
  1019→            />
  1020→          )}
  1021→        </div>
  1022→
  1023→        <div className="grid grid-cols-2 gap-3">
  1024→          <GlassInput
  1025→            label="Quantity"
  1026→            type="number"
  1027→            inputMode="decimal"
  1028→            placeholder="0"
  1029→            value={quantity}
  1030→            onChange={(e) => setQuantity(e.target.value)}
  1031→            error={errors.quantity}
  1032→          />
  1033→          <div className="space-y-1.5">
  1034→            <label className="text-xs font-medium text-muted-foreground ml-1">
  1035→              Unit
  1036→            </label>
  1037→            <Select value={unit} onValueChange={(v) => setUnit(v)}>
  1038→              <SelectTrigger className="w-full h-11 rounded-2xl">
  1039→                <SelectValue />
  1040→              </SelectTrigger>
  1041→              <SelectContent>
  1042→                {UNIT_OPTIONS.map((u) => (
  1043→                  <SelectItem key={u} value={u}>
  1044→                    {u}
  1045→                  </SelectItem>
  1046→                ))}
  1047→                <SelectItem value="CUSTOM">Custom</SelectItem>
  1048→              </SelectContent>
  1049→            </Select>
  1050→            {unit === "CUSTOM" && (
  1051→              <GlassInput
  1052→                placeholder="Enter custom unit…"
  1053→                value={customUnit}
  1054→                onChange={(e) => setCustomUnit(e.target.value)}
  1055→                error={errors.customUnit}
  1056→                icon={<PencilLine className="h-4 w-4" />}
  1057→              />
  1058→            )}
  1059→          </div>
  1060→        </div>
  1061→
  1062→        <GlassInput
  1063→          label="Cost (₹)"
  1064→          type="number"
  1065→          inputMode="decimal"
  1066→          placeholder="0.00"
  1067→          value={amount}
  1068→          onChange={(e) => setAmount(e.target.value)}
  1069→          error={errors.amount}
  1070→          icon={<IndianRupee />}
  1071→        />
  1072→
  1073→        <div className="space-y-1.5">
  1074→          <label className="text-xs font-medium text-muted-foreground ml-1">
  1075→            Date
  1076→          </label>
  1077→          <GlassInput
  1078→            type="date"
  1079→            value={date}
  1080→            onChange={(e) => setDate(e.target.value)}
  1081→            error={errors.date}
  1082→            icon={<Calendar />}
  1083→          />
  1084→        </div>
  1085→
  1086→        <GlassTextarea
  1087→          label="Notes (optional)"
  1088→          placeholder="Add any notes about this expense…"
  1089→          value={description}
  1090→          onChange={(e) => setDescription(e.target.value)}
  1091→          rows={3}
  1092→        />
  1093→      </div>
  1094→
  1095→      <SheetFooter className="px-6 py-4 border-t border-border/40 flex-row gap-2">
  1096→        <GlassButton variant="ghost" className="flex-1" onClick={onCancel}>
  1097→          Cancel
  1098→        </GlassButton>
  1099→        <GlassButton

  1101→          onClick={handleSubmit}
