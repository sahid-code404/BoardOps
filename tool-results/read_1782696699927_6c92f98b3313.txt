   190→function formatINR(n: number) {
   191→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   192→}
   193→
   194→function formatDate(iso: string) {
   195→  return new Date(iso).toLocaleDateString("en-IN", {
   196→    day: "numeric",
   197→    month: "short",
   198→    year: "numeric",
   199→  });
   200→}
   201→
   202→/** Full date-time format: "29 June 2026, 05:28 am" */
   203→function formatDateTime(iso: string) {
   204→  const d = new Date(iso);
   205→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric" });
   206→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   207→  return `${datePart}, ${timePart}`;
   208→}

   211→  "from-violet-500 to-fuchsia-500",

   214→  "from-rose-500 to-pink-500",

   219→function gradientFor(name: string) {
   220→  const idx = name.split("").reduce((acc, c) => acc + c.charCodeAt(0), 0) % AVATAR_GRADIENTS.length;
   221→  return AVATAR_GRADIENTS[idx];
   222→}

   224→function initials(name: string) {
   225→  return name.split(" ").filter(Boolean).slice(0, 2).map((p) => p[0]?.toUpperCase() ?? "").join("");

  1024→function PaymentRow({
  1025→  payment,
  1026→  isAdmin,
  1027→  onApprove,
  1028→  onReject,
  1029→  onEdit,
  1030→  onDelete,
  1031→  onVoid,
  1032→  onRestore,
  1033→}: {
  1034→  payment: Payment;
  1035→  isAdmin: boolean;
  1036→  onApprove: () => void;
  1037→  onReject: () => void;
  1038→  onEdit: () => void;
  1039→  onDelete: () => void;
  1040→  onVoid: () => void;
  1041→  onRestore: () => void;
  1042→}) {
  1043→  const isDeleted = !!payment.deletedAt;
  1044→
  1045→  // Build the actions list — admin-only. Deleted rows show Restore; otherwise
  1046→  // PENDING rows get Approve/Reject, non-VOID rows get Edit/Void, all get Delete.
  1047→  const actions: {
  1048→    label: string;
  1049→    icon: typeof CheckCircle2;
  1050→    onClick: () => void;
  1051→    variant?: "destructive";
  1052→  }[] = [];
  1053→
  1054→  if (isDeleted) {
  1055→    if (isAdmin) {
  1056→      actions.push({ label: "Restore Payment", icon: RotateCcw, onClick: onRestore });
  1057→    }
  1058→  } else {
  1059→    if (isAdmin && payment.status === "PENDING") {
  1060→      actions.push({ label: "Approve Payment", icon: CheckCircle2, onClick: onApprove });
  1061→      actions.push({
  1062→        label: "Reject Payment",
  1063→        icon: XCircle,
  1064→        onClick: onReject,
  1065→        variant: "destructive",
  1066→      });
  1067→    }
  1068→    if (isAdmin && payment.status !== "VOID") {
  1069→      actions.push({ label: "Edit Payment", icon: PencilLine, onClick: onEdit });
  1070→      actions.push({
  1071→        label: "Void Payment",
  1072→        icon: Ban,
  1073→        onClick: onVoid,
  1074→        variant: "destructive",
  1075→      });
  1076→    }
  1077→    if (isAdmin) {
  1078→      actions.push({
  1079→        label: "Delete Payment",
  1080→        icon: Trash2,
  1081→        onClick: onDelete,
  1082→        variant: "destructive",
  1083→      });
  1084→    }
  1085→  }
  1086→
  1087→  const methodMeta = METHOD_META[payment.method];
  1088→  const statusMeta = STATUS_STYLES[payment.status];
  1089→
  1090→  return (
  1091→    <GlassCard className="p-4" hover={false}>
  1092→      <div className="flex items-start gap-3">
  1093→        <Avatar className="h-10 w-10 rounded-xl shrink-0">
  1094→          <AvatarFallback
  1095→            className={cn(
  1096→              "rounded-xl bg-gradient-to-br text-white font-semibold text-xs",
  1097→              gradientFor(payment.user.name)
  1098→            )}
  1099→          >
  1100→            {initials(payment.user.name) || "U"}
  1101→          </AvatarFallback>
  1102→        </Avatar>
  1103→
  1104→        <div className="flex-1 min-w-0">
  1105→          <div className="flex items-start justify-between gap-2">
  1106→            <div className="min-w-0">
  1107→              <div className="flex items-center gap-2 flex-wrap">
  1108→                <h3
  1109→                  className={cn(
  1110→                    "font-medium text-sm truncate",
  1111→                    isDeleted && "text-muted-foreground line-through"
  1112→                  )}
  1113→                >
  1114→                  {isAdmin ? payment.user.name : methodMeta.label}
  1115→                </h3>
  1116→                {isDeleted ? (
  1117→                  <Badge
  1118→                    variant="outline"
  1119→                    className="text-[10px] bg-destructive/15 text-destructive border-destructive/30"
  1120→                  >
  1121→                    <Clock className="h-2.5 w-2.5" />{" "}
  1122→                    {formatDeletionCountdown(new Date(payment.deletedAt!))}
  1123→                  </Badge>
  1124→                ) : (
  1125→                  <>
  1126→                    <Badge
  1127→                      variant="outline"
  1128→                      className={cn("text-[10px]", statusMeta.className)}
  1129→                    >
  1130→                      {statusMeta.label}
  1131→                    </Badge>
  1132→                    <Badge
  1133→                      variant="outline"
  1134→                      className={cn("text-[10px]", methodMeta.className)}
  1135→                    >
  1136→                      {methodMeta.label}
  1137→                    </Badge>
  1138→                  </>
  1139→                )}
  1140→              </div>
  1141→              {/* Transaction strip — Amount with label, same size as billing */}
  1142→              <div className="flex flex-wrap items-baseline gap-x-4 gap-y-1 mt-2">
  1143→                <div className="flex items-baseline gap-1">
  1144→                  <span className="text-[10px] uppercase tracking-wide text-muted-foreground">Amount</span>
  1145→                  <span className="text-base font-bold tabular-nums">{formatINR(payment.amount)}</span>
  1146→                </div>
  1147→                {isDeleted && payment.deletionReason && (
  1148→                  <span className="inline-flex items-start gap-1 text-[11px] text-destructive/80">
  1149→                    <AlertTriangle className="h-3 w-3 shrink-0 mt-0.5" />
  1150→                    Reason: {payment.deletionReason}
  1151→                  </span>
  1152→                )}
  1153→              </div>
  1154→              {/* Date + reference + notes row */}
  1155→              <div className="flex flex-wrap items-center gap-x-3 gap-y-0.5 mt-1.5 text-[11px] text-muted-foreground">
  1156→                <span className="inline-flex items-center gap-1">
  1157→                  <Clock className="h-3 w-3" /> {formatDateTime(payment.createdAt)}
  1158→                </span>
  1159→                {payment.reference && (
  1160→                  <span className="inline-flex items-center gap-1 truncate">
  1161→                    <ArrowUpRight className="h-3 w-3" /> Ref {payment.reference}
  1162→                  </span>
  1163→                )}
  1164→                {payment.notes && (
  1165→                  <span className="truncate max-w-[200px]">
  1166→                    · {payment.notes}
  1167→                  </span>
  1168→                )}
  1169→              </div>
  1170→            </div>
  1171→
  1172→            {/* Dropdown — only render if there are actions */}
  1173→            {actions.length > 0 && (
  1174→              <DropdownMenu>
  1175→                <DropdownMenuTrigger asChild>
  1176→                  <GlassButton
  1177→                    variant="ghost"
  1178→                    size="icon"
  1179→                    className="shrink-0"

  1443→function PaymentEditBody({
  1444→  payment,
  1445→  onSubmit,
  1446→  loading,
  1447→  onCancel,
  1448→}: {
  1449→  payment: Payment | null;
  1450→  onSubmit: (
  1451→    id: string,
  1452→    payload: {
  1453→      amount?: number;
  1454→      method?: PaymentMethod;
  1455→      reference?: string | null;
  1456→      notes?: string | null;
  1457→    }
  1458→  ) => void;
  1459→  loading: boolean;
  1460→  onCancel: () => void;
  1461→}) {
  1462→  const [amount, setAmount] = useState(payment ? String(payment.amount) : "");
  1463→  const [method, setMethod] = useState<PaymentMethod>(payment?.method ?? "UPI");
  1464→  const [reference, setReference] = useState(payment?.reference ?? "");
  1465→  const [notes, setNotes] = useState(payment?.notes ?? "");
  1466→  const [errors, setErrors] = useState<Record<string, string>>({});
  1467→
  1468→  // Cannot edit amount on APPROVED + bill-linked payments — backend will
  1469→  // reject with 422. Lock the field and explain why; admin must void + resubmit.
  1470→  const amountLocked = payment?.status === "APPROVED" && !!payment?.billId;
  1471→
  1472→  function handleSubmit() {
  1473→    if (!payment) return;
  1474→    const next: Record<string, string> = {};
  1475→    const amt = parseFloat(amount);
  1476→    if (!amountLocked) {
  1477→      if (!amt || amt <= 0) next.amount = "Enter a valid amount";
  1478→    }
  1479→    setErrors(next);
  1480→    if (Object.keys(next).length > 0) return;
  1481→
  1482→    const payload: {
  1483→      amount?: number;
  1484→      method?: PaymentMethod;
  1485→      reference?: string | null;
  1486→      notes?: string | null;
  1487→    } = {
  1488→      method,
  1489→      reference: reference.trim() || null,
  1490→      notes: notes.trim() || null,
  1491→    };
  1492→    if (!amountLocked) payload.amount = amt;
  1493→    onSubmit(payment.id, payload);
  1494→  }
  1495→
  1496→  return (
  1497→    <>
  1498→      <SheetHeader className="px-6 pt-6 pb-2">
  1499→        <SheetTitle className="flex items-center gap-2 text-xl">
  1500→          <PencilLine className="h-5 w-5 text-primary" />
  1501→          Edit Payment
  1502→        </SheetTitle>
  1503→        <SheetDescription>
  1504→          Update the details of this payment from {payment?.user.name}.
  1505→        </SheetDescription>
  1506→      </SheetHeader>
  1507→
  1508→      <div className="flex-1 overflow-y-auto px-6 py-4 space-y-3 no-scrollbar">
  1509→        <GlassInput
  1510→          label="Amount (₹)"
  1511→          type="number"
  1512→          inputMode="decimal"
  1513→          placeholder="0.00"
  1514→          value={amount}
  1515→          onChange={(e) => setAmount(e.target.value)}
  1516→          error={errors.amount}
  1517→          icon={<IndianRupee />}
  1518→          disabled={amountLocked}
  1519→          hint={
  1520→            amountLocked
  1521→              ? "Amount locked — this approved payment is linked to a bill. Void it and submit a new payment to change the amount."
  1522→              : undefined
  1523→          }
  1524→        />
  1525→
  1526→        <div className="space-y-1.5">
  1527→          <label className="text-xs font-medium text-muted-foreground ml-1">
  1528→            Method
  1529→          </label>
  1530→          <Select
  1531→            value={method}
  1532→            onValueChange={(v) => setMethod(v as PaymentMethod)}
  1533→          >
  1534→            <SelectTrigger className="w-full h-11 rounded-2xl">
  1535→              <SelectValue />
  1536→            </SelectTrigger>
  1537→            <SelectContent>
  1538→              {Object.entries(METHOD_META).map(([k, m]) => (
  1539→                <SelectItem key={k} value={k}>
  1540→                  {m.label}
  1541→                </SelectItem>
  1542→              ))}
  1543→            </SelectContent>
  1544→          </Select>
  1545→        </div>
  1546→
  1547→        <GlassInput
  1548→          label="Reference (optional)"
  1549→          placeholder="UTR / Txn ID"

  1551→          onChange={(e) => setReference(e.target.value)}
  1552→          icon={<ArrowUpRight />}

  1555→        <GlassTextarea

  1559→          onChange={(e) => setNotes(e.target.value)}
