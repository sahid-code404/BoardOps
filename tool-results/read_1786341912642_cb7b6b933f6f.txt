   582→// PRD DEC-029: Partial refunds supported. A Refund tracks the full refund amount,
   583→// how much has been paid so far, and the remaining amount. Multiple partial
   584→// payments can be recorded until the refund is fully completed.
   585→model Refund {
   586→  id              String   @id @default(cuid())
   587→  refundNumber    String?  // Human-readable reference (e.g. REF-2026-0001)

   590→  billingCycleId  String?  // the cycle that generated this refund

   592→  amount          Float
   593→  // How much has been paid out so far (sum of all partial refund payments)
   594→  paidAmount      Float    @default(0)
   595→  // Remaining = amount - paidAmount (computed, but stored for query convenience)
   596→  remainingAmount Float    @default(0)
   597→  status          String   @default("PENDING") // PENDING | PARTIALLY_PAID | COMPLETED | CANCELLED
   598→  method          String?  // UPI | CASH | BANK_TRANSFER | CHEQUE

   600→  notes           String?
   601→  processedBy     String?
   602→  processedAt     DateTime?
   603→  completedAt     DateTime?
   604→  createdAt       DateTime @default(now())
   605→  updatedAt       DateTime @updatedAt
   606→  user            User     @relation(fields: [userId], references: [id], onDelete: Cascade)
   607→  bill            Bill?    @relation(fields: [billId], references: [id], onDelete: SetNull)
   608→  transactions    RefundTransaction[]
   609→
   610→  @@index([userId, status])
   611→  @@index([billingCycleId])
   612→}

   655→// ─────────────────────────────────────────────────────────────
   656→// FINANCIAL LEDGER (PRD: Resident Ledger — accounting backbone)
   657→// Every financial event creates a ledger entry. Balances are derived
   658→// from the sum of entries, never stored directly. This is the single
   659→// source of truth for a resident's financial position.
   660→// ─────────────────────────────────────────────────────────────
   661→
   662→model LedgerEntry {
   663→  id              String   @id @default(cuid())
   664→  userId          String
   665→  // What type of event
   666→  type            String   // DEPOSIT | BILL_SETTLEMENT | REFUND | ADJUSTMENT | PURCHASE | EXPENSE
   667→  // Credit/debit (from the resident's perspective)
   668→  // DEPOSIT: credit (+) — money in
   669→  // BILL_SETTLEMENT: debit (-) — money out (bill deducted)
   670→  // REFUND: debit (-) — money returned to resident
   671→  // ADJUSTMENT: credit or debit depending on sign
   672→  amount          Float    // positive = credit (adds to balance), negative = debit
   673→  // Running balance after this entry (for quick display — the authoritative
   674→  // balance is always the sum of all entries, but storing the running balance
   675→  // avoids re-summing on every query)
   676→  runningBalance  Float    @default(0)
   677→  // Reference to the source entity
   678→  entityType      String   // Payment | Bill | Refund | Adjustment | Expense | Purchase
   679→  entityId        String?
   680→  // Description for the human-readable ledger
   681→  description     String
   682→  // Billing cycle this entry applies to
   683→  billingMonth    Int?
   684→  billingYear     Int?
   685→  createdAt       DateTime @default(now())
   686→  user            User     @relation(fields: [userId], references: [id], onDelete: Cascade)
   687→  payment         Payment? @relation(fields: [entityId], references: [id], onDelete: SetNull)
   688→
   689→  @@index([userId, createdAt])
   690→  @@index([entityType, entityId])
   691→  @@index([billingMonth, billingYear])
   692→}
