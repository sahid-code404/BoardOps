   198→function formatDate(iso: string) {
   199→  return new Date(iso).toLocaleDateString("en-IN", {
   200→    day: "numeric",
   201→    month: "short",
   202→    year: "numeric",
   203→  });
   204→}
   205→
   206→/** Format quantity + unit for display (e.g. "5 kg", "2 piece"). */
   207→function formatQuantity(qty: number, unit: string): string {
   208→  if (!qty && !unit) return "";
   209→  if (!unit) return String(qty);
   210→  if (!qty) return unit;
   211→  return `${qty} ${unit}`;
   212→}
   213→
   214→/** An expense is locked when its status is LOCKED or it belongs to a past month. */
   215→function isExpenseLocked(expense: Expense): boolean {
   216→  if (expense.status === "LOCKED") return true;
   217→  const expDate = new Date(expense.expenseDate);
   218→  const now = new Date();
   219→  const expYM = expDate.getFullYear() * 12 + expDate.getMonth();
   220→  const todayYM = now.getFullYear() * 12 + now.getMonth();
   221→  return expYM < todayYM;
   222→}

   877→          onSubmit={onSubmit}

   879→          onCancel={() => onOpenChange(false)}

   881→      </SheetContent>

   886→function ExpenseFormBody({
   887→  expense,
   888→  onSubmit,
   889→  loading,
   890→  onCancel,
   891→}: {
   892→  expense: Expense | null;
   893→  onSubmit: (payload: ExpensePayload, id?: string) => void;
   894→  loading: boolean;
   895→  onCancel: () => void;
   896→}) {
   897→  const today = new Date().toISOString().slice(0, 10);
   898→  const isEdit = !!expense;
   899→
   900→  // Derive initial category (predefined or CUSTOM) from the expense.
   901→  const initialCategory: ExpenseCategory = (() => {
   902→    if (!expense?.category) return "GROCERY";
   903→    if (
   904→      CATEGORY_ORDER.includes(expense.category as ExpenseCategory) &&
   905→      expense.category !== "CUSTOM"
   906→    ) {
   907→      return expense.category as ExpenseCategory;
   908→    }
   909→    return "CUSTOM";
   910→  })();
   911→  const initialCustomCategory =
   912→    initialCategory === "CUSTOM" ? expense?.category ?? "" : "";
   913→
   914→  // Derive initial unit (predefined or CUSTOM) from the expense.
   915→  const initialUnit: string = (() => {
   916→    if (!expense?.unit) return "piece";
   917→    if (UNIT_OPTIONS.includes(expense.unit)) return expense.unit;
   918→    return "CUSTOM";
   919→  })();
   920→  const initialCustomUnit = initialUnit === "CUSTOM" ? expense?.unit ?? "" : "";
   921→
   922→  const [title, setTitle] = useState(expense?.title ?? "");
   923→  const [amount, setAmount] = useState(
   924→    expense?.amount ? String(expense.amount) : ""
   925→  );
   926→  const [quantity, setQuantity] = useState(
   927→    expense?.quantity ? String(expense.quantity) : ""
   928→  );
   929→  const [category, setCategory] = useState<ExpenseCategory>(initialCategory);
   930→  const [customCategory, setCustomCategory] = useState(initialCustomCategory);
   931→  const [unit, setUnit] = useState<string>(initialUnit);
   932→  const [customUnit, setCustomUnit] = useState(initialCustomUnit);
   933→  const [date, setDate] = useState(
   934→    expense
   935→      ? new Date(expense.expenseDate).toISOString().slice(0, 10)
   936→      : today
   937→  );
   938→  const [description, setDescription] = useState(expense?.description ?? "");
   939→  const [errors, setErrors] = useState<Record<string, string>>({});
   940→
   941→  function handleSubmit() {
   942→    const next: Record<string, string> = {};
   943→    if (!title.trim() || title.trim().length < 2) next.title = "Item name is required";
   944→    const amt = parseFloat(amount);
   945→    if (!amt || amt <= 0) next.amount = "Enter a valid cost";
   946→    if (!date) next.date = "Date is required";
   947→    if (category === "CUSTOM" && customCategory.trim().length < 2) {
   948→      next.customCategory = "Enter a custom category name";
   949→    }
   950→    const qty = quantity ? parseFloat(quantity) : 0;
   951→    if (quantity && (!qty || qty <= 0)) next.quantity = "Enter a valid quantity";
   952→    if (unit === "CUSTOM" && customUnit.trim().length < 1) {
   953→      next.customUnit = "Enter a custom unit";
   954→    }
   955→    setErrors(next);
   956→    if (Object.keys(next).length > 0) return;
   957→
   958→    const finalCategory =
   959→      category === "CUSTOM"
   960→        ? customCategory.trim().toUpperCase().replace(/\s+/g, "_")
   961→        : category;
   962→    const finalUnit = unit === "CUSTOM" ? customUnit.trim() : unit;
   963→
   964→    onSubmit(
   965→      {
   966→        title: title.trim(),
   967→        amount: amt,
   968→        category: finalCategory,
   969→        quantity: qty || 0,
   970→        unit: finalUnit,
   971→        expenseDate: new Date(date).toISOString(),
   972→        description: description.trim() || undefined,
   973→      },
   974→      isEdit ? expense!.id : undefined
   975→    );
   976→  }
   977→
   978→  return (
   979→    <>
   980→      <SheetHeader className="px-6 pt-6 pb-2">
   981→        <SheetTitle className="flex items-center gap-2 text-xl">
   982→          <PencilLine className="h-5 w-5 text-primary" />
   983→          {isEdit ? "Edit Expense" : "Add Expense"}
   984→        </SheetTitle>
   985→        <SheetDescription>
   986→          {isEdit
   987→            ? "Update the details of this expense."
   988→            : "Record a new operational expense. It will be visible immediately."}
   989→        </SheetDescription>
   990→      </SheetHeader>
   991→
   992→      <div className="flex-1 overflow-y-auto px-6 py-4 space-y-3 no-scrollbar">
   993→        <GlassInput
   994→          label="Item"
   995→          placeholder="e.g. Monthly groceries"
   996→          value={title}
   997→          onChange={(e) => setTitle(e.target.value)}
   998→          error={errors.title}
   999→          icon={<Receipt />}
  1000→        />
  1001→
  1002→        <div className="space-y-1.5">
  1003→          <label className="text-xs font-medium text-muted-foreground ml-1">
  1004→            Category
  1005→          </label>
  1006→          <Select
  1007→            value={category}
  1008→            onValueChange={(v) => setCategory(v as ExpenseCategory)}
  1009→          >
  1010→            <SelectTrigger className="w-full h-11 rounded-2xl">
  1011→              <SelectValue />
  1012→            </SelectTrigger>
  1013→            <SelectContent>
  1014→              {CATEGORY_ORDER.map((c) => (
  1015→                <SelectItem key={c} value={c}>
  1016→                  {CATEGORY_META[c].label}
  1017→                </SelectItem>
  1018→              ))}
  1019→            </SelectContent>
  1020→          </Select>
  1021→          {category === "CUSTOM" && (
  1022→            <GlassInput
  1023→              placeholder="Enter custom category name…"
  1024→              value={customCategory}
  1025→              onChange={(e) => setCustomCategory(e.target.value)}
  1026→              error={errors.customCategory}
  1027→              icon={<PencilLine className="h-4 w-4" />}
  1028→            />
  1029→          )}
  1030→        </div>
  1031→
  1032→        <div className="grid grid-cols-2 gap-3">
  1033→          <GlassInput
  1034→            label="Quantity"
  1035→            type="number"
  1036→            inputMode="decimal"
  1037→            placeholder="0"
  1038→            value={quantity}
  1039→            onChange={(e) => setQuantity(e.target.value)}
  1040→            error={errors.quantity}
  1041→          />
  1042→          <div className="space-y-1.5">
  1043→            <label className="text-xs font-medium text-muted-foreground ml-1">
  1044→              Unit
  1045→            </label>
  1046→            <Select value={unit} onValueChange={(v) => setUnit(v)}>
  1047→              <SelectTrigger className="w-full h-11 rounded-2xl">
  1048→                <SelectValue />
  1049→              </SelectTrigger>
  1050→              <SelectContent>
  1051→                {UNIT_OPTIONS.map((u) => (
  1052→                  <SelectItem key={u} value={u}>
  1053→                    {u}
  1054→                  </SelectItem>
  1055→                ))}
  1056→                <SelectItem value="CUSTOM">Custom</SelectItem>
  1057→              </SelectContent>
  1058→            </Select>
  1059→            {unit === "CUSTOM" && (
  1060→              <GlassInput
  1061→                placeholder="Enter custom unit…"
  1062→                value={customUnit}
  1063→                onChange={(e) => setCustomUnit(e.target.value)}
  1064→                error={errors.customUnit}
  1065→                icon={<PencilLine className="h-4 w-4" />}
  1066→              />
  1067→            )}
  1068→          </div>
  1069→        </div>
  1070→
  1071→        <GlassInput
  1072→          label="Cost (₹)"
  1073→          type="number"
  1074→          inputMode="decimal"
  1075→          placeholder="0.00"
  1076→          value={amount}
  1077→          onChange={(e) => setAmount(e.target.value)}
  1078→          error={errors.amount}
  1079→          icon={<IndianRupee />}
  1080→        />
  1081→
  1082→        <div className="space-y-1.5">
  1083→          <label className="text-xs font-medium text-muted-foreground ml-1">
  1084→            Date
  1085→          </label>
  1086→          <GlassInput

  1088→            value={date}
  1089→            onChange={(e) => setDate(e.target.value)}

  1094→
  1095→        <GlassTextarea
  1096→          label="Notes (optional)"
  1097→          placeholder="Add any notes about this expense…"

  1099→          onChange={(e) => setDescription(e.target.value)}
  1100→          rows={3}
