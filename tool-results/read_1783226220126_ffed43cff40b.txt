    95→function toDateString(d: Date): string {
    96→  const y = d.getFullYear();
    97→  const m = String(d.getMonth() + 1).padStart(2, "0");
    98→  const day = String(d.getDate()).padStart(2, "0");
    99→  return `${y}-${m}-${day}`;
   100→}
   101→
   102→function to12h(t: string): string {
   103→  const [h, m] = t.split(":").map(Number);
   104→  const period = h >= 12 ? "PM" : "AM";
   105→  const hr = h % 12 || 12;
   106→  return `${hr}:${String(m).padStart(2, "0")} ${period}`;
   107→}
   108→
   109→function parseDateStr(s: string): Date {
   110→  const [y, m, d] = s.split("-").map(Number);
   111→  return new Date(y, m - 1, d);
   112→}

   120→export function UserMealsView() {
   121→  const qc = useQueryClient();
   122→  const now = new Date();
   123→  const [viewMode, setViewMode] = useState<ViewMode>("agenda");
   124→  const [selectedMonth, setSelectedMonth] = useState(now.getMonth());

   128→  const [expandedDay, setExpandedDay] = useState<string | null>(toDateString(now));
   129→

   132→    queryKey: ["user-meals", { month: selectedMonth, year: selectedYear }],
   133→    queryFn: async () => {
   134→      const r = await api.get<ApiResponse<EntriesResponse>>("/meals/entries", {
   135→        params: { month: selectedMonth, year: selectedYear },
   136→      });

   138→    },
   139→    placeholderData: (prev) => prev,
   140→    enabled: viewMode !== "day",
   141→  });

   144→  const dayStr = toDateString(selectedDay);
   145→  const { data: dayData, isLoading: dayLoading } = useQuery({
   146→    queryKey: ["user-meals-day", dayStr],
   147→    queryFn: async () => {
   148→      const r = await api.get<ApiResponse<EntriesResponse>>("/meals/entries", {

   154→    enabled: viewMode === "day",

   160→    },

   162→      qc.invalidateQueries({ queryKey: ["user-meals"] });
   163→      qc.invalidateQueries({ queryKey: ["user-meals-day"] });
   164→      qc.invalidateQueries({ queryKey: ["kitchen"] });

   168→
   169→  // Normalize month data into sorted days array
   170→  const days = useMemo(() => {
   171→    const src = monthData;

   174→    const result: { dateStr: string; date: Date; entries: MealEntry[] }[] = [];

   176→    for (const dateStr of dateKeys) {
   177→      const flatEntries = src.byDate[dateStr];
   178→      if (!flatEntries || flatEntries.length === 0) continue;

   180→        const meal = mealMap.get(f.mealId);
   181→        return {
   182→          id: f.id, status: f.status, locked: f.locked,
   183→          editableUntil: f.editableUntil, serviceDate: f.serviceDate,
   184→          meal: {
   185→            id: f.mealId, name: f.mealName, displayName: f.mealDisplayName,
   186→            icon: f.mealIcon, color: f.mealColor,
   187→            startTime: f.startTime, endTime: f.endTime,
   188→            cutoffTime: meal?.cutoffTime ?? "",
   189→          },
   190→        };
   191→      });
   192→      result.push({ dateStr, date: parseDateStr(dateStr), entries });
   193→    }
   194→    return result;
   195→  }, [monthData]);
   196→
   197→  // Map for quick lookup by dateStr
   198→  const dayMap = useMemo(() => {
   199→    const m = new Map<string, MealEntry[]>();
   200→    for (const d of days) m.set(d.dateStr, d.entries);
   201→    return m;
   202→  }, [days]);
   203→
   204→  // Day view entries
   205→  const dayEntries = useMemo(() => {
   206→    if (!dayData) return [];
   207→    const mealMap = new Map(dayData.meals.map((m) => [m.id, m]));
   208→    const flat = dayData.byDate[dayStr] ?? [];
   209→    return flat.map<MealEntry>((f) => {
   210→      const meal = mealMap.get(f.mealId);
   211→      return {
   212→        id: f.id, status: f.status, locked: f.locked,
   213→        editableUntil: f.editableUntil, serviceDate: f.serviceDate,
   214→        meal: {
   215→          id: f.mealId, name: f.mealName, displayName: f.mealDisplayName,
   216→          icon: f.mealIcon, color: f.mealColor,
   217→          startTime: f.startTime, endTime: f.endTime,
   218→          cutoffTime: meal?.cutoffTime ?? "",
   219→        },
   220→      };
   221→    });
   222→  }, [dayData, dayStr]);
   223→
   224→  // Month stats
   225→  const stats = useMemo(() => {
   226→    let on = 0, off = 0, locked = 0;
   227→    for (const day of days) {
   228→      for (const e of day.entries) {
   229→        if (e.status === "ON" || e.status === "LOCKED") on++;
   230→        else if (e.status === "OFF") off++;
   231→        if (e.locked || e.status === "LOCKED") locked++;
   232→      }
   233→    }
   234→    return { on, off, locked };
   235→  }, [days]);
   236→
   237→  // Day stats
   238→  const dayStats = useMemo(() => {
   239→    const on = dayEntries.filter((e) => e.status === "ON" || e.status === "LOCKED").length;
   240→    const off = dayEntries.filter((e) => e.status === "OFF").length;
   241→    const locked = dayEntries.filter((e) => e.locked || e.status === "LOCKED").length;
   242→    return { on, off, locked };
   243→  }, [dayEntries]);
   244→
   245→  const handleToggleDay = useCallback((dateStr: string) => {
   246→    setExpandedDay((prev) => (prev === dateStr ? null : dateStr));
   247→  }, []);
   248→
   249→  // Auto-scroll to today's row when agenda view loads or month changes
   250→  const todayRef = useRef<HTMLDivElement>(null);
   251→  const todayDateStr = toDateString(now);
   252→  const isLoading = viewMode === "day" ? dayLoading : monthLoading;
   253→  useEffect(() => {
   254→    if (viewMode !== "agenda" || isLoading || days.length === 0) return;
   255→    // Only auto-scroll if today is in the current month's data
   256→    const hasToday = days.some((d) => d.dateStr === todayDateStr);
   257→    if (!hasToday) return;
   258→    // Small delay to let the DOM render after loading
   259→    const t = setTimeout(() => {
   260→      todayRef.current?.scrollIntoView({ behavior: "smooth", block: "center" });
   261→    }, 300);
   262→    return () => clearTimeout(t);
   263→  }, [viewMode, isLoading, days, todayDateStr]);
   264→
   265→  if (isLoading) {
   266→    return (
   267→      <div className="space-y-4">
   268→        <ShimmerSkeleton className="h-14 w-full" />
   269→        <div className="grid grid-cols-3 gap-3">
   270→          <ShimmerSkeleton className="h-28" />
   271→          <ShimmerSkeleton className="h-28" />
   272→          <ShimmerSkeleton className="h-28" />
   273→        </div>
   274→        <div className="space-y-3">
   275→          {Array.from({ length: 5 }).map((_, i) => (
   276→            <ShimmerSkeleton key={i} className="h-16" />
   277→          ))}
   278→        </div>
   279→      </div>
   280→    );
   281→  }
   282→
   283→  return (
   284→    <StaggerGroup className="space-y-4 pb-6">
   285→      {/* Picker — month picker for agenda/calendar, day picker for day view */}
   286→      {viewMode === "day" ? (
   287→        <StaggerItem>
   288→          <div className="flex items-center justify-center gap-4">
   289→            <motion.button
   290→              whileTap={{ scale: 0.9 }}
   291→              onClick={() => setSelectedDay((d) => addDays(d, -1))}
   292→              aria-label="Previous day"
   293→              className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   294→            >
   295→              <ChevronLeft className="h-5 w-5" />
   296→            </motion.button>
   297→            <button
   298→              onClick={() => !isSameDay(selectedDay, now) && setSelectedDay(new Date())}
   299→              className="flex-1 max-w-[280px] flex items-center justify-center gap-2.5 glass-soft rounded-full px-6 py-2.5 transition-all hover:ring-1 hover:ring-primary/30"
   300→            >
   301→              <Calendar className="h-4 w-4 text-primary shrink-0" />
   302→              <div className="leading-tight text-center">
   303→                <p className="text-sm font-bold text-primary">
   304→                  {isSameDay(selectedDay, now) ? "Today" : format(selectedDay, "d MMM")}
   305→                </p>
   306→                <p className="text-[11px] text-muted-foreground">
   307→                  {format(selectedDay, "EEE, d MMM yyyy")}
   308→                </p>
   309→              </div>
   310→              {!isSameDay(selectedDay, now) && (
   311→                <RotateCcw className="h-3 w-3 text-muted-foreground shrink-0" />
   312→              )}
   313→            </button>
   314→            <motion.button
   315→              whileTap={{ scale: 0.9 }}
   316→              onClick={() => setSelectedDay((d) => addDays(d, 1))}
   317→              aria-label="Next day"
   318→              className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   319→            >
   320→              <ChevronRight className="h-5 w-5" />
   321→            </motion.button>
   322→          </div>
   323→        </StaggerItem>
   324→      ) : (
   325→        <StaggerItem>
   326→          <div className="flex items-center justify-center gap-4">
   327→            <motion.button
   328→              whileTap={{ scale: 0.9 }}
   329→              onClick={() => {
   330→                const d = new Date(selectedYear, selectedMonth - 1, 1);
   331→                setSelectedMonth(d.getMonth());
   332→                setSelectedYear(d.getFullYear());
   333→              }}
   334→              aria-label="Previous month"
   335→              className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   336→            >
   337→              <ChevronLeft className="h-5 w-5" />
   338→            </motion.button>
   339→            <button
   340→              onClick={() => {
   341→                if (!isThisMonth) {
   342→                  setSelectedMonth(now.getMonth());
   343→                  setSelectedYear(now.getFullYear());
   344→                }
   345→              }}
   346→              className="flex-1 max-w-[280px] flex items-center justify-center gap-2.5 glass-soft rounded-full px-6 py-2.5 transition-all hover:ring-1 hover:ring-primary/30"
   347→            >
   348→              <Calendar className="h-4 w-4 text-primary shrink-0" />
   349→              <div className="leading-tight text-center">
   350→                <p className="text-sm font-bold text-primary">
   351→                  {new Date(selectedYear, selectedMonth).toLocaleDateString("en-US", { month: "long" })}
   352→                </p>
   353→                <p className="text-[11px] text-muted-foreground">{selectedYear}</p>
   354→              </div>
   355→              {!isThisMonth && (
   356→                <RotateCcw className="h-3 w-3 text-muted-foreground shrink-0" />
   357→              )}
   358→            </button>
   359→            <motion.button
   360→              whileTap={{ scale: 0.9 }}
   361→              onClick={() => {
   362→                const d = new Date(selectedYear, selectedMonth + 1, 1);
   363→                setSelectedMonth(d.getMonth());
   364→                setSelectedYear(d.getFullYear());
   365→              }}
   366→              aria-label="Next month"
   367→              className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   368→            >
   369→              <ChevronRight className="h-5 w-5" />
   370→            </motion.button>
   371→          </div>
   372→        </StaggerItem>
   373→      )}
   374→
   375→      {/* KPI cards */}
   376→      <StaggerItem>
   377→        <div className="grid grid-cols-3 gap-3">
   378→          <GlassCard className="p-4" glow="success" hover={false}>
   379→            <div className="grid place-items-center h-10 w-10 rounded-2xl bg-success/15 text-success mb-3">
   380→              <Check className="h-5 w-5" />
   381→            </div>
   382→            <p className="text-xs text-muted-foreground">Meals ON</p>
   383→            <div className="text-2xl font-bold tracking-tight">
   384→              <AnimatedCounter value={viewMode === "day" ? dayStats.on : stats.on} />
   385→            </div>
   386→          </GlassCard>
   387→          <GlassCard className="p-4" glow="warning" hover={false}>
   388→            <div className="grid place-items-center h-10 w-10 rounded-2xl bg-warning/15 text-warning mb-3">
   389→              <X className="h-5 w-5" />
   390→            </div>
   391→            <p className="text-xs text-muted-foreground">Meals OFF</p>
   392→            <div className="text-2xl font-bold tracking-tight">
   393→              <AnimatedCounter value={viewMode === "day" ? dayStats.off : stats.off} />
   394→            </div>
   395→          </GlassCard>
   396→          <GlassCard className="p-4" glow="danger" hover={false}>
   397→            <div className="grid place-items-center h-10 w-10 rounded-2xl bg-destructive/15 text-destructive mb-3">
   398→              <Lock className="h-5 w-5" />
   399→            </div>
   400→            <p className="text-xs text-muted-foreground">Locked</p>
   401→            <div className="text-2xl font-bold tracking-tight">
   402→              <AnimatedCounter value={viewMode === "day" ? dayStats.locked : stats.locked} />
   403→            </div>
   404→          </GlassCard>
   405→        </div>
   406→      </StaggerItem>
   407→
   408→      {/* View mode toggle — segmented control (below KPIs) */}
   409→      <StaggerItem>
   410→        <div className="flex items-center justify-center">
   411→          <div className="inline-flex items-center gap-1 glass-soft rounded-2xl p-1">
   412→            {([
   413→              { mode: "agenda" as const, label: "Agenda", icon: List },
   414→              { mode: "calendar" as const, label: "Calendar", icon: CalendarDays },
   415→              { mode: "day" as const, label: "Day", icon: Sun },
   416→            ]).map((opt) => {
   417→              const active = viewMode === opt.mode;
   418→              const Icon = opt.icon;
   419→              return (
   420→                <button
   421→                  key={opt.mode}
   422→                  onClick={() => setViewMode(opt.mode)}
   423→                  className={cn(
   424→                    "inline-flex items-center gap-1.5 h-8 px-3 rounded-xl text-[11px] font-medium transition-all shrink-0",
   425→                    active
   426→                      ? "bg-primary text-primary-foreground shadow-md shadow-primary/30"
   427→                      : "text-muted-foreground hover:text-foreground"
   428→                  )}
   429→                >
   430→                  <Icon className="h-3.5 w-3.5" />
   431→                  {opt.label}
   432→                </button>
   433→              );
   434→            })}
   435→          </div>
   436→        </div>
   437→      </StaggerItem>
   438→
   439→      {/* Content — agenda / calendar / day */}
   440→      {viewMode === "agenda" && (
   441→        <StaggerItem>
   442→          {days.length === 0 ? (
   443→            <GlassCard className="p-10 text-center" hover={false}>
   444→              <Utensils className="h-8 w-8 mx-auto text-muted-foreground mb-2" />
   445→              <p className="text-sm text-muted-foreground">
   446→                No meals configured for {new Date(selectedYear, selectedMonth).toLocaleDateString("en-US", { month: "long", year: "numeric" })}.
   447→              </p>
   448→            </GlassCard>
   449→          ) : (
   450→            <div className="space-y-2">
   451→              <AnimatePresence mode="popLayout">
   452→                {days.map((day) => (
   453→                  <motion.div
   454→                    key={day.dateStr}
   455→                    ref={day.dateStr === todayDateStr ? todayRef : undefined}
   456→                    layout
   457→                    initial={{ opacity: 0, y: 8 }}
   458→                    animate={{ opacity: 1, y: 0 }}
   459→                    exit={{ opacity: 0, scale: 0.98 }}
   460→                    transition={{ type: "spring", stiffness: 280, damping: 26 }}
   461→                  >
   462→                    <DayRow
   463→                      dateStr={day.dateStr}
   464→                      date={day.date}
   465→                      entries={day.entries}
   466→                      isExpanded={expandedDay === day.dateStr}
   467→                      onToggleExpand={() => handleToggleDay(day.dateStr)}
   468→                      onToggleMeal={(entryId, status) =>
   469→                        toggleMutation.mutate({ entryId, status })
   470→                      }
   471→                      loading={toggleMutation.isPending}
   472→                    />
   473→                  </motion.div>
   474→                ))}
   475→              </AnimatePresence>
   476→            </div>
   477→          )}
   478→        </StaggerItem>
   479→      )}
   480→
   481→      {viewMode === "calendar" && (
   482→        <StaggerItem>
   483→          <GlassCard className="p-4" hover={false}>
   484→            {/* Weekday headers */}
   485→            <div className="grid grid-cols-7 gap-1 mb-2">
   486→              {WEEKDAYS.map((wd) => (
   487→                <div key={wd} className="text-center text-[10px] font-medium text-muted-foreground py-1">
   488→                  {wd}
   489→                </div>
   490→              ))}
   491→            </div>
   492→            {/* Calendar grid */}
   493→            <div className="grid grid-cols-7 gap-1">
   494→              {(() => {
   495→                const monthStart = startOfMonth(new Date(selectedYear, selectedMonth, 1));
   496→                const monthEnd = endOfMonth(monthStart);
   497→                const startPad = getDay(monthStart);
   498→                const allDays = eachDayOfInterval({ start: monthStart, end: monthEnd });
   499→                const cells: (Date | null)[] = [];
   500→                for (let i = 0; i < startPad; i++) cells.push(null);
   501→                allDays.forEach((d) => cells.push(d));
   502→                while (cells.length % 7 !== 0) cells.push(null);
   503→
   504→                return cells.map((date, i) => {
   505→                  if (!date) return <div key={`pad-${i}`} className="aspect-square" />;
   506→                  const ds = toDateString(date);
   507→                  const entries = dayMap.get(ds) ?? [];
   508→                  const isToday = ds === toDateString(now);
   509→                  const onCount = entries.filter((e) => e.status === "ON" || e.status === "LOCKED").length;
   510→                  const offCount = entries.filter((e) => e.status === "OFF").length;
   511→                  const hasLocked = entries.some((e) => e.locked || e.status === "LOCKED");
   512→                  const isPast = date < now && !isToday;
   513→
   514→                  return (
   515→                    <button
   516→                      key={ds}
   517→                      onClick={() => {
   518→                        setSelectedDay(date);
   519→                        setViewMode("day");
   520→                      }}
   521→                      className={cn(
   522→                        "aspect-square rounded-xl flex flex-col items-center justify-center gap-0.5 transition-all text-[10px]",
   523→                        isToday
   524→                          ? "bg-primary/15 ring-1 ring-primary/40"
   525→                          : entries.length > 0
   526→                            ? "glass-soft hover:ring-1 hover:ring-primary/30"
   527→                            : "opacity-40",
   528→                        isPast && "opacity-60"
   529→                      )}
   530→                    >
