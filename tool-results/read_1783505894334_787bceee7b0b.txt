   231→/**
   232→ * Custom 12-hour time picker with AM/PM select.
   233→ * Always displays 12-hour AM/PM format regardless of browser locale.
   234→ * Stores/returns "HH:mm" (24-hour) format for backend compatibility.
   235→ */
   236→function TimePicker12({
   237→  value,
   238→  onChange,
   239→  error,
   240→}: {
   241→  value: string;
   242→  onChange: (v: string) => void;
   243→  error?: string;
   244→}) {
   245→  // Parse "HH:mm" → { hour: 1-12, minute, period: AM|PM }
   246→  const [h24, m] = (value || "08:00").split(":").map(Number);
   247→  const period = h24 >= 12 ? "PM" : "AM";
   248→  const hour12 = h24 % 12 === 0 ? 12 : h24 % 12;
   249→  const minute = m || 0;
   250→
   251→  const update = (newHour12: number, newMinute: number, newPeriod: "AM" | "PM") => {
   252→    let h = newHour12 % 12;
   253→    if (newPeriod === "PM") h += 12;
   254→    onChange(`${String(h).padStart(2, "0")}:${String(newMinute).padStart(2, "0")}`);
   255→  };
   256→
   257→  return (
   258→    <div className="flex items-center gap-1.5">
   259→      {/* Hour (1-12) */}
   260→      <select
   261→        value={hour12}
   262→        onChange={(e) => update(Number(e.target.value), minute, period)}
   263→        className="h-10 rounded-2xl glass px-2 text-sm outline-none focus:ring-2 focus:ring-primary/40 cursor-pointer"
   264→      >
   265→        {Array.from({ length: 12 }, (_, i) => i + 1).map((h) => (
   266→          <option key={h} value={h}>{h}</option>
   267→        ))}
   268→      </select>
   269→      <span className="text-muted-foreground text-sm">:</span>
   270→      {/* Minute (00-59, step 5) */}
   271→      <select
   272→        value={minute}
   273→        onChange={(e) => update(hour12, Number(e.target.value), period)}
   274→        className="h-10 rounded-2xl glass px-2 text-sm outline-none focus:ring-2 focus:ring-primary/40 cursor-pointer"
   275→      >
   276→        {Array.from({ length: 12 }, (_, i) => i * 5).map((m) => (
   277→          <option key={m} value={m}>{String(m).padStart(2, "0")}</option>
   278→        ))}
   279→      </select>
   280→      {/* AM / PM */}
   281→      <select
   282→        value={period}
   283→        onChange={(e) => update(hour12, minute, e.target.value as "AM" | "PM")}
   284→        className="h-10 rounded-2xl glass px-2 text-sm outline-none focus:ring-2 focus:ring-primary/40 cursor-pointer"
   285→      >
   286→        <option value="AM">AM</option>
   287→        <option value="PM">PM</option>
   288→      </select>
   289→      {error && <p className="text-[11px] text-destructive ml-1">{error}</p>}
   290→    </div>
   291→  );
   292→}
   293→
   294→function formatTime12(t: string): string {
   295→  const [h, m] = t.split(":").map(Number);
   296→  const period = h >= 12 ? "PM" : "AM";
   297→  const hr = h % 12 === 0 ? 12 : h % 12;
   298→  return `${hr}:${String(m || 0).padStart(2, "0")} ${period}`;
   299→}
   300→
   301→function computeCutoffPreview(
   302→  strategy: string,
   303→  cutoffTime: string,
   304→  offsetMinutes: number
   305→): string {
   306→  const time12 = formatTime12(cutoffTime);
   307→  switch (strategy) {
   308→    case "PREVIOUS_DAY":
   309→      return `Previous day, ${time12}`;
   310→    case "SAME_DAY":
   311→      return `Service day, ${time12}`;
   312→    case "CUSTOM_OFFSET":
   313→      return `${offsetMinutes} min before service`;
   314→    default:
   315→      return time12;
   316→  }
   317→}
