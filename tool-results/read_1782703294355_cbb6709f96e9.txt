   190→   190→function formatINR(n: number) {
   191→   191→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   192→   192→}

   194→   194→function formatDate(iso: string) {
   195→   195→  const d = new Date(iso);
   196→   196→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" });
   197→   197→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   198→   198→  return `${datePart}, ${timePart}`;
   199→   199→}
   200→   200→
   201→   201→/** Full date-time format: "29 June 2026, 05:28 am" */
   202→   202→function formatDateTime(iso: string) {
   203→   203→  const d = new Date(iso);
   204→   204→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric" });
   205→   205→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   206→   206→  return `${datePart}, ${timePart}`;
   207→   207→}

   447→   447→

   460→   460→          ))}

   462→   462→      </div>

   941→   941→function KpiCard({
   942→   942→  label,
   943→   943→  value,
   944→   944→  icon,
   945→   945→  color,
   946→   946→  prefix,

   955→   955→    color === "primary"

  1471→  1471→function PaymentEditBody({

  1484→  1484→      notes?: string | null;

  1489→  1489→}) {

  1499→  1499→

  1504→  1504→    if (!amountLocked) {

  1506→  1506→    }
  1507→  1507→    setErrors(next);
  1508→  1508→    if (Object.keys(next).length > 0) return;
  1509→  1509→
  1510→  1510→    const payload: {
  1511→  1511→      amount?: number;
  1512→  1512→      method?: PaymentMethod;

  1514→  1514→      notes?: string | null;

  1517→  1517→      reference: reference.trim() || null,
  1518→  1518→      notes: notes.trim() || null,
  1519→  1519→    };
  1520→  1520→    if (!amountLocked) payload.amount = amt;
  1521→  1521→    onSubmit(payment.id, payload);

  1541→  1541→          placeholder="0.00"
  1542→  1542→          value={amount}
  1543→  1543→          onChange={(e) => setAmount(e.target.value)}

  1550→  1550→              : undefined
