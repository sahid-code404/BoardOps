   173→   173→function formatINR(n: number) {

   177→   177→function formatMonthYear(month: number, year: number) {
   178→   178→  return `${MONTHS[month] ?? "—"} ${year}`;

   180→   180→
   181→   181→function formatDate(iso: string | null) {
   182→   182→  if (!iso) return "—";
   183→   183→  const d = new Date(iso);
   184→   184→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" });
   185→   185→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   186→   186→  return `${datePart}, ${timePart}`;
   187→   187→}
   188→   188→
   189→   189→/** Full date-time format: "29 June 2026, 05:28 am" */
   190→   190→function formatDateTime(iso: string | null) {
   191→   191→  if (!iso) return "—";
   192→   192→  const d = new Date(iso);
   193→   193→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric" });
   194→   194→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   195→   195→  return `${datePart}, ${timePart}`;
   196→   196→}
   197→   197→
   198→   198→/** A bill is overdue if it has a due date in the past AND there's still an outstanding due amount. */
   199→   199→function isOverdue(bill: Bill): boolean {
   200→   200→  if (!bill.dueDate || bill.deletedAt) return false;
   201→   201→  if (bill.status === "PAID" || bill.status === "VOID") return false;
   202→   202→  return new Date(bill.dueDate) < new Date() && bill.dueAmount > 0;
   203→   203→}

   411→   411→  }, [sourceBills, search, statusFilter]);
   412→   412→
   413→   413→  if (isLoading) {

   419→   419→          ))}
   420→   420→        </div>
   421→   421→        <ShimmerSkeleton className="h-14 w-full" />
   422→   422→        <div className="grid gap-3">
   423→   423→          {Array.from({ length: 5 }).map((_, i) => (
   424→   424→            <ShimmerSkeleton key={i} className="h-20" />

   430→   430→
   431→   431→  return (

   433→   433→      {/* Subtle refetch indicator — thin animated bar at the top of the list.

   443→   443→            style={{ transformOrigin: "left" }}

   446→   446→          />
   447→   447→        )}

   449→   449→      {/* Month picker — centered capsule with circular arrows + click-to-reset */}
   450→   450→      <StaggerItem>
   451→   451→        <div className="flex items-center justify-center gap-4">
   452→   452→          <motion.button
   453→   453→            whileTap={{ scale: 0.9 }}
   454→   454→            onClick={() => {
