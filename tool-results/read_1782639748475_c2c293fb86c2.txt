    84→type BillStatus =
    85→  | "DRAFT"
    86→  | "GENERATED"

    95→  periodMonth: number;

    98→  otherCharges: number;

   100→  totalAmount: number;
   101→  paidAmount: number;
   102→  dueAmount: number;
   103→  status: BillStatus;
   104→  dueDate: string | null;
   105→  generatedAt: string | null;
   106→  createdAt: string;
   107→  deletedAt: string | null;
   108→  deletionReason: string | null;
   109→  user: { name: string; email: string; room: string | null; avatarUrl: string | null };

   119→const AVATAR_GRADIENTS = [
   120→  "from-violet-500 to-fuchsia-500",
   121→  "from-emerald-500 to-teal-500",
   122→  "from-amber-500 to-orange-500",
   123→  "from-rose-500 to-pink-500",
   124→  "from-cyan-500 to-blue-500",
   125→  "from-indigo-500 to-purple-500",

   127→
   128→function gradientFor(name: string) {
   129→  const idx = name.split("").reduce((acc, c) => acc + c.charCodeAt(0), 0) % AVATAR_GRADIENTS.length;
   130→  return AVATAR_GRADIENTS[idx];
   131→}
   132→
   133→function initials(name: string) {
   134→  return name.split(" ").filter(Boolean).slice(0, 2).map((p) => p[0]?.toUpperCase() ?? "").join("");
   135→}

   191→function formatINR(n: number) {
   192→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   193→}
   194→
   195→function formatMonthYear(month: number, year: number) {
   196→  return `${MONTHS[month] ?? "—"} ${year}`;
   197→}
   198→
   199→function formatDate(iso: string | null) {
   200→  if (!iso) return "—";
   201→  return new Date(iso).toLocaleDateString("en-IN", {
   202→    day: "numeric",
   203→    month: "short",
   204→    year: "numeric",
   205→  });
   206→}

   211→
   212→export function BillingView() {
   213→  const user = useAuthStore((s) => s.user);
   214→  const setView = useAppStore((s) => s.setView);
   215→  const isAdmin =
   216→    user?.role === "ADMIN";
   217→
   218→  const qc = useQueryClient();
   219→  const [search, setSearch] = useState("");
   220→  const [statusFilter, setStatusFilter] = useState<BillStatus | "ALL" | "DELETED">("ALL");
   221→  const [generateOpen, setGenerateOpen] = useState(false);
   222→  const [genMonth, setGenMonth] = useState<number>(new Date().getMonth());
   223→  const [genYear, setGenYear] = useState<number>(new Date().getFullYear());
   224→  const [selectedBill, setSelectedBill] = useState<Bill | null>(null);
   225→  const [voidTarget, setVoidTarget] = useState<Bill | null>(null);
   226→  const now = new Date();
   227→  const [selectedMonth, setSelectedMonth] = useState(now.getMonth());
   228→  const [selectedYear, setSelectedYear] = useState(now.getFullYear());
   229→
   230→  const { data: bills = [], isLoading } = useQuery({
   231→    queryKey: ["bills", { month: selectedMonth, year: selectedYear }],
   232→    queryFn: async () => {
   233→      const r = await api.get<ApiResponse<Bill[]>>("/bills", {
   234→        params: { month: selectedMonth, year: selectedYear },
   235→      });
   236→      return r.data;
   237→    },
   238→  });
   239→
   240→  // Fetch soft-deleted bills (deletion queue) — admin only
   241→  const { data: deletedBills = [] } = useQuery({
   242→    queryKey: ["bills", "deleted", { month: selectedMonth, year: selectedYear }],
   243→    queryFn: async () => {
   244→      const r = await api.get<ApiResponse<Bill[]>>("/bills", {
   245→        params: { month: selectedMonth, year: selectedYear, includeDeleted: "true" },
   246→      });
   247→      return r.data;
   248→    },
   249→    enabled: isAdmin,
   250→  });
   251→
   252→  const generateMutation = useMutation({
   253→    mutationFn: () =>
   254→      api.post<ApiResponse<{ generated: number; month: number; year: number }>>(
   255→        "/bills",
   256→        { month: genMonth, year: genYear }
   257→      ),
   258→    onSuccess: (r) => {
   259→      toast.success(`Generated ${r.data.generated} bills for ${MONTHS[genMonth]} ${genYear}`);
   260→      setGenerateOpen(false);
   261→      qc.invalidateQueries({ queryKey: ["bills"] });
   262→    },
   263→    onError: (e: Error) => toast.error(e.message || "Failed to generate bills"),
   264→  });
   265→
   266→  const voidMutation = useMutation({
   267→    mutationFn: (id: string) =>
   268→      api.delete<ApiResponse<{ success: boolean }>>(`/bills/${id}`),
   269→    onSuccess: () => {
   270→      toast.success("Bill voided successfully");
   271→      setVoidTarget(null);
   272→      qc.invalidateQueries({ queryKey: ["bills"] });
   273→    },
   274→    onError: (e: Error) => toast.error(e.message || "Failed to void bill"),
   275→  });
   276→
   277→  const [deleteTarget, setDeleteTarget] = useState<Bill | null>(null);
   278→  const [deleteAllOpen, setDeleteAllOpen] = useState(false);
   279→  const [deleteReason, setDeleteReason] = useState("");
   280→  const [deleteAllReason, setDeleteAllReason] = useState("");
   281→
   282→  const deleteMutation = useMutation({
   283→    mutationFn: async (id: string) => {
   284→      await api.delete(`/bills/${id}`, { body: JSON.stringify({ reason: deleteReason || undefined }) });
   285→    },
   286→    onSuccess: () => {
   287→      toast.success("Bill scheduled for deletion — permanently removed in 7 days");
   288→      setDeleteTarget(null);
   289→      setDeleteReason("");
   290→      qc.invalidateQueries({ queryKey: ["bills"] });
   291→    },
   292→    onError: (e: Error) => toast.error(e.message || "Failed to delete bill"),
   293→  });
   294→
   295→  const deleteAllMutation = useMutation({
   296→    mutationFn: async () => {
   297→      const r = await api.delete<ApiResponse<{ deleted: number }>>("/bills", {
   298→        params: { month: selectedMonth, year: selectedYear },
   299→        body: JSON.stringify({ reason: deleteAllReason || undefined }),
   300→      });
   301→      return r.data;
   302→    },
   303→    onSuccess: (data) => {
   304→      toast.success(`${data.deleted} bills scheduled for deletion — permanently removed in 7 days`);
   305→      setDeleteAllOpen(false);
   306→      setDeleteAllReason("");
   307→      qc.invalidateQueries({ queryKey: ["bills"] });
   308→    },
   309→    onError: (e: Error) => toast.error(e.message || "Failed to delete bills"),
   310→  });
   311→
   312→  const restoreMutation = useMutation({
   313→    mutationFn: async (id: string) => {
   314→      const r = await api.post<ApiResponse<Bill>>(`/bills/${id}/restore`);
   315→      return r.data;
   316→    },
   317→    onSuccess: () => {
   318→      toast.success("Bill restored successfully");
   319→      qc.invalidateQueries({ queryKey: ["bills"] });
   320→    },
   321→    onError: (e: Error) => toast.error(e.message || "Failed to restore bill"),
   322→  });
   323→
   324→  const restoreAllMutation = useMutation({
   325→    mutationFn: async () => {
   326→      const results = await Promise.all(
   327→        deletedBills.map((b) => api.post(`/bills/${b.id}/restore`).catch(() => null))
   328→      );
   329→      return results.filter((r) => r !== null).length;
   330→    },
   331→    onSuccess: (count) => {
   332→      toast.success(`${count} bills restored successfully`);
   333→      qc.invalidateQueries({ queryKey: ["bills"] });
   334→    },
   335→    onError: (e: Error) => toast.error(e.message || "Failed to restore bills"),
   336→  });
   337→
   338→  // ── Derived KPIs ──
   339→  const kpis = useMemo(() => {
   340→    const active = bills.filter((b) => b.status !== "VOID");
   341→    const totalBilled = active.reduce((s, b) => s + b.totalAmount, 0);
   342→    const totalCollected = active.reduce((s, b) => s + b.paidAmount, 0);
   343→    const totalOutstanding = active.reduce((s, b) => s + b.dueAmount, 0);
   344→    const overdueCount = active.filter((b) => b.status === "OVERDUE").length;
   345→    return { totalBilled, totalCollected, totalOutstanding, overdueCount };
   346→  }, [bills]);
   347→
   348→  // ── Filtered list ──
   349→  // Use deleted bills when filter is DELETED, otherwise use active bills
   350→  const sourceBills = statusFilter === "DELETED" ? deletedBills : bills;
   351→
   352→  const filtered = useMemo(() => {
   353→    const q = search.trim().toLowerCase();
   354→    return sourceBills.filter((b) => {
   355→      if (statusFilter !== "ALL" && statusFilter !== "DELETED" && b.status !== statusFilter) return false;
   356→      if (!q) return true;
   357→      return (
   358→        b.user.name?.toLowerCase().includes(q) ||
   359→        b.user.email?.toLowerCase().includes(q) ||
   360→        (b.user.room || "").toLowerCase().includes(q)
   361→      );
   362→    });
   363→  }, [sourceBills, search, statusFilter]);
   364→
   365→  if (isLoading) {
   366→    return (
   367→      <div className="space-y-4">
   368→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   369→          {Array.from({ length: 4 }).map((_, i) => (
   370→            <ShimmerSkeleton key={i} className="h-32" />
   371→          ))}
   372→        </div>
   373→        <ShimmerSkeleton className="h-14 w-full" />
   374→        <div className="grid gap-3">
   375→          {Array.from({ length: 5 }).map((_, i) => (
   376→            <ShimmerSkeleton key={i} className="h-20" />
   377→          ))}
   378→        </div>
   379→      </div>
   380→    );
   381→  }
   382→
   383→  return (
   384→    <StaggerGroup className="space-y-4 md:space-y-5">
   385→      {/* Month picker — centered, capsule design (matches expenses page) */}
   386→      <StaggerItem>
   387→        <div className="flex items-center justify-center gap-4">
   388→          <motion.button
   389→            whileTap={{ scale: 0.9 }}
   390→            onClick={() => {
   391→              const d = new Date(selectedYear, selectedMonth - 1, 1);
   392→              setSelectedMonth(d.getMonth());
   393→              setSelectedYear(d.getFullYear());
   394→            }}
   395→            aria-label="Previous month"
   396→            className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   397→          >
   398→            <ChevronLeft className="h-5 w-5" />
   399→          </motion.button>
   400→
   401→          <div className="flex-1 max-w-[280px] flex items-center justify-center gap-2.5 glass-soft rounded-full px-6 py-2.5">
   402→            <Calendar className="h-4 w-4 text-primary shrink-0" />
   403→            <div className="leading-tight text-center">
   404→              <p className="text-sm font-bold text-primary">
   405→                {new Date(selectedYear, selectedMonth).toLocaleDateString("en-US", { month: "long" })}
   406→              </p>
   407→              <p className="text-[11px] text-muted-foreground">
   408→                {selectedYear}
   409→              </p>
   410→            </div>
   411→          </div>
   412→
   413→          <motion.button
   414→            whileTap={{ scale: 0.9 }}
   415→            onClick={() => {
   416→              const d = new Date(selectedYear, selectedMonth + 1, 1);
   417→              setSelectedMonth(d.getMonth());
   418→              setSelectedYear(d.getFullYear());
   419→            }}
   420→            aria-label="Next month"
   421→            className="grid place-items-center h-10 w-10 rounded-full glass-strong shrink-0 ring-1 ring-border/40 hover:ring-primary/40 transition-all"
   422→          >
   423→            <ChevronRight className="h-5 w-5" />
   424→          </motion.button>
   425→        </div>
   426→      </StaggerItem>
   427→
   428→      {/* Compact action bar */}
   429→      {isAdmin && (
   430→        <StaggerItem>
   431→          <div className="flex items-center justify-end gap-3">
   432→            <p className="text-sm text-muted-foreground hidden sm:block">
   433→              Generate and track resident bills
   434→            </p>
   435→            <GlassButton
   436→              onClick={() => setGenerateOpen(true)}
   437→              className="shrink-0"
   438→            >
   439→              <Plus className="h-4 w-4" />
   440→              Generate Bills
   441→            </GlassButton>
   442→          </div>
   443→        </StaggerItem>
   444→      )}
   445→
   446→      {/* KPIs */}
   447→      <StaggerItem>
   448→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   449→          <KpiCard
   450→            label="Total Billed"
   451→            value={kpis.totalBilled}
   452→            icon={<Wallet className="h-5 w-5" />}
   453→            color="primary"
   454→            prefix="₹"
   455→          />
   456→          <KpiCard
   457→            label="Total Collected"
   458→            value={kpis.totalCollected}
   459→            icon={<CheckCircle2 className="h-5 w-5" />}
   460→            color="success"
   461→            prefix="₹"
   462→          />
   463→          <KpiCard
   464→            label="Outstanding"
   465→            value={kpis.totalOutstanding}
   466→            icon={<TrendingUp className="h-5 w-5" />}
   467→            color="warning"
   468→            prefix="₹"
   469→          />
   470→          <KpiCard
   471→            label="Overdue Count"
   472→            value={kpis.overdueCount}
   473→            icon={<AlertCircle className="h-5 w-5" />}
   474→            color="danger"
   475→          />
   476→        </div>
   477→      </StaggerItem>
   478→
   479→      {/* Search + Filter pills (expenses-style design) */}
   480→      <StaggerItem>
   481→        <div className="space-y-3">
   482→          <GlassInput
   483→            placeholder="Search by name, email, room…"
   484→            value={search}
   485→            onChange={(e) => setSearch(e.target.value)}
   486→            icon={<Search />}
   487→          />
   488→          <div className="flex items-center gap-2 overflow-x-auto no-scrollbar">
   489→            {(
   490→              [
   491→                "ALL",
   492→                "GENERATED",
   493→                "PARTIALLY_PAID",
   494→                "PAID",
   495→                "OVERDUE",
   496→                "VOID",
   497→                ...(isAdmin ? ["DELETED" as const] : []),
   498→              ] as const
   499→            ).map((s) => {
   500→              const active = statusFilter === s;
   501→              const label = s === "DELETED" ? "Deletion Queue" : s === "ALL" ? "All" : BILL_STATUS_STYLES[s as BillStatus]?.label || s;
   502→              const badge = s === "DELETED" && deletedBills.length > 0 ? deletedBills.length : null;
   503→              return (
   504→                <button
   505→                  key={s}
   506→                  onClick={() => setStatusFilter(s)}
   507→                  className={cn(
   508→                    "inline-flex items-center gap-1 h-8 px-2.5 rounded-xl text-[11px] font-medium whitespace-nowrap transition-all",
   509→                    active
   510→                      ? "bg-primary text-primary-foreground shadow-md shadow-primary/30"
   511→                      : "glass-soft text-muted-foreground hover:text-foreground"
   512→                  )}
   513→                >
   514→                  {label}
   515→                  {badge !== null && (
   516→                    <span className={cn(
   517→                      "text-[9px] rounded-full px-1.5 py-0.5 leading-none font-bold min-w-[16px] text-center",
   518→                      active && s !== "DELETED"
   519→                        ? "bg-primary-foreground/20 text-primary-foreground"
   520→                        : active && s === "DELETED"
   521→                          ? "bg-primary-foreground/20 text-primary-foreground"
   522→                          : "bg-destructive text-white"
   523→                    )}>
   524→                      {badge}
   525→                    </span>
   526→                  )}
   527→                </button>
   528→              );
   529→            })}
   530→          </div>
   531→          {/* Action buttons below sorting bar */}
   532→          {isAdmin && statusFilter !== "DELETED" && bills.length > 0 && (
   533→            <div className="flex justify-end pt-1">
   534→              <button
   535→                onClick={() => setDeleteAllOpen(true)}
   536→                className="inline-flex items-center gap-1.5 text-[11px] font-medium text-destructive/70 hover:text-destructive transition-colors"
   537→              >
   538→                <Trash2 className="h-3 w-3" />
   539→                Delete All ({bills.length})
   540→              </button>
   541→            </div>
   542→          )}
   543→          {/* Restore All — only in deletion queue */}
   544→          {isAdmin && statusFilter === "DELETED" && deletedBills.length > 0 && (
   545→            <div className="flex justify-end pt-1">
   546→              <button
   547→                onClick={() => restoreAllMutation.mutate()}
   548→                disabled={restoreAllMutation.isPending}
   549→                className="inline-flex items-center gap-1.5 text-[11px] font-medium text-success/70 hover:text-success transition-colors"
   550→              >
   551→                <RotateCcw className="h-3 w-3" />
   552→                {restoreAllMutation.isPending ? "Restoring…" : `Restore All (${deletedBills.length})`}
   553→              </button>
   554→            </div>
   555→          )}
   556→        </div>
   557→      </StaggerItem>
   558→
   559→      {/* List */}
   560→      <StaggerItem>
   561→        {filtered.length === 0 ? (
   562→          <GlassCard className="p-10 md:p-16" hover={false}>
   563→            <div className="flex flex-col items-center justify-center text-center gap-3">
   564→              <div className="grid place-items-center h-14 w-14 rounded-3xl bg-muted/40">
   565→                <Receipt className="h-6 w-6 text-muted-foreground" />
   566→              </div>
   567→              <div>
   568→                <p className="font-medium">No bills found</p>
